+/** A fulfilled native call is not evidence that the selected bytes were saved.
+ * The caller retains the existing journal and cleanup ownership on any failure. */
+export async function writeVerifiedFile(uri: string, value: string, assertCurrent: () => void,
+  options?: FileSystem.WritingOptions): Promise<void> {
+  assertCurrent();
+  if (options) await FileSystem.writeAsStringAsync(uri, value, options);
+  else await FileSystem.writeAsStringAsync(uri, value);
+  assertCurrent();
+  const actual = await FileSystem.readAsStringAsync(uri, options?.encoding ? { encoding: options.encoding } : undefined);
+  assertCurrent();
+  if (actual !== value) throw new Error('FILE_WRITE_UNCONFIRMED');

+export type WebRuntimeStatus = 'checking' | 'admitted' | 'occupied' | 'unavailable';
+export type RuntimeLockPort = {
+  request(name: string, options: { mode: 'exclusive'; ifAvailable: true },
+    callback: (lock: unknown | null) => Promise<void>): Promise<unknown>;
+};
+export const PRIVATE_WEB_RUNTIME_LOCK = 'layerwell.private-runtime.v1';
+
+/** One application runtime per browser origin. Holding the lock for the
+ * document lifetime also covers direct cleanup/key/control-record operations
+ * which do not participate in privateKV's per-key mutation queues.
+ * Never steal or release on blur/navigation/component unmount. The browser
+ * releases the lock when this context ends, including a crash or reload. */
+export function createWebRuntimeAdmission(port: RuntimeLockPort | undefined) {
+  let status: WebRuntimeStatus = 'checking';
+  let pending = false;
+  const listeners = new Set<() => void>();
+  const publish = (next: WebRuntimeStatus) => {
+    status = next;
+    for (const listener of [...listeners]) {
+      try { listener(); } catch { /* A listener cannot release runtime ownership. */ }
+    }
+  };
+  const retry = () => {
+    if (pending || status === 'admitted') return;
+    if (!port) { publish('unavailable'); return; }
+    pending = true; publish('checking');
+    try {
+      void port.request(PRIVATE_WEB_RUNTIME_LOCK, { mode: 'exclusive', ifAvailable: true }, async lock => {
+        if (lock === null) { publish('occupied'); return; }
+        publish('admitted');
+        await new Promise<void>(() => { /* Deliberately retained until context destruction. */ });
+      }).then(() => { pending = false; if (status === 'admitted') publish('unavailable'); }, () => {
+        pending = false; publish('unavailable');
+      });
+    } catch { pending = false; publish('unavailable'); }
+  };
+  return { snapshot: () => status, retry, subscribe: (listener: () => void) => {
+    listeners.add(listener); return () => { listeners.delete(listener); };
+  } };
+}
+
+type Admission = ReturnType<typeof createWebRuntimeAdmission>;
+const runtimeGlobal = globalThis as typeof globalThis & { __layerwellPrivateWebRuntime?: Admission };
+// Preserve ownership during development Fast Refresh. No storage or private
+// content is placed on this singleton; it is only browser lock admission.
+export const webRuntimeAdmission = runtimeGlobal.__layerwellPrivateWebRuntime ??= createWebRuntimeAdmission(
+  (globalThis as unknown as { navigator?: { locks?: RuntimeLockPort } }).navigator?.locks,
+);
+
+/** auth-js initializes at module construction, before React mounts the root.
+ * Keep that initialization dormant until this document owns private storage.
+ * Refusal is never translated into an absent session or permission to erase. */
+export function gateWebRuntimeStorage(storage: {
+  getItem(key: string): Promise<string | null>;
+  setItem(key: string, value: string): Promise<void>;
+  removeItem(key: string): Promise<void>;
+}, admission = webRuntimeAdmission) {
+  const ready = () => new Promise<void>(resolve => {
+    if (admission.snapshot() === 'admitted') { resolve(); return; }
+    const unsubscribe = admission.subscribe(() => {
+      if (admission.snapshot() === 'admitted') { unsubscribe(); resolve(); }
+    });
+  });
+  return {
+    async getItem(key: string) { await ready(); return storage.getItem(key); },
+    async setItem(key: string, value: string) { await ready(); return storage.setItem(key, value); },
+    async removeItem(key: string) { await ready(); return storage.removeItem(key); },
+  };
+}
