+/** Retire the wait even if a transport or body reader ignores cancellation.
+ * Late resolution/rejection is consumed and cannot resume the retired caller. */
+export function untilAborted<T>(signal: AbortSignal, work: Promise<T>): Promise<T> {
+  return new Promise((resolve, reject) => {
+    const abort = () => reject(new Error('REQUEST_RETIRED'));
+    if (signal.aborted) abort();
+    else signal.addEventListener('abort', abort, { once: true });
+    work.then(value => {
+      if (!signal.aborted) resolve(value);
+    }, reject).finally(() => signal.removeEventListener('abort', abort));
+  });
+}

+      const result = await this.projection(token, record);
+      tx.beforeCommit(async () => { await this.resolve(tx, token); });
+      return result;
+    });
+  }

+  /** WorkspaceTransaction.authority must use this same-transaction resolver at
+   * initial AND final publication. A request-time cached session is insufficient. */
+  async resolve(tx: EncryptedTransaction, token: string): Promise<RestoredSession> {
+    if (!tokenValid(token)) return fail();
+    const digest = await digestToken(token);
+    const record = storedSession(await tx.read(`session/${digest}`), digest, tx.now());
+    const epoch = subject(await tx.read(`subject/${record.identity.subjectId}`), record.identity.subjectId);
+    if (epoch.epoch !== record.epoch || epoch.serial < record.serial) return fail();
+    let live: VerifiedIdentity | null;
+    try { live = await this.verifierRequired().current(record.identity.providerSessionId); }
+    catch { throw new ProfessionalAccessError('unavailable'); }
+    // No A→B→A adoption: the authoritative link revision must change on every
+    // replacement. A later MFA proof requires explicit reauthentication.
+    if (canonical(identity(live, tx.now())) !== canonical(record.identity)) return fail();
+    storedSession(record, digest, tx.now());
+    const result = await this.projection(token, record);
+    storedSession(record, digest, tx.now());
+    return result;
+  }
+
+  async restore(token: string, touch = false): Promise<RestoredSession> {
+    return this.custody.transaction(async tx => {
+      const initial = await this.resolve(tx, token);
+      if (touch) {
+        const digest = await digestToken(token);
+        const record = storedSession(await tx.read(`session/${digest}`), digest, tx.now());
+        await tx.write(`session/${digest}`, { ...record, lastActivityAt: tx.now() });
+      }
+      tx.beforeCommit(async () => { const final = await this.resolve(tx, token);
+        if (final.context !== initial.context) return fail(); });
+      // The API touch occurs once; later service rechecks never extend activity.
+      return touch ? this.resolve(tx, token) : initial;
+    });
+  }
+
+  private async retire(tx: EncryptedTransaction, token: string): Promise<void> {
+    const digest = await digestToken(token); const value = await tx.read(`session/${digest}`) as Session | null;
+    if (value !== null) {
+      if (value.digest !== digest || value.version !== 1 || typeof value.revoked !== 'boolean') return fail();
+      await tx.write(`session/${digest}`, { ...value, revoked: true });
+    }
+  }
+  async revoke(token: string): Promise<void> {
+    if (!tokenValid(token)) return fail();
+    await this.custody.transaction(tx => this.retire(tx, token));
+  }
+  async revokeAll(token: string): Promise<void> {
+    await this.custody.transaction(async tx => {
+      const restored = await this.resolve(tx, token);
+      const old = subject(await tx.read(`subject/${restored.session.subjectId}`), restored.session.subjectId);
+      if (!Number.isSafeInteger(old.epoch + 1)) return fail();
+      await tx.write(`subject/${old.subjectId}`, { ...old, epoch: old.epoch + 1 });
+    });
+  }
+  private async projection(token: string, record: Session): Promise<RestoredSession> {
+    const i = record.identity;
+    return { token, mode: i.method === 'owner_email_otp' ? 'owner' : 'staff',
+      context: await recordHash(['onskin-session-context-v1', record.digest, record.epoch, record.serial, i]),
+      session: { subjectId: i.subjectId, ownerId: i.ownerId, workspaceId: i.workspaceId, epoch: record.serial,
+        startedAt: i.authenticatedAt, lastActivityAt: record.lastActivityAt, stepUpAt: i.stepUpAt,
+        aal: i.aal, totp: i.totp, adult: i.adult, revoked: false } };
+  }
+}

+export function sessionCookie(token: string): string {
+  if (!tokenValid(token)) return fail();
+  return `${SESSION_COOKIE}=${token}; Path=/; Secure; HttpOnly; SameSite=Strict; Max-Age=28800`;
+}

+  if (matches.length !== 1) return null;
+  const token = matches[0]!.slice(SESSION_COOKIE.length + 1);
+  return tokenValid(token) ? token : null;

+import { ProfessionalAccessError } from './authority';
+import { canonical } from './grants';
+import type { WorkspaceTransaction } from './workspaceService';
+
+const READS = new Set<keyof WorkspaceTransaction>([
+  'authority', 'lifecycle', 'grants', 'billing', 'capacity', 'plans', 'receivedPlans',
+  'receipts', 'planSource', 'activationAvailable', 'reconciliation', 'reconciliationSource', 'previsitRecords',
+]);
+
+/** A repository read is a value, never permission to adopt a later mutable
+ * alias. Keep its original bytes as well as a private copy: cloning alone
+ * hides retirement during a subsequent awaited final check. This guard does
+ * not replace the repository's serial transaction/current-source contract. */
+export function snapshotWorkspaceTransaction(tx: WorkspaceTransaction): WorkspaceTransaction {
+  const retained: { value: unknown; fingerprint: string }[] = [];
+  function assertUnchanged() {
+    if (retained.some(read => canonical(read.value) !== read.fingerprint)) {
+      throw new ProfessionalAccessError('changed');
+    }
+  }
+  tx.beforeCommit(async () => assertUnchanged());
+  // The port may itself be frozen. An empty target avoids proxy invariants
+  // that would otherwise prohibit wrapping its non-writable methods.
+  return new Proxy({} as WorkspaceTransaction, {
+    get(_target, key) {
+      const method = Reflect.get(tx, key, tx);
+      if (key === 'beforeCommit') {
+        return (check: () => Promise<void>) => tx.beforeCommit(async () => {
+          assertUnchanged();
+          await check();
+          assertUnchanged();
+        });
+      }
+      if (READS.has(key as keyof WorkspaceTransaction) && typeof method === 'function') {
+        return async (...args: unknown[]) => {
+          const value: unknown = await Reflect.apply(method, tx, structuredClone(args));
+          assertUnchanged();
+          const captured = structuredClone(value);
+          retained.push({ value, fingerprint: canonical(captured) });
+          return captured;
+        };
+      }
+      return typeof method === 'function' ? method.bind(tx) : method;
+    },
+  });
+}

+  saveOwnerReceiptAndCopy(receipts: readonly PlanReceipt[], plan: IntendedPlan): Promise<void>;
+  beforeCommit(check: () => Promise<void>): void;
+  // Missing unconfigured ports refuse; no synthetic event or empty-log fallback.
+  reconciliation?(ownerId: string, workspaceId: string): Promise<ReconciliationLog>;
+  reconciliationSource?(ownerId: string, workspaceId: string): Promise<ReconciliationSource>;
+  saveReconciliation?(value: ReconciliationLog): Promise<void>;
+  previsitRecords?(ownerId: string, workspaceId: string, fields: readonly PrevisitRecordField[]): Promise<PrevisitRecordSource>;

+const unavailable = (): never => { throw new ProfessionalAccessError('unavailable'); };
+const changed = (): never => { throw new ProfessionalAccessError('changed'); };
+function admit(authority: Authority, request: AccessRequest, now: number) {
+  const denial = accessDenial(authority, request, now);
+  if (denial) throw new ProfessionalAccessError(denial);
+}
+const ownerRequest = (ownerId: string, operation: 'read' | 'write' | 'export'): AccessRequest => ({
+  action: 'personal_read_write_export_delete', ownerId, workspaceId: null, fields: [], purpose: null,
+  grantVersion: null, grantEpoch: null, operation,
+});
+async function liveLifecycle(tx: WorkspaceTransaction, ownerId: string): Promise<OwnerLifecycle> {
+  const lifecycle = structuredClone(await tx.lifecycle(ownerId));
+  if (lifecycle.ownerId !== ownerId || !Number.isSafeInteger(lifecycle.epoch) || lifecycle.epoch < 1 || lifecycle.deletionPending) changed();
+  return lifecycle;
+}
+async function ownerFence(tx: WorkspaceTransaction, token: string, ownerId: string, operation: 'read' | 'write' | 'export') {
+  const request = ownerRequest(ownerId, operation);
+  const initial = structuredClone(await tx.authority(token, request)); admit(initial, request, tx.now());
+  const lifecycle = await liveLifecycle(tx, ownerId);
+  const fingerprint = canonical([initial.session, lifecycle]);
+  const recheck = async () => {
+    const current = structuredClone(await tx.authority(token, request)); admit(current, request, tx.now());
+    const latest = await tx.lifecycle(ownerId);
+    if (canonical([current.session, latest]) !== fingerprint || latest.deletionPending) changed();
+  };
+  return { initial, recheck };
+}
+async function validatedReceivedCopies(tx: WorkspaceTransaction, ownerId: string) {
+  const plans = structuredClone(await tx.receivedPlans(ownerId)); const receipts = await decodePlanReceipts(await tx.receipts(ownerId), ownerId, tx.now());
+    if (plans.length > 1000 || new Set(plans.map(plan => plan.id)).size !== plans.length) changed();
+    for (const receipt of receipts) {
+      // A valid receipt without its exact retained copy is unavailable history,
+      // never an empty index or proof that a different copy was received.
+      if (!plans.some(plan => plan.id === receipt.planId && plan.contentHash === receipt.planHash && plan.createdAt <= receipt.recordedAt)) changed();
+    }
+  for (const plan of plans) {
+    await validateIntendedPlan(plan, tx.now());
+    if (plan.ownerId !== ownerId || !receipts.some(receipt => receipt.kind === 'received' && receipt.planId === plan.id && receipt.planHash === plan.contentHash)) changed();
+  }
+  return { plans, receipts };
+}
+function planDecisions(receipts: readonly PlanReceipt[], planId: string, planHash: string) {
+  return receipts.filter(receipt => receipt.planId === planId && receipt.planHash === planHash && receipt.kind !== 'received');
+}
+/** A new response follows the exact decision the owner reviewed. Exact retry
+ * checks its original predecessor, never borrows a later response. */
+function assertReviewedResponse(receipts: readonly PlanReceipt[], command: { receiptId: string; planId: string; planHash: string; expectedResponseHash: string | null }) {
+  if (command.expectedResponseHash !== null && !/^[a-f0-9]{64}$/u.test(command.expectedResponseHash)) changed();
+  const decisions = planDecisions(receipts, command.planId, command.planHash);
+  const existing = decisions.findIndex(receipt => receipt.id === command.receiptId);
+  const predecessor = existing < 0 ? decisions.at(-1) : decisions[existing - 1];
+  if ((predecessor?.contentHash ?? null) !== command.expectedResponseHash) changed();
+}
+async function currentBilling(tx: WorkspaceTransaction, workspaceId: string) {
+  return decodeWorkspaceBilling(await tx.billing(workspaceId), workspaceId);

+export async function publishWorkspacePlan(repository: WorkspaceRepository | null, token: string,
+  request: AccessRequest, command: IntendedPlanCommand, expectedSourceHash?: string): Promise<IntendedPlan> {
+  if (!repository) unavailable();
+  const captured = structuredClone({ request, command });
+  const r = captured.request;
+  if (r.action !== 'client_plan_issue_respond_export' || r.operation !== 'write' || !r.workspaceId || !r.ownerId ||
+      canonical(r.fields) !== canonical(['intended_plans']) || r.purpose !== 'routine_review') throw new ProfessionalAccessError('invalid_request');

+    admit(a, r, tx.now());
+    admit(a, productRead, tx.now());
+    const lifecycle = await liveLifecycle(tx, r.ownerId!);

+    const plans = await issueIntendedPlan(await tx.plans(r.ownerId!, r.workspaceId!), a, captured.command, tx.now(), source);
+    tx.beforeCommit(async () => {
+      const latestBilling = await currentBilling(tx, r.workspaceId!);
+      const current = { ...await tx.authority(token, r), billing: workspaceBillingAccess(latestBilling, tx.now()) };
+      admit(current, r, tx.now());
+      admit(current, productRead, tx.now());
+      if (canonical([current, await tx.planSource(r.ownerId!, r.workspaceId!), latestBilling, await tx.capacity(r.workspaceId!), await liveLifecycle(tx, r.ownerId!)]) !== initial ||
+          canonical(await tx.plans(r.ownerId!, r.workspaceId!)) !== canonical(plans)) changed();
+    });
+    await tx.savePlans(plans);
+    return plans.find(plan => plan.id === captured.command.id)!;
+  });

+/** New suggestions are disclosed only under the current owner-controlled text
+ * grant and health authority. A list or review is not a receipt or activation.
+ * The owner explicitly retains the exact copy through receiveWorkspacePlan. */
+export async function ownerIncomingPlans(repository: WorkspaceRepository | null, token: string, ownerId: string,
+  selection?: { planId: string; planHash: string }) {
+  if (!repository) unavailable();
+  const selected = selection ? Object.freeze({ ...selection }) : null;
+  return repository!.transaction({ workspaceId: null, ownerId }, async rawTx => {
+    const tx = snapshotWorkspaceTransaction(rawTx);
+    const owner = await ownerFence(tx, token, ownerId, 'read');
+    const grants = await decodeGrants(await tx.grants(ownerId), ownerId, tx.now()); const grant = grants.active;
+    const health = owner.initial.health;
+    if (!grant || grant.expiresAt <= tx.now() || !grant.fields.includes('intended_plans') || !grant.purposes.includes('routine_review') || !grant.permissions.includes('read') ||
+        !health?.active || health.deletionPending || health.ownerId !== ownerId || health.epoch !== grant.healthEpoch ||
+        health.consentReceiptId !== grant.consentReceiptId) throw new ProfessionalAccessError('grant');
+    const issued = await decodeIntendedPlans(await tx.plans(ownerId, grant!.workspaceId), ownerId, grant!.workspaceId, tx.now());
+    const retained = await validatedReceivedCopies(tx, ownerId);
+    if (issued.some(plan => retained.plans.some(copy => copy.id === plan.id && copy.contentHash !== plan.contentHash))) changed();
+    const incoming = issued.filter(plan => plan.grantId === grant!.id && plan.grantVersion === grant!.version && plan.grantEpoch === grant!.epoch &&
+      plan.healthEpoch === grant!.healthEpoch && !retained.plans.some(copy => copy.id === plan.id));
+    const chosen = selected ? incoming.find(plan => plan.id === selected.planId && plan.contentHash === selected.planHash) : null;
+    if (selected && !chosen) changed();
+    const authorityFingerprint = canonical([grants, health]);
+    const fingerprint = canonical([issued, retained]);
+    tx.beforeCommit(async () => {
+      await owner.recheck();
+      const current = await tx.authority(token, ownerRequest(ownerId, 'read'));
+      if (grant!.expiresAt <= tx.now() || canonical([await tx.grants(ownerId), current.health]) !== authorityFingerprint) changed();
+      if (canonical([await tx.plans(ownerId, grant!.workspaceId),
+        { plans: await tx.receivedPlans(ownerId), receipts: await tx.receipts(ownerId) }]) !== fingerprint) changed();
+    });
+    return { workspaceId: grant!.workspaceId,
+      plans: incoming.map(plan => ({ planId: plan.id, planHash: plan.contentHash, sequence: plan.sequence, issuedAt: plan.createdAt, latestResponse: null })),
+      text: chosen ? ['New intended suggestion — not yet received',
+        'Review only. No received copy, acknowledgement, applied routine, use or accepted-plan place has been recorded by this review.',
+        `Issued ${new Date(chosen.createdAt).toISOString()} · intended version ${chosen.sequence}`,
+        `Morning: ${chosen.am.map(step => step.name).join(' → ') || 'No steps recorded'}`,
+        `Evening: ${chosen.pm.map(step => step.name).join(' → ') || 'No steps recorded'}`].join('\n\n') : null };
+  });

+/** Acquire a received copy through the independently authenticated owner. A
+ * current text grant is required for this new disclosure; afterwards that
+ * owner's retained copy survives practice revocation and billing expiry. */
+export async function receiveWorkspacePlan(repository: WorkspaceRepository | null, token: string, ownerId: string,
+  workspaceId: string, command: { receiptId: string; planId: string; planHash: string; confirmed: true }): Promise<void> {
+  if (!repository) unavailable();
+  const c = Object.freeze({ ...command });
+  await repository!.transaction({ workspaceId, ownerId }, async rawTx => {
+    const tx = snapshotWorkspaceTransaction(rawTx);
+    const owner = await ownerFence(tx, token, ownerId, 'write');
+    const grants = await decodeGrants(await tx.grants(ownerId), ownerId, tx.now()); const grant = grants.active;
+    const health = owner.initial.health;
+    if (!grant || grant.workspaceId !== workspaceId || grant.expiresAt <= tx.now() || !grant.fields.includes('intended_plans') || !grant.purposes.includes('routine_review') ||
+        !health?.active || health.deletionPending || health.ownerId !== ownerId || health.epoch !== grant.healthEpoch ||
+        health.consentReceiptId !== grant.consentReceiptId) throw new ProfessionalAccessError('grant');
+    const plans = await decodeIntendedPlans(await tx.plans(ownerId, workspaceId), ownerId, workspaceId, tx.now());
+    const plan = plans.find(item => item.id === c.planId && item.contentHash === c.planHash && item.grantId === grant!.id &&
+      item.grantVersion === grant!.version && item.grantEpoch === grant!.epoch && item.healthEpoch === grant!.healthEpoch);
+    if (!plan) changed();
+    const retained = await validatedReceivedCopies(tx, ownerId);
+    if (retained.plans.some(copy => copy.id === plan!.id && copy.contentHash !== plan!.contentHash)) changed();
+    const expectedCopies = retained.plans.some(copy => copy.id === plan!.id) ? retained.plans : [...retained.plans, plan!];
+    const next = await appendPlanReceipt(retained.receipts, { ownerId, actorId: owner.initial.session.subjectId,
+      healthEpoch: grant!.healthEpoch, consentReceiptId: grant!.consentReceiptId, current: true, adult: true, deletionPending: false }, plan!,
+    { id: c.receiptId, kind: 'received', expectedPlanHash: c.planHash, confirmed: c.confirmed }, tx.now());
+    const authorityFingerprint = canonical([grants, health]);
+    const fingerprint = canonical(plans);
+    tx.beforeCommit(async () => {
+      await owner.recheck();
+      const current = await tx.authority(token, ownerRequest(ownerId, 'write'));
+      const latest = await tx.grants(ownerId);
+      if (canonical([latest, current.health]) !== authorityFingerprint || !latest.active || latest.active.expiresAt <= tx.now()) changed();
+      if (canonical(await tx.plans(ownerId, workspaceId)) !== fingerprint || canonical(await tx.receipts(ownerId)) !== canonical(next) ||
+          canonical(await tx.receivedPlans(ownerId)) !== canonical(expectedCopies)) changed();
+    });
+    await tx.saveOwnerReceiptAndCopy(next, plan!);
+  });

+/** Owner explicitly accepts an exact issued version. Acknowledgement and
+ * capacity become atomic; neither applies the owner's actual routine. */
+export async function acceptWorkspacePlan(repository: WorkspaceRepository | null, token: string,
+  ownerId: string, workspaceId: string, command: { receiptId: string; planId: string; planHash: string; capacityId: string; expectedCapacityRevision: number; expectedResponseHash: string | null; confirmed: true }): Promise<void> {
+  if (!repository) unavailable();
+  const c = Object.freeze({ ...command });
+  if (c.confirmed !== true) changed();
+  await repository!.transaction({ workspaceId, ownerId }, async rawTx => {
+    const tx = snapshotWorkspaceTransaction(rawTx);
+    const owner = await ownerFence(tx, token, ownerId, 'write');
+    const grants = await decodeGrants(await tx.grants(ownerId), ownerId, tx.now());
+    const active = grants.active;
+    if (!active ||
+        active.workspaceId !== workspaceId || active.expiresAt <= tx.now() || !active.permissions.includes('plan_write') || !active.fields.includes('intended_plans') || !active.purposes.includes('routine_review')) changed();
+    const health = owner.initial.health;
+    if (!health?.active || health.deletionPending || health.ownerId !== ownerId || health.epoch !== active!.healthEpoch ||
+        health.consentReceiptId !== active!.consentReceiptId) throw new ProfessionalAccessError('health');
+    const plans = await decodeIntendedPlans(await tx.plans(ownerId, workspaceId), ownerId, workspaceId, tx.now());
+    const plan = plans.find(item => item.id === c.planId && item.contentHash === c.planHash);
+    if (!plan || plan.grantId !== active!.id || plan.grantEpoch !== active!.epoch || plan.grantVersion !== active!.version || plan.healthEpoch !== active!.healthEpoch) changed();
+    const billing = await currentBilling(tx, workspaceId);
+    if (workspaceBillingAccess(billing, tx.now()) !== 'write_allowed') throw new ProfessionalAccessError('billing');
+    if (!await tx.activationAvailable(workspaceId)) throw new ProfessionalAccessError('coverage');
+    const capacity = await decodeWorkspaceCapacity(await tx.capacity(workspaceId), workspaceId, tx.now());
+    const retained = await validatedReceivedCopies(tx, ownerId);
+    if (!retained.plans.some(copy => copy.id === plan!.id && copy.contentHash === plan!.contentHash)) changed();
+    const received = retained.receipts;
+    assertReviewedResponse(received, c);
+    const nextReceipts = await appendPlanReceipt(received, { ownerId, actorId: owner.initial.session.subjectId,
+      healthEpoch: active!.healthEpoch, consentReceiptId: active!.consentReceiptId, current: true, adult: true, deletionPending: false }, plan!, {
+      id: c.receiptId, kind: 'acknowledged', expectedPlanHash: c.planHash, confirmed: true,
+    }, tx.now());
+    if (nextReceipts.filter(receipt => receipt.planId === c.planId && receipt.planHash === c.planHash && receipt.kind !== 'received').at(-1)?.id !== c.receiptId) changed();
+    const tier = billing.receipts.at(-1)!.tier;
+    const reservation = capacity.active.find(item => item.ownerId === ownerId);
+    // Only a validated fresh reservation can report capacity exhaustion. Stale
+    // revisions, reused IDs, mismatched grants and corrupt history stay unknown.
+    if (/^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/u.test(c.capacityId) &&
+        c.expectedCapacityRevision === capacity.revision && !capacity.events.some(event => event.id === c.capacityId) &&
+        (!reservation || (reservation.grantId === active!.id && reservation.grantVersion === active!.version && reservation.grantEpoch === active!.epoch)) &&
+        ((!reservation && capacity.active.length >= WORKSPACE_LIMITS[tier].plans) ||
+          (capacity.active.length > WORKSPACE_LIMITS[tier].plans && !capacityPublicationAllowed(capacity, tier, ownerId)))) {
+      throw new ProfessionalAccessError('capacity');
+    }
+    const intendedReservation = { ownerId, planId: plan!.id, planHash: plan!.contentHash,
+      grantId: active!.id, grantVersion: active!.version, grantEpoch: active!.epoch };
+    const nextCapacity = await transitionWorkspaceCapacity(capacity, { id: c.capacityId, expectedRevision: c.expectedCapacityRevision,
+      kind: 'accept', selectedOwnerIds: null, reservation: intendedReservation }, owner.initial.session.subjectId, billing.receipts.at(-1)!.tier, tx.now());
+    // The retained command proves a historical acceptance, not that it still
+    // owns a place after a newer acceptance or staff release. Keep the receipts
+    // intact while refusing a misleading current-success response.
+    if (canonical(nextCapacity.active.find(item => item.ownerId === ownerId) ?? null) !== canonical(intendedReservation) ||
+        nextCapacity.events.filter(event => event.reservation?.ownerId === ownerId).at(-1)?.id !== c.capacityId) changed();
+    const source = canonical([grants, owner.initial.health, billing]);
+    tx.beforeCommit(async () => {
+      await owner.recheck();
+      const current = await tx.authority(token, ownerRequest(ownerId, 'write'));
+      const latestBilling = await currentBilling(tx, workspaceId);
+      const currentGrant = await tx.grants(ownerId);
+      if (canonical([currentGrant, current.health, latestBilling]) !== source || !currentGrant.active || currentGrant.active.expiresAt <= tx.now() ||
+          workspaceBillingAccess(latestBilling, tx.now()) !== 'write_allowed' || !await tx.activationAvailable(workspaceId) ||
+          canonical(await tx.plans(ownerId, workspaceId)) !== canonical(plans) || canonical(await tx.capacity(workspaceId)) !== canonical(nextCapacity) ||
+          canonical(await tx.receipts(ownerId)) !== canonical(nextReceipts) ||
+          canonical(await tx.receivedPlans(ownerId)) !== canonical(retained.plans)) changed();
+    });
+    await tx.saveCapacity(nextCapacity);
+    await tx.saveOwnerReceiptAndCopy(nextReceipts, plan!);
+  });
+}

+/** Billing admin may select the active set after downgrade; practitioners and
+ * clients cannot select other clients. Release also remains possible read-only
+ * so cancellation cannot trap scarce capacity. Historical events stay intact. */
+export async function manageWorkspaceCapacity(repository: WorkspaceRepository | null, token: string,
+  workspaceId: string, command: CapacityCommand): Promise<void> {
+  if (!repository) unavailable();
+  const captured = structuredClone(command);
+  if (!['select', 'release'].includes(captured.kind)) throw new ProfessionalAccessError('invalid_request');
+  const r: AccessRequest = { action: 'workspace_billing_manage', workspaceId, ownerId: null, fields: [], purpose: null, grantVersion: null, grantEpoch: null, operation: 'write' };
+  await repository!.transaction({ workspaceId, ownerId: null }, async rawTx => {
+    const tx = snapshotWorkspaceTransaction(rawTx);
+    const a = structuredClone(await tx.authority(token, r)); admit(a, r, tx.now());
+    const capacity = await decodeWorkspaceCapacity(await tx.capacity(workspaceId), workspaceId, tx.now());
+    const billing = captured.kind === 'select' ? await currentBilling(tx, workspaceId) : null;
+    const tier = billing?.receipts.at(-1)?.tier ?? (captured.kind === 'release' ? capacity.events.at(-1)?.tier : null);
+    if (!tier) throw new ProfessionalAccessError('billing');
+    const next = await transitionWorkspaceCapacity(capacity, captured, a.session.subjectId, tier!, tx.now());
+    // Exact retries are successful only while their selected/released state
+    // still holds. A historical event cannot certify a superseding command.
+    if (captured.kind === 'release' && (next.active.some(item => item.ownerId === captured.reservation!.ownerId) ||
+          next.events.filter(event => event.reservation?.ownerId === captured.reservation!.ownerId).at(-1)?.id !== captured.id) ||
+        captured.kind === 'select' && (canonical(next.selected) !== canonical({ tier, ownerIds: captured.selectedOwnerIds }) ||
+          next.events.filter(event => event.kind === 'select').at(-1)?.id !== captured.id)) changed();
+    tx.beforeCommit(async () => {
+      const current = structuredClone(await tx.authority(token, r)); admit(current, r, tx.now());
+      if (canonical(current) !== canonical(a) || (billing && canonical(await tx.billing(workspaceId)) !== canonical(billing)) ||
+          canonical(await tx.capacity(workspaceId)) !== canonical(next)) changed();
+    });
+    await tx.saveCapacity(next);
+  });
+}

+export async function ownerReceivedPlanReport(repository: WorkspaceRepository | null, token: string, ownerId: string,
+  selection: readonly { planId: string; planHash: string }[], operation: 'read' | 'export'): Promise<string> {
