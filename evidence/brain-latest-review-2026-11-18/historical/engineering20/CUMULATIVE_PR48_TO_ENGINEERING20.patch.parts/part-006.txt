+  // available so Tab can wrap, and isolate the application at the body boundary.

+  if (branch?.parentElement?.tagName === 'BODY') {
+    for (const sibling of Array.from(branch.parentElement.children)) {
+      if (sibling !== branch && sibling instanceof HTMLElement) next.add(sibling);
+    }

+    });
+  };

+export type WebRuntimeStatus = 'checking' | 'admitted' | 'occupied' | 'unavailable';
+export type RuntimeLockPort = {
+  request(name: string, options: { mode: 'exclusive'; ifAvailable: true },
+    callback: (lock: unknown | null) => Promise<void>): Promise<unknown>;
+};
+export const PRIVATE_WEB_RUNTIME_LOCK = 'layerwell.private-runtime.v1';
+
+/** One application runtime per browser origin. Holding the lock for the
+ * document lifetime also covers direct cleanup/key/control-record operations
+ * which do not participate in privateKV's per-key mutation queues.
+ * Never steal or release on blur/navigation/component unmount. The browser
+ * releases the lock when this context ends, including a crash or reload. */
+export function createWebRuntimeAdmission(port: RuntimeLockPort | undefined) {
+  let status: WebRuntimeStatus = 'checking';
+  let pending = false;
+  const listeners = new Set<() => void>();
+  const publish = (next: WebRuntimeStatus) => {
+    status = next;
+    for (const listener of [...listeners]) {
+      try { listener(); } catch { /* A listener cannot release runtime ownership. */ }
+    }
+  };
+  const retry = () => {
+    if (pending || status === 'admitted') return;
+    if (!port) { publish('unavailable'); return; }
+    pending = true; publish('checking');
+    try {
+      void port.request(PRIVATE_WEB_RUNTIME_LOCK, { mode: 'exclusive', ifAvailable: true }, async lock => {
+        if (lock === null) { publish('occupied'); return; }
+        publish('admitted');
+        await new Promise<void>(() => { /* Deliberately retained until context destruction. */ });
+      }).then(() => { pending = false; if (status === 'admitted') publish('unavailable'); }, () => {
+        pending = false; publish('unavailable');
+      });
+    } catch { pending = false; publish('unavailable'); }
+  };
+  return { snapshot: () => status, retry, subscribe: (listener: () => void) => {
+    listeners.add(listener); return () => { listeners.delete(listener); };
+  } };
+}

+// content is placed on this singleton; it is only browser lock admission.
+export const webRuntimeAdmission = runtimeGlobal.__layerwellPrivateWebRuntime ??= createWebRuntimeAdmission(

+/** No browser persistence, health query cache, configurable remote origin or
+ * default account. Only a future trusted same-origin adapter can admit access. */
+export function createConnectionController(request: typeof fetch, render: (state: 'loading' | 'cleared' | 'unavailable') => void) {
+  let generation = 0; let pending: AbortController | null = null;
+  return {
+    clear() { generation += 1; pending?.abort(); pending = null; render('cleared'); },
+    async check() {
+      if (pending) return;
+      const admitted = ++generation; const controller = new AbortController(); pending = controller; render('loading');
+      try {
+        // Do not read a response body: this build has no admitted server data
+        // contract. A forged 200 or SPA fallback must never become success.
+        await request('/api/professional/status', { method: 'GET', cache: 'no-store', credentials: 'same-origin', redirect: 'error',
+          headers: { Accept: 'application/json' }, signal: controller.signal });
+      } catch { /* Service failure stays content-free and retryable. */ }
+      finally {
+        if (generation === admitted) { pending = null; render('unavailable'); }
+      }
+    },
+  };

+      (key.algorithm as AesKeyAlgorithm).length !== 256 || !key.usages.includes(usage)) fail();

+function hex(bytes: ArrayBuffer | Uint8Array): string { return Array.from(new Uint8Array(bytes), b => b.toString(16).padStart(2, '0')).join(''); }

+function jsonValue(value: unknown, ancestors = new Set<object>()): void {

+  if (!value || typeof value !== 'object' || ancestors.has(value) || ancestors.size >= 128) fail();

+  } else if (Object.getPrototypeOf(object) !== Object.prototype && Object.getPrototypeOf(object) !== null) fail();

+    if (Array.isArray(object) && key === 'length') continue;
+    const descriptor = Object.getOwnPropertyDescriptor(object, key)!;
+    if (typeof key !== 'string' || !descriptor.enumerable || !Object.hasOwn(descriptor, 'value')) fail();

+/** Authenticated envelopes bind record identity and retained key version. A
+ * missing/corrupt value stays unavailable; only explicit provisioning can
+ * create domain-empty aggregates. All mutations are read back before handoff. */
+export class EncryptedCustody {
+  constructor(private readonly driver: CiphertextCustody | null, private readonly keys: CustodyKeys | null) {}
+
+  async transaction<T>(work: (tx: EncryptedTransaction) => Promise<T>): Promise<T> {
+    const driver = this.driver; const keys = this.keys;
+    if (!driver || !keys) return fail();
+    let close = () => {}; let checked = false;
+    try {
+      const result = await driver.transaction(async raw => {
+        let open = true; let writable = true;
+        close = () => { open = false; };
+        const checkOpen = () => { if (!open) fail(); };
+        const originals = new Map<string, string | null>();
+        const values = new Map<string, unknown>();
+        const loading = new Map<string, Promise<unknown | null>>();
+        const dirty = new Set<string>(); const checks: (() => Promise<void>)[] = [];
+        const tx: EncryptedTransaction = {
+          now: () => { checkOpen(); const now = raw.now(); if (!Number.isSafeInteger(now) || now < 0) fail(); return now; },
+          read: async key => {
+            checkOpen(); keyValid(key);
+            if (values.has(key)) return structuredClone(values.get(key)!);
+            const pending = loading.get(key);
+            if (pending) { const result = await pending; checkOpen(); return structuredClone(result); }
+            const load = async () => {
+              const stored = await raw.read(key); checkOpen(); originals.set(key, stored);
+              if (stored === null) { values.set(key, null); return null; }
+              if (stored.length > MAX_BYTES * 2 + 1024) return fail();
+              const envelope = JSON.parse(stored) as { version: number; keyId: string; iv: string; ciphertext: string };
+              if (!envelope || Object.keys(envelope).sort().join(',') !== 'ciphertext,iv,keyId,version' || envelope.version !== 1 ||
+                  typeof envelope.keyId !== 'string' || !/^[A-Za-z0-9:_-]{1,128}$/u.test(envelope.keyId) ||
+                  typeof envelope.iv !== 'string' || envelope.iv.length !== 24 || typeof envelope.ciphertext !== 'string' ||
+                  envelope.ciphertext.length < 32) return fail();
+              const keyObject = aesKey(await keys.retained(key, envelope.keyId), 'decrypt');
+              const bytes = await crypto.subtle.decrypt({ name: 'AES-GCM', iv: unhex(envelope.iv), additionalData: aad(key, envelope.keyId) }, keyObject, unhex(envelope.ciphertext));
+              if (bytes.byteLength > MAX_BYTES) return fail();
+              const text = new TextDecoder('utf-8', { fatal: true }).decode(bytes);
+              const value: unknown = JSON.parse(text);
+              if (value === null || canonical(value) !== text) return fail();
+              checkOpen();
+              values.set(key, value); return value;
+            };
+            const operation = load(); loading.set(key, operation);
+            const result = await operation; checkOpen(); return structuredClone(result);
+          },
+          write: async (key, value) => {
+            checkOpen(); if (!writable) fail(); keyValid(key);
+            // Capture before any await so caller mutation cannot alter a command.
+            jsonValue(value);
+            const snapshot: unknown = structuredClone(value);
+            if (snapshot === null || snapshot === undefined || encoder.encode(canonical(snapshot)).byteLength > MAX_BYTES) fail();
+            await tx.read(key);
+            checkOpen(); if (!writable) fail();
+            values.set(key, snapshot); dirty.add(key);
+          },
+          beforeCommit: check => { checkOpen(); if (!writable) fail(); checks.push(check); },
+        };
+        try {
+          const result = await work(tx);
+          writable = false;
+          for (const key of dirty) {
+            // Key adapters can rotate their current selection while retained
+            // lookup/encryption is pending. Bind AAD, encryption and envelope
+            // metadata to the same captured version instead of that live object.
+            const current = await keys.current(key);
+            const selected = current && { id: current.id, key: current.key };
+            if (!selected || typeof selected.id !== 'string' || !/^[A-Za-z0-9:_-]{1,128}$/u.test(selected.id)) fail();
+            const iv = crypto.getRandomValues(new Uint8Array(12));
+            const parameters = { name: 'AES-GCM', iv, additionalData: aad(key, selected.id) };
+            const plaintext = canonical(values.get(key));
+            const ciphertext = await crypto.subtle.encrypt(parameters, aesKey(selected.key, 'encrypt'), encoder.encode(plaintext));
+            // An interrupted key rotation must not acknowledge a record that
+            // the retained-version resolver cannot read after process restart.
+            const readback = await crypto.subtle.decrypt(parameters, aesKey(await keys.retained(key, selected.id), 'decrypt'), ciphertext);
+            if (new TextDecoder('utf-8', { fatal: true }).decode(readback) !== plaintext) fail();
+            const stored = JSON.stringify({ version: 1, keyId: selected.id, iv: hex(iv), ciphertext: hex(ciphertext) });
+            await raw.compareAndSet(key, originals.get(key)!, stored);
+            if (await raw.read(key) !== stored) fail();
+          }
+          raw.beforeCommit(async () => {
+            try { if (checked) fail(); for (const check of checks) await check(); checked = true; }
+            finally { open = false; }
+          });
+          return result;
+        } catch (error) { open = false; throw error; }
+      });
+      if (!checked) return fail();
+      return result;
+    } catch (error) {
+      if (error instanceof ProfessionalAccessError) throw error;
+      return fail(); // Provider bodies, ciphertext, keys and private text never escape.
+    } finally { close(); }
+  }
