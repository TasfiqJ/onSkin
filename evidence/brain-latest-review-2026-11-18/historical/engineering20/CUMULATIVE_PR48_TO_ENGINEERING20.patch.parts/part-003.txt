+import type { PlanHistoryStep45 } from './planHistory45';

+type Sequence = Readonly<{ am: readonly PlanHistoryStep45[]; pm: readonly PlanHistoryStep45[] }>;
+/** Free factual comparison of exact bottle identity, names, phase and relative
+ * order. No formula equivalence, use, compatibility or advanced advice inferred. */
+export function sequenceComparisonLines(previous: Sequence | null, reviewed: Sequence): readonly string[] {
+  if (!previous) return ['No previous owner-applied sequence is available for comparison. This will be the first applied version.'];
+  const lines: string[] = [];
+  for (const phase of ['am', 'pm'] as const) {
+    const label = phase === 'am' ? 'Morning' : 'Evening';
+    const before = previous[phase]; const after = reviewed[phase];
+    const oldIds = new Set(before.map(step => step.productId)); const newIds = new Set(after.map(step => step.productId));
+    const added = after.filter(step => !oldIds.has(step.productId)); const removed = before.filter(step => !newIds.has(step.productId));
+    for (const step of added) lines.push(`${label}: add ${step.name} to this phase.`);
+    for (const step of removed) lines.push(`${label}: remove ${step.name} from this phase. Earlier versions and use entries stay unchanged.`);
+    for (const step of after) {
+      const prior = before.find(item => item.productId === step.productId);
