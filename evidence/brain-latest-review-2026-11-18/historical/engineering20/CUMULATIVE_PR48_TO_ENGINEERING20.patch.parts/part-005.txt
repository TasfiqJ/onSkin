+import { parseIngredientText } from '@/features/catalog/ingredientParser';
+
+import {
+  validateOwnedBottleHistory38,
+  type OwnedBottleHistorySnapshot38,
+} from './ownedBottleVersion38';
+
+/** IDs authenticate owner observations, never a manufacturer's formulation identity. */
+export function compareRecordedFormulaVersions(
+  first: OwnedBottleHistorySnapshot38,
+  second: OwnedBottleHistorySnapshot38,
+  firstVersionId: string | null,
+  secondVersionId: string | null,
+  ownerUserId: string,
+) {
+  try {
+    for (const snapshot of [first, second]) {
+      if (snapshot.ownedUnitId !== snapshot.productId || !snapshot.currentVersionId) return null;
+      validateOwnedBottleHistory38({
+        productId: snapshot.productId,
+        ownerUserId,
+        revision: snapshot.revision,
+        currentVersionId: snapshot.currentVersionId,
+        versions: snapshot.versions,
+      });
+    }
+  } catch {
+    return null;
+  }
+  const left = first.versions.find((version) => version.id === firstVersionId);
+  const right = second.versions.find((version) => version.id === secondVersionId);
+  if (!left || !right) return null;
+  const confirmed = [left, right].every(
+    (version) =>
+      version.formulaVersion.labelStatus === 'user_confirmed_label' &&
+      version.basis === 'label_in_hand',
+  );
+  const entries = (text: string | null) => {
+    const result = new Map<string, string>();
+    for (const token of parseIngredientText(text ?? '').tokens) {
+      // Preserve section qualifiers; an optional "may contain" entry is not a main ingredient.
+      const display = `${token.section === 'main' ? '' : `${token.section}: `}${token.rawToken}`;
+      const key = display.trim().replace(/\s+/gu, ' ').toLowerCase();
+      if (key && !result.has(key)) result.set(key, display);
+    }
+    return result;
+  };
+  const leftEntries = confirmed
+    ? entries(left.formulaVersion.labelText)
+    : new Map<string, string>();
+  const rightEntries = confirmed
+    ? entries(right.formulaVersion.labelText)
+    : new Map<string, string>();
+  const comparable = confirmed && leftEntries.size > 0 && rightEntries.size > 0;
+  return {
+    left,
+    right,
+    sameRecord: left.id === right.id,
+    manufacturerFormulaDifference: 'UNCONFIRMED' as const,
+    onlyFirst: comparable
+      ? [...leftEntries].filter(([key]) => !rightEntries.has(key)).map(([, text]) => text)
+      : null,
+    onlySecond: comparable
+      ? [...rightEntries].filter(([key]) => !leftEntries.has(key)).map(([, text]) => text)
+      : null,
+  };
+}

+import { categoryLabel } from './categories';
+import { validLocalDate } from './freshness';
+import type { ShelfProduct } from './store';
+
+export type ComparableProduct = Pick<
+  ShelfProduct,
+  | 'id'
+  | 'name'
+  | 'brand'
+  | 'category'
+  | 'status'
+  | 'isOpened'
+  | 'openedAt'
+  | 'paoMonths'
+  | 'paoSource'
+  | 'expiryDate'
+  | 'expirySource'
+  | 'addedVia'
+  | 'ingredients'
+>;
+
+function facts(product: ComparableProduct): Record<string, string> {
+  return {
+    Brand: product.brand || 'Not recorded',
+    Category: categoryLabel(product.category) ?? 'Not recorded',
+    Status: { active: 'Active', finished: 'Finished', discarded: 'Discarded' }[product.status],
+    'Opened date': !product.isOpened ? 'Unopened' : (validLocalDate(product.openedAt) ?? 'Unknown'),
+    'Printed expiry':
+      product.expirySource === 'printed'
+        ? (validLocalDate(product.expiryDate) ?? 'Not recorded')
+        : 'Not recorded',
+    'Period after opening':
+      product.paoMonths !== null &&
+      (product.paoSource === 'label' || product.paoSource === 'catalog')
+        ? `${product.paoMonths} months · ${product.paoSource}`
+        : 'Not recorded',
+    'Added through': {
+      manual: 'Manual entry',
+      onboarding: 'Onboarding',
+      barcode: 'Barcode',
+      search: 'Catalog search',
+      ocr: 'Label text',
+    }[product.addedVia],
+  };
+}
+
+/** Compare recorded text only. No synonym inference, completeness claim,
+ * compatibility, clinical ranking, quota, persistence or provider lookup. */
+export function compareSavedProducts<T extends ComparableProduct>(
+  products: readonly T[],
+  leftId: string | null,
+  rightId: string | null,
+) {
+  if (!leftId || !rightId || leftId === rightId) return null;
+  const left = products.find((product) => product.id === leftId);
+  const right = products.find((product) => product.id === rightId);
+  if (!left || !right) return null;
+  const leftFacts = facts(left);
+  const rightFacts = facts(right);
+  const ingredientKey = (text: string) => text.trim().replace(/\s+/gu, ' ').toLowerCase();
+  const leftIngredients = new Map<string, string>();
+  for (const text of left.ingredients) {
+    const key = ingredientKey(text);
+    if (key && !leftIngredients.has(key)) leftIngredients.set(key, text.trim());
+  }
+  const rightIngredients = new Set(
+    right.ingredients.filter((text) => text.trim()).map(ingredientKey),
+  );
+  // Keep the first recorded spelling as well as the source's order.
+  const sharedIngredients =
+    leftIngredients.size && rightIngredients.size
+      ? [...leftIngredients].filter(([key]) => rightIngredients.has(key)).map(([, text]) => text)
+      : null;
+  return {
+    left,
+    right,
+    sharedIngredients,
+    rows: Object.entries(leftFacts).map(([label, value]) => ({
+      label,
+      left: value,
+      right: rightFacts[label]!,
+    })),
+  };
+}

+  lease: HealthDataWriteLease;
+  diaryAccess?: DiaryBriefAccess;
+}>;
+
+/** Text-only native sharing creates no plaintext cache file or second storage system.
+ * Once explicitly handed to another app, a recipient's copy cannot be revoked. */
+export async function shareReviewedAppointmentBrief(
+  review: ReviewedAppointmentBrief,
+  assertReviewCurrent: () => void,
+): Promise<boolean> {
+  if (review.selection.includePhoto) throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+  const assertCurrent = () => {
+    assertOriginalHealthDataWriteLease(review.lease);
+    assertReviewCurrent();
+    if (review.selection.recordIds?.some((id) => id.startsWith('diary:'))) {
+      if (!review.diaryAccess) throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+      review.diaryAccess.assertCurrent();
+    }
+  };
+  assertCurrent();
+  return runHealthDataOperation(review.lease.ownerUserId, async (lease) => {
+    assertCurrent();
+    const proof = await revalidateAppointmentBrief(review, assertCurrent);
+    lease.assertCurrent(); assertCurrent();
+    const result = await dispatchWithBriefSourceProof(proof, review.lease, assertCurrent, () => Share.share({
+      title: 'My factual appointment brief',
+      message: review.text,
+    }));
+    lease.assertCurrent();
+    assertCurrent();
+    return result.action === Share.sharedAction;
+  });
+}

+  assertCurrent();
+  const proof = await captureBriefSourceProof(review.lease, review.selection.recordIds, review.selection.includePhoto === true);

+    }
+  } catch (error) {
+    assertCurrent();
+    if (error instanceof Error && error.message === 'APPOINTMENT_BRIEF_INVALID_OR_STALE') throw error;
+    throw new Error('APPOINTMENT_BRIEF_SOURCE_UNAVAILABLE');
+  }
+  assertCurrent();
+  return proof;
+}

+/** A correction/retraction refers to an existing chain, never overwrites it. */
+export function observationHistory(events: readonly OptionalChangeEvent41[]) {
+  const roots = new Map<string, string>();
+  const latest = new Map<string, OptionalChangeEvent41>();
+  for (const event of events) {
+    const root = event.targetEventId === null ? event.id : roots.get(event.targetEventId);
+    if (!root || roots.has(event.id)) throw new Error('OBSERVATION_HISTORY_INVALID');
+    roots.set(event.id, root);
+    latest.set(root, event);
+  }
+  const current = events.filter(
+    (event) => latest.get(roots.get(event.id)!) === event && event.kind !== 'retraction',
+  );
+  const status = new Map(
+    events.map((event) => {
+      const last = latest.get(roots.get(event.id)!)!;
+      return [
+        event.id,
+        event.kind === 'retraction'
+          ? 'Retraction record'
+          : last.kind === 'retraction'
+            ? 'Retracted observation'
+            : last.id === event.id
+              ? 'Current observation'
+              : 'Earlier observation',
+      ] as const;
+    }),
+  );
