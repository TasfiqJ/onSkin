+export function createReconciliationController(request: typeof fetch, onChange: (state: QueueViewState) => void, newId: () => string = () => crypto.randomUUID()) {
+  let state: QueueViewState = { kind: 'idle' }; let generation = 0; let abort: AbortController | null = null;
+  const show = (next: QueueViewState) => { state = next; onChange(next); };
+  async function post(input: unknown, operation: number) {
+    const control = new AbortController(); abort?.abort(); abort = control;
+    const timer = setTimeout(() => control.abort(), 12_000);
+    try {
+      const response = await untilAborted(control.signal, request('/api/professional/reconciliation', { method: 'POST', credentials: 'same-origin', cache: 'no-store', redirect: 'error',
+        headers: { 'Content-Type': 'application/json', 'X-OnSkin-Request': 'reconciliation-v1' }, body: JSON.stringify(input), signal: control.signal }));
+      if (generation !== operation || control.signal.aborted) return fail();
+      if (response.status === 409) throw new Error('QUEUE_CHANGED');
+      if ([400, 401, 403].includes(response.status)) throw new Error('QUEUE_DENIED');
+      if (!response.ok || !response.headers.get('Cache-Control')?.toLowerCase().split(',').some(v => v.trim() === 'no-store') || !response.headers.get('Content-Type')?.startsWith('application/json')) return fail();
+      const raw = await untilAborted(control.signal, response.text()); if (raw.length > 3_000_000) return fail();
+      const result = JSON.parse(raw); if (!result || Object.keys(result).sort().join(',') !== 'context,kind,model' || !safe(result.context)) return fail();
+      const model = await decodeModel(result.model); if (generation !== operation || control.signal.aborted) return fail();
+      return { ...result, model } as { kind: string; context: string; model: ReconciliationModel };
+    } finally { clearTimeout(timer); if (abort === control) abort = null; }
+  }
+  const api = {
+    get state() { return state; }, get generation() { return generation; },
+    clear() { generation++; abort?.abort(); abort = null; show({ kind: 'idle' }); },
+    async load(admitted: number) {
+      if (admitted !== generation || state.kind === 'saving' || state.kind === 'loading') return;
+      generation++; const op = generation; show({ kind: 'loading' });
+      try { const result = await post({ kind: 'source', context: null }, op); if (generation !== op) return;
+        if (result.kind !== 'source') return fail(); generation++; show({ kind: 'source', scope: { context: result.context, model: result.model } });
+      } catch (error) { if (generation === op) { generation++; show({ kind: error instanceof Error && error.message === 'QUEUE_CHANGED' ? 'changed' : 'unavailable' }); } }
+    },
+    filter(admitted: number, filter: QueueFilter) {
+      if (admitted !== generation || !['source', 'saved'].includes(state.kind) || !('scope' in state) ||
+          !['all', 'open', ...Object.keys(QUEUE_TRANSITIONS)].includes(filter.state) || !['all', ...MATERIAL_EVENTS].includes(filter.kind) ||
+          Object.keys(filter).sort().join(',') !== 'kind,state') return;
+      generation++; show({ ...state, scope: { ...state.scope, filter: Object.freeze({ ...filter }) } });
+    },
+    reviewEvent(admitted: number, eventId: string, eventVersion: number) {
+      if (admitted !== generation || !['source', 'saved'].includes(state.kind) || !('scope' in state) || !state.scope.model.writeAvailable) return;
+      const scope = state.scope;
+      const event = scope.model.events.find(e => e.id === eventId && e.version === eventVersion &&
+        (!scope.filter || scope.filter.kind === 'all' || e.kind === scope.filter.kind)); if (!event) return;
+      generation++; show({ kind: 'review', scope: state.scope, command: { kind: 'enqueue', sourceHash: state.scope.model.sourceHash,
+        eventId, eventVersion, eventHash: event.contentHash, caseId: newId(), confirmed: true } });
+    },
+    edit(admitted: number, caseId: string) {
+      if (admitted !== generation || !['source', 'saved'].includes(state.kind) || !('scope' in state)) return;
+      const row = filterCurrentCases(state.scope.model.rows, state.scope.filter ?? allRecords).find(row => row.caseId === caseId); if (!row) return;
+      generation++; show({ kind: 'editing', scope: state.scope, row });
+    },
+    reviewTransition(admitted: number, fields: Omit<Extract<QueueCommand, { kind: 'transition' }>, 'kind' | 'sourceHash' | 'caseId' | 'expectedHash' | 'confirmed'>): boolean {
+      if (admitted !== generation || state.kind !== 'editing' || !state.scope.model.writeAvailable || !QUEUE_TRANSITIONS[state.row.state].includes(fields.next) || !safe(fields.reason) ||
+          (['resolved', 'declined_referred', 'relationship_closed'].includes(fields.next) && (!safe(fields.communication) ||
+            ((fields.changedPlanHash ?? state.row.changedPlanHash) !== null && fields.acknowledgementId === null && !safe(fields.nonAcknowledgementReason))))) return false;
+      const effectivePlanHash = fields.changedPlanHash ?? state.row.changedPlanHash;
+      if (fields.acknowledgementId !== null && !state.scope.model.acknowledgements.some(a => a.id === fields.acknowledgementId && a.contentHash === fields.acknowledgementHash && a.planHash === effectivePlanHash)) return false;
+      const command: QueueCommand = structuredClone({ kind: 'transition', sourceHash: state.scope.model.sourceHash, caseId: state.row.caseId, expectedHash: state.row.contentHash, confirmed: true, ...fields });
+      generation++; show({ kind: 'review', scope: state.scope, command }); return true;
+    },
+    back(admitted: number) {
+      if (admitted !== generation || !['review', 'editing'].includes(state.kind) || !('scope' in state)) return;
+      generation++; show({ kind: 'source', scope: state.scope });
+    },
+    async confirm(admitted: number) {
+      if (admitted !== generation || !['review', 'unknown'].includes(state.kind) || !('command' in state)) return;
+      const { scope, command } = state; generation++; const op = generation; show({ kind: 'saving', scope, command });
+      try { const result = await post({ kind: 'save', context: scope.context, grantVersion: scope.model.grantVersion, grantEpoch: scope.model.grantEpoch, command }, op);
+        if (generation !== op) return;
+        if (result.kind !== 'save' || result.context !== scope.context || result.model.sourceHash !== command.sourceHash ||
+            result.model.grantVersion !== scope.model.grantVersion || result.model.grantEpoch !== scope.model.grantEpoch) return fail();
+        if (command.kind === 'enqueue' ? !result.model.rows.some(row => row.events.some(e => e.id === command.eventId && e.version === command.eventVersion && e.contentHash === command.eventHash)) :
+          ![...result.model.rows].reverse().find(row => row.caseId === command.caseId && row.previousHash === command.expectedHash && row.state === command.next && row.reason === command.reason && row.communication === command.communication)) return fail();
+        generation++; show({ kind: 'saved', scope: { ...scope, context: result.context, model: result.model } });
+      } catch (error) { if (generation !== op) return; generation++;
+        show(error instanceof Error && error.message === 'QUEUE_CHANGED' ? { kind: 'changed' } : error instanceof Error && error.message === 'QUEUE_DENIED' ? { kind: 'unavailable' } : { kind: 'unknown', scope, command });
+      }
+    },
+  }; return api;

+/** Retire the wait even if a transport or body reader ignores cancellation.
+ * Late resolution/rejection is consumed and cannot resume the retired caller. */
+export function untilAborted<T>(signal: AbortSignal, work: Promise<T>): Promise<T> {
+  return new Promise((resolve, reject) => {
+    const abort = () => reject(new Error('REQUEST_RETIRED'));
+    if (signal.aborted) abort();
+    else signal.addEventListener('abort', abort, { once: true });
+    work.then(value => {
+      if (!signal.aborted) resolve(value);
+    }, reject).finally(() => signal.removeEventListener('abort', abort));
+  });

+    if (typeof proof !== 'string' || !proof || proof.length > 8192) return fail();
+    const verifier = this.verifierRequired();
+    let verified: VerifiedIdentity | null;
+    try { const result = await verifier.exchange(proof); verified = result ? structuredClone(result) : null; }
+    catch { throw new ProfessionalAccessError('unavailable'); }
+    const token = Array.from(crypto.getRandomValues(new Uint8Array(32)), byte => byte.toString(16).padStart(2, '0')).join('');
+    const digest = await digestToken(token);
+    return this.custody.transaction(async tx => {
+      const current = identity(verified, tx.now());
+      if (tx.now() - current.authenticatedAt >= 10 * 60_000) return fail();
+      // Verified account provisioning owns this monotonic record. Missing
+      // revocation history cannot be guessed as a first login after restore.
+      const old = subject(await tx.read(`subject/${current.subjectId}`), current.subjectId);
+      if (!Number.isSafeInteger(old.serial + 1)) return fail();
+      const next: Subject = { ...old, serial: old.serial + 1 };
+      // Re-authentication replaces only this bearer. Other devices remain valid
+      // until explicit all-session revocation or a verified binding retirement.
+      if (replaceToken && tokenValid(replaceToken)) await this.retire(tx, replaceToken);
+      if (await tx.read(`session/${digest}`) !== null) return fail();
+      await tx.write(`subject/${current.subjectId}`, next);
+      const record: Session = { version: 1, digest, identity: current, epoch: next.epoch, serial: next.serial,
+        lastActivityAt: tx.now(), revoked: false };
+      await tx.write(`session/${digest}`, record);
+      const result = await this.projection(token, record);
+      tx.beforeCommit(async () => { await this.resolve(tx, token); });
+      return result;
+    });
+  }
+
+  /** WorkspaceTransaction.authority must use this same-transaction resolver at
+   * initial AND final publication. A request-time cached session is insufficient. */
+  async resolve(tx: EncryptedTransaction, token: string): Promise<RestoredSession> {
+    if (!tokenValid(token)) return fail();
+    const digest = await digestToken(token);
+    const record = storedSession(await tx.read(`session/${digest}`), digest, tx.now());
+    const epoch = subject(await tx.read(`subject/${record.identity.subjectId}`), record.identity.subjectId);
+    if (epoch.epoch !== record.epoch || epoch.serial < record.serial) return fail();
+    let live: VerifiedIdentity | null;
+    try { live = await this.verifierRequired().current(record.identity.providerSessionId); }
+    catch { throw new ProfessionalAccessError('unavailable'); }
+    // No A→B→A adoption: the authoritative link revision must change on every
+    // replacement. A later MFA proof requires explicit reauthentication.
+    if (canonical(identity(live, tx.now())) !== canonical(record.identity)) return fail();
+    storedSession(record, digest, tx.now());
+    const result = await this.projection(token, record);
+    storedSession(record, digest, tx.now());
+    return result;
+  }
+
+  async restore(token: string, touch = false): Promise<RestoredSession> {
+    return this.custody.transaction(async tx => {
+      const initial = await this.resolve(tx, token);
+      if (touch) {
+        const digest = await digestToken(token);
+        const record = storedSession(await tx.read(`session/${digest}`), digest, tx.now());
+        await tx.write(`session/${digest}`, { ...record, lastActivityAt: tx.now() });
+      }
+      tx.beforeCommit(async () => { const final = await this.resolve(tx, token);
+        if (final.context !== initial.context) return fail(); });
+      // The API touch occurs once; later service rechecks never extend activity.
+      return touch ? this.resolve(tx, token) : initial;
+    });
+  }
+
+  private async retire(tx: EncryptedTransaction, token: string): Promise<void> {
+    const digest = await digestToken(token); const value = await tx.read(`session/${digest}`) as Session | null;
+    if (value !== null) {
+      if (value.digest !== digest || value.version !== 1 || typeof value.revoked !== 'boolean') return fail();
+      await tx.write(`session/${digest}`, { ...value, revoked: true });
+    }
+  }
+  async revoke(token: string): Promise<void> {
+    if (!tokenValid(token)) return fail();
+    await this.custody.transaction(tx => this.retire(tx, token));
+  }
+  async revokeAll(token: string): Promise<void> {
+    await this.custody.transaction(async tx => {
+      const restored = await this.resolve(tx, token);
+      const old = subject(await tx.read(`subject/${restored.session.subjectId}`), restored.session.subjectId);
+      if (!Number.isSafeInteger(old.epoch + 1)) return fail();
+      await tx.write(`subject/${old.subjectId}`, { ...old, epoch: old.epoch + 1 });
+    });
+  }
+  private async projection(token: string, record: Session): Promise<RestoredSession> {
+    const i = record.identity;
+    return { token, mode: i.method === 'owner_email_otp' ? 'owner' : 'staff',
+      context: await recordHash(['onskin-session-context-v1', record.digest, record.epoch, record.serial, i]),
+      session: { subjectId: i.subjectId, ownerId: i.ownerId, workspaceId: i.workspaceId, epoch: record.serial,
+        startedAt: i.authenticatedAt, lastActivityAt: record.lastActivityAt, stepUpAt: i.stepUpAt,
+        aal: i.aal, totp: i.totp, adult: i.adult, revoked: false } };
+  }
+}

+export function sessionCookie(token: string): string {
+  if (!tokenValid(token)) return fail();
+  return `${SESSION_COOKIE}=${token}; Path=/; Secure; HttpOnly; SameSite=Strict; Max-Age=28800`;
+}

+  const matches = (request.headers.get('Cookie') ?? '').split(';').map(part => part.trim()).filter(part => part.split('=')[0] === SESSION_COOKIE);
+  if (matches.length !== 1) return null;
+  const token = matches[0]!.slice(SESSION_COOKIE.length + 1);
+  return tokenValid(token) ? token : null;
+}
