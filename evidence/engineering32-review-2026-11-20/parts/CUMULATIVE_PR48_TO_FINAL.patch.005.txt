+import type { PlanHistoryStep45 } from './planHistory45';
+
+type Sequence = Readonly<{ am: readonly PlanHistoryStep45[]; pm: readonly PlanHistoryStep45[] }>;
+/** Free factual comparison of exact bottle identity, names, phase and relative
+ * order. No formula equivalence, use, compatibility or advanced advice inferred. */
+export function sequenceComparisonLines(previous: Sequence | null, reviewed: Sequence): readonly string[] {
+  if (!previous) return ['No previous owner-applied sequence is available for comparison. This will be the first applied version.'];
+  const lines: string[] = [];
+  for (const phase of ['am', 'pm'] as const) {
+    const label = phase === 'am' ? 'Morning' : 'Evening';
+    const before = previous[phase]; const after = reviewed[phase];
+    const oldIds = new Set(before.map(step => step.productId)); const newIds = new Set(after.map(step => step.productId));
+    const added = after.filter(step => !oldIds.has(step.productId)); const removed = before.filter(step => !newIds.has(step.productId));
+    for (const step of added) lines.push(`${label}: add ${step.name} to this phase.`);
+    for (const step of removed) lines.push(`${label}: remove ${step.name} from this phase. Earlier versions and use entries stay unchanged.`);
+    for (const step of after) {
+      const prior = before.find(item => item.productId === step.productId);
+      if (prior && prior.name !== step.name) lines.push(`${label}: recorded name changes from ${prior.name} to ${step.name} for the same saved bottle.`);
+    }
+    const oldOrder = before.filter(step => newIds.has(step.productId)).map(step => step.productId);
+    const newOrder = after.filter(step => oldIds.has(step.productId)).map(step => step.productId);
+    const reordered = JSON.stringify(oldOrder) !== JSON.stringify(newOrder);
+    if (reordered) lines.push(`${label}: relative order of retained bottles changes.`);
+    if (!added.length && !removed.length && !reordered && after.every(step => before.find(item => item.productId === step.productId)?.name === step.name)) {
+      lines.push(`${label}: same saved bottles, recorded names and order.`);
+    }
+  }
+  lines.push('Names and sequence only. Bottle observations or other source facts may have changed even when these sequences match. This is not a formula, compatibility or use comparison.');
+  return Object.freeze(lines);
+}

+export function DataExportControls({ ownerUserId, source, disabled = false, onStart }: {

+
+type Confirmation = Readonly<{ scope: object; state: DataRightsState; generation: number }>;
+
+/** Exact rendered confirmation, original account and original consent view.
+ * Export has a separate flight so it can never block deletion priority. */
+export function useDataRightsConfirmation(ownerUserId: string | null, source?: unknown) {
+  const identity = captureAccountIdentityGeneration();
+  const health = useSyncExternalStore(subscribeActiveHealthProcessingLeaseChanges, activeHealthProcessingLeaseSnapshot, () => null);
+  const scope = useMemo(() => Object.freeze({ ownerUserId, identity, health, source }), [ownerUserId, identity, health, source]);
+  const liveScope = useRef<object | null>(scope);
+  const live = useRef<Confirmation | null>(null);
+  const [rendered, setRendered] = useState<Confirmation | null>(null);
+  useLayoutEffect(() => {
+    liveScope.current = scope;
+    return () => { if (liveScope.current === scope) { liveScope.current = null; live.current = null; } };
+  }, [scope]);
+  function current() {
+    try { assertAccountIdentityGeneration(identity); } catch { return false; }
+    return liveScope.current === scope && activeHealthProcessingLeaseSnapshot() === health;
+  }
+  function open(action: PendingDataRightsAction) {
+    if (!current() || live.current) return;
+    let generation: number | null = null;
+    // Capture only; this operation never retains a lease across confirmation.
+    void runAccountGenerationOperation((lease) => { generation = lease.generation; }).catch(() => {});
+    if (generation === null) return;
+    const confirmation = Object.freeze({ scope, state: beginDataRightsConfirmation(IDLE_DATA_RIGHTS_STATE, action)!, generation });
+    live.current = confirmation;
+    setRendered(confirmation);
+  }
+  function cancel(confirmation: Confirmation | null) {
+    if (!current() || !confirmation || live.current !== confirmation || confirmation.state.phase !== 'confirming') return;
+    live.current = null;
+    setRendered(null);
+  }
+  function begin(confirmation: Confirmation | null, action: PendingDataRightsAction) {
+    if (!current() || !confirmation || live.current !== confirmation || !ownerUserId) return null;
+    const running = beginDataRightsOperation(confirmation.state, action, 1);
+    if (!running) return null;
+    const operation = Object.freeze({ ...confirmation, state: running });
+    live.current = operation; // Same-turn latch before invoking any service.
+    setRendered(operation);
+    return Object.freeze({
+      expectedInitiator: { generation: confirmation.generation, userId: ownerUserId },
+      isCurrent: current,
+      finish: () => { if (live.current === operation) { live.current = null; setRendered(null); } },
+    });
+  }
+  return { busy: rendered?.scope === scope && rendered.state.phase === 'running', confirmation: rendered?.scope === scope ? rendered : null, open, cancel, begin };

+import { Pressable, ScrollView, TextInput, View } from 'react-native';
+import { useSafeAreaInsets } from 'react-native-safe-area-context';
+
+import { Button, Text } from '@/components/ui';
+import {
+  captureHealthDataWriteLease,
+  assertOriginalHealthDataWriteLease,
+} from '@/lib/consent/healthDataWriteAdmission';
+import { localDateString } from '@/features/today/useToday';
+import { activeHealthProcessingLeaseSnapshot } from '@/lib/consent/healthProcessingEpoch';
+
+import { buildAppointmentBrief, type AppointmentBriefSelection } from './appointmentBrief';
+import { findSavedProducts } from './quickFind';
+import {
+  shareReviewedAppointmentBrief,
+  revalidateAppointmentBrief,
+  type ReviewedAppointmentBrief,
+} from './shareAppointmentBrief';
+import { useShelf } from './useShelf';
+import { loadAppointmentBriefRecords, type BriefRecordedItem, type DiaryBriefAccess } from './appointmentBriefRecords';
+import type { ReviewedDiaryPhoto } from './diaryBriefPhoto';
+
+const DiaryBriefPhotoPreview = lazy(() => import('./DiaryBriefPhotoPreview'));
+
+type Review = ReviewedAppointmentBrief & {
+  source: NonNullable<ReturnType<typeof useShelf>['data']>;
+  photo: ReviewedDiaryPhoto | null;
+};
+
+export function AppointmentBriefEditor({ onClose, diaryAccess, assertEntryCurrent }: { onClose: () => void; diaryAccess?: DiaryBriefAccess; assertEntryCurrent: () => void }) {
+  const shelf = useShelf();
+  const qc = useQueryClient();
+  const insets = useSafeAreaInsets();
+  const [selected, setSelected] = useState<string[]>([]);
+  const [query, setQuery] = useState('');
+  const [includeDates, setDates] = useState(false);
+  const [observations, setObservations] = useState('');
+  const [questions, setQuestions] = useState('');
+  const [recordIds, setRecordIds] = useState<string[]>([]);
+  const [recordSource, setRecordSource] = useState<{
+    source: Review['source'];
+    lease: Review['lease'];
+    items: readonly BriefRecordedItem[];
+  } | null>(null);
+  const recordChoiceSource = useRef<typeof recordSource>(null);
+  const [recordsLoading, setRecordsLoading] = useState(false);
+  const [recordsFailed, setRecordsFailed] = useState(false);
+  const [recordsNeedRefresh, setRecordsNeedRefresh] = useState(false);
+  const [recordQuery, setRecordQuery] = useState('');
+  const recordsFlight = useRef<object | null>(null);
+  const [review, setReview] = useState<Review | null>(null);
+  const [feedback, setFeedback] = useState<string | null>(null);
+  const [busy, setBusy] = useState(false);
+  const [selectedPhoto, setSelectedPhoto] = useState<ReviewedDiaryPhoto | null>(null);
+  const [photoLoading, setPhotoLoading] = useState(false);
+  const [photoFailed, setPhotoFailed] = useState(false);
+  const [photoPreviewReady, setPhotoPreviewReady] = useState(false);
+  const [cleanupPending, setCleanupPending] = useState(false);
+  const [cleanupBusy, setCleanupBusy] = useState(false);
+  const photoFlight = useRef<object | null>(null);
+  const reviewRef = useRef<Review | null>(null);
+  const alive = useRef(true);
+  const sharing = useRef(false);
+  const reviewFlight = useRef<object | null>(null);
+  const [verifying, setVerifying] = useState(false);
+  useEffect(() => {
+    alive.current = true;
+    return () => {
+      alive.current = false;
+      reviewRef.current = null;
+      photoFlight.current = null;
+      recordsFlight.current = null;
+      recordChoiceSource.current = null;
+      reviewFlight.current = null;
+    };
+  }, []);
+  const products = [...(shelf.data?.items ?? []), ...(shelf.data?.archive ?? [])].map(
+    (item) => item.product,
+  );
+  const choices = findSavedProducts(products, query);
+  const assertSource = (source: Review['source']) => {
+    assertEntryCurrent();
+    diaryAccess?.assertCurrent();
+    const state = qc.getQueryState(['shelf']);
+    if (
+      qc.getQueryData(['shelf']) !== source ||
+      activeHealthProcessingLeaseSnapshot()?.generation !== source.sourceGeneration ||
+      state?.status !== 'success' ||
+      state.fetchStatus !== 'idle'
+    ) {
+      throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+    }
+  };
+  let records: readonly BriefRecordedItem[] = [];
+  if (recordSource && !recordsLoading && !recordsFailed) {
+    try {
+      assertOriginalHealthDataWriteLease(recordSource.lease);
+      assertSource(recordSource.source);
+      records = recordSource.items;
+    } catch {
+      /* Retired choices must not supply private text. */
+    }
+  }
+  const recordMatches = records.filter((item) =>
+    `${item.title} ${item.lines.join(' ')}`
+      .toLowerCase()
+      .includes(recordQuery.trim().toLowerCase()),
+  );
+  async function openRecords() {
+    const source = shelf.data;
+    if (!source || !alive.current || recordsFlight.current || sharing.current) return;
+    const flight = {};
+    recordsFlight.current = flight;
+    recordChoiceSource.current = null;
+    setRecordsLoading(true);
+    setRecordsFailed(false);
+    try {
+      const lease = captureHealthDataWriteLease();
+      assertSource(source);
+      const items = await loadAppointmentBriefRecords(lease, products, diaryAccess);
+      assertOriginalHealthDataWriteLease(lease);
+      assertSource(source);
+      if (!alive.current || recordsFlight.current !== flight) return;
+      const choiceSource = { source, lease, items };
+      recordChoiceSource.current = choiceSource;
+      setRecordSource(choiceSource);
+      setRecordsNeedRefresh(false);
+      setFeedback(null);
+      // A Diary note can change in place under the same ID. Retain a choice
+      // only when its complete reviewed projection is still identical.
+      setRecordIds((previous) =>
+        previous.filter((id) => items.some((item) => item.id === id && item.available &&
+          JSON.stringify(item) === JSON.stringify(recordSource?.items.find((old) => old.id === id)))),
+      );
+    } catch {
+      if (alive.current && recordsFlight.current === flight) setRecordsFailed(true);
+    } finally {
+      if (recordsFlight.current === flight) {
+        recordsFlight.current = null;
+        if (alive.current) setRecordsLoading(false);
+      }
+    }
+  }
+  function cancelRecords() {
+    if (!recordsFlight.current || sharing.current) return;
+    recordsFlight.current = null;
+    recordChoiceSource.current = null;
+    setRecordsLoading(false); setRecordsFailed(false); setRecordsNeedRefresh(true);
+    setRecordSource(null); setRecordIds([]); setRecordQuery('');
+    setFeedback('Recorded loading canceled and recorded selections cleared. Your product selections, questions and observations are kept. Retry loading or prepare a products/questions-only brief.');
+  }
+  let currentReview = false;
+  if (review) {
+    try {
+      assertOriginalHealthDataWriteLease(review.lease);
+      assertSource(review.source);
+      currentReview = true;
+    } catch {
+      /* Never render retired preview text. */
+    }
+  }
+  async function prepare() {
+    if (!shelf.data || sharing.current || reviewFlight.current || !alive.current || photoLoading || recordsLoading || cleanupPending) return;
+    const flight = {}; reviewFlight.current = flight; setVerifying(true); setFeedback(null);
+    let verifyingSource = false;
+    try {
+      assertSource(shelf.data);
+      const lease = selectedPhoto?.lease ?? captureHealthDataWriteLease();
+      const selection: AppointmentBriefSelection = Object.freeze({
+        productIds: Object.freeze([...selected]),
+        includeDates,
+        observations,
+        questions,
+        recordIds: Object.freeze([...recordIds]),
+        includePhoto: selectedPhoto !== null,
+      });
+      const preparedOn = localDateString();
+      const accepted = Object.freeze({
+        source: shelf.data,
+        lease,
+        diaryAccess,
+        photo: selectedPhoto,
+        selection,
+        preparedOn,
+        text: buildAppointmentBrief(products, selection, preparedOn, records),
+      });
+      assertOriginalHealthDataWriteLease(lease);
+      // Review must describe a fresh authenticated selection. Sharing repeats
+      // this read and fences final dispatch; this check grants no recipient access.
+      verifyingSource = true;
+      await revalidateAppointmentBrief(accepted, () => {
+        if (!alive.current || reviewFlight.current !== flight) throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+        assertSource(accepted.source);
+      });
+      if (!alive.current || reviewFlight.current !== flight) return;
+      assertOriginalHealthDataWriteLease(lease); assertSource(accepted.source);
+      reviewRef.current = accepted;
+      setReview(accepted);
+      setPhotoPreviewReady(false);
+      setFeedback(null);
+    } catch (error) {
+      if (!alive.current || reviewFlight.current !== flight) return;
+      const code = error instanceof Error ? error.message : '';
+      if (verifyingSource && (code === 'APPOINTMENT_BRIEF_INVALID_OR_STALE' || code === 'APPOINTMENT_BRIEF_SOURCE_UNAVAILABLE')) {
+        reviewRef.current = null; setReview(null); recordChoiceSource.current = null;
+        setRecordSource(null); setRecordIds([]); setSelected([]); setSelectedPhoto(null); setPhotoPreviewReady(false); setRecordsNeedRefresh(true);
+        setFeedback(code === 'APPOINTMENT_BRIEF_INVALID_OR_STALE'
+          ? 'Selected records changed before review. Refresh recorded choices, select exact sources again, and review. Your questions and observations are kept. Nothing was shared.'
+          : 'Current records could not be verified before review. No records were reset. Refresh choices and select sources again. Your questions and observations are kept. Nothing was shared.');
+      } else setFeedback('Choose current products or write a question/observation. Remove unsupported control characters and keep each text field within 2,000 characters. If records changed, close and start again.');
+    } finally {
+      if (reviewFlight.current === flight) { reviewFlight.current = null; if (alive.current) setVerifying(false); }
+    }
+  }
+  function cancelVerification() {
+    reviewFlight.current = null; setVerifying(false);
+    setFeedback('Brief verification canceled. Nothing was shared. Your selections, questions and observations are kept for a fresh review.');
+  }
+  async function choosePhoto() {
+    if (!diaryAccess || sharing.current || cleanupPending) return;
+    if (photoLoading || selectedPhoto) {
+      photoFlight.current = null; setSelectedPhoto(null); setPhotoLoading(false); setPhotoFailed(false); return;
+    }
+    const flight = {}; photoFlight.current = flight;
+    setPhotoLoading(true); setPhotoFailed(false);
+    try {
+      const lease = captureHealthDataWriteLease(); diaryAccess.assertCurrent();
+      const { prepareDiaryBriefPhoto } = await import('./diaryBriefPhoto');
+      assertOriginalHealthDataWriteLease(lease); diaryAccess.assertCurrent();
+      const photo = await prepareDiaryBriefPhoto(lease, diaryAccess);
+      if (alive.current && photoFlight.current === flight) setSelectedPhoto(photo);
+    } catch {
+      if (alive.current && photoFlight.current === flight) setPhotoFailed(true);
+    } finally {
+      if (alive.current && photoFlight.current === flight) { photoFlight.current = null; setPhotoLoading(false); }
+    }
+  }
+  async function retryCleanup() {
+    if (cleanupBusy) return; setCleanupBusy(true);
+    try {
+      await (await import('@/lib/storage/plaintextStaging')).retryPlaintextStagingRecovery();
+      if (alive.current) { setCleanupPending(false); setFeedback('Local temporary exports were removed. Recipient copies cannot be revoked.'); }
+    } catch { if (alive.current) setFeedback('Temporary-file removal is still unconfirmed. Retry cleanup before exporting again.'); }
+    finally { if (alive.current) setCleanupBusy(false); }
+  }
+  async function share() {
+    // Capture the review represented by this exact rendered button. An old
+    // callback cannot adopt a later review, owner, consent grant or selection.
+    const accepted = review;
+    if (!accepted || sharing.current || reviewRef.current !== accepted || cleanupPending || (accepted.photo && !photoPreviewReady)) return;
+    sharing.current = true;
+    setBusy(true);
+    setFeedback(null);
+    try {
+      const assertAccepted = () => {
+        assertSource(accepted.source);
+        if (!alive.current || reviewRef.current !== accepted)
+          throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+      };
+      if (accepted.photo) {
+        const { exportReviewedPhotoBrief } = await import('./diaryBriefPhoto');
+        assertAccepted();
+        const result = await exportReviewedPhotoBrief(accepted, accepted.photo, assertAccepted);
+        if (alive.current) {
+          setCleanupPending(result.cleanup === 'pending');
+          setFeedback(`${result.handoff === 'completed' ? 'The system sharing call completed; delivery or cancellation is not confirmed.' : 'Image-file sharing is unavailable on this device.'} ${result.cleanup === 'pending' ? 'Temporary-file removal is unconfirmed. Retry cleanup before another export.' : result.cleanup === 'removed' ? 'The local temporary file was removed.' : 'No temporary export file was created.'}`);
+        }
+      } else {
+        const shared = await shareReviewedAppointmentBrief(accepted, assertAccepted);
+        if (alive.current)
+        setFeedback(
+          shared
+            ? 'The system share sheet reported completion. Recipient receipt, review or action is not confirmed.'
+            : 'Sharing was not completed. Your brief is still here.',
+        );
+      }
+    } catch (error) {
+      let admitted = false;
+      try { assertOriginalHealthDataWriteLease(accepted.lease); admitted = true; } catch { /* Retired approval owns no recovery UI. */ }
+      if (alive.current && admitted && reviewRef.current === accepted) {
+        const removalPending = typeof error === 'object' && error !== null && 'cleanup' in error && error.cleanup === 'pending';
+        if (removalPending) setCleanupPending(true);
+        const code = error instanceof Error ? error.message : '';
+        if (code === 'APPOINTMENT_BRIEF_INVALID_OR_STALE' || code === 'APPOINTMENT_BRIEF_SOURCE_UNAVAILABLE') {
+          reviewRef.current = null; setReview(null);
+          setSelectedPhoto(null); setPhotoPreviewReady(false);
+          recordChoiceSource.current = null;
+          setRecordSource(null); setRecordIds([]); setSelected([]); setRecordsFailed(false); setRecordsNeedRefresh(true);
+          setFeedback((code === 'APPOINTMENT_BRIEF_INVALID_OR_STALE'
+            ? 'Selected records changed. Approval was cleared. Refresh recorded choices, select sources again, and review. Your questions and observations are kept.'
+            : 'Current records could not be verified. Approval was cleared. Refresh recorded choices and select sources again. Your questions and observations are kept; no records were reset.') +
+            (removalPending ? ' Temporary-file removal is unconfirmed. Retry cleanup before another export.' : ''));
+        } else setFeedback(accepted.photo
+          ? `Image export was not confirmed. ${removalPending ? 'Temporary-file removal is unconfirmed; retry cleanup before another export.' : 'Review the selection and try again; existing Diary records were not changed.'}`
+          : 'Sharing was not confirmed by the device. Your reviewed brief is still here. Retry sharing or edit it; no new brief file was saved.');
+      }
+    } finally {
+      sharing.current = false;
+      if (alive.current) setBusy(false);
+    }
+  }
+  return (
+    <View
+      className="flex-1 px-5"
+      style={{ paddingTop: insets.top + 12, paddingBottom: insets.bottom + 8 }}
+    >
+      <View className="flex-row flex-wrap items-center justify-between gap-2">
+        <Text variant="titleSm" className="flex-1">
+          My appointment brief
+        </Text>
+        <Button label="Close appointment brief" variant="ghost" onPress={onClose} />
+      </View>
+      <ScrollView
+        key={review ? 'review' : 'selection'}
+        className="mt-3 flex-1"
+        contentContainerClassName="gap-3 pb-8"
+        keyboardShouldPersistTaps="handled"
+      >
+        <Text variant="bodySm" tone="muted">
+          Free and optional. This draft stays in memory until you close it. Nothing is automatically
+          sent. Names, brands and status are included only for products you select. No appointment
+          or clinician advice is generated. Images are excluded unless you separately select the opened Diary photo.
+        </Text>
+        {verifying ? <View className="gap-3" accessibilityLiveRegion="polite">
+          <Text>Verifying selected source records before review… Nothing has been shared.</Text>
+          <Button label="Cancel brief verification" variant="ghost" onPress={cancelVerification} />
+        </View> : review ? (
+          <>
+            <Text accessibilityRole="header" variant="bodySm" className="font-sans-semibold">
+              Review everything before sharing
+            </Text>
+            {currentReview ? (
+              <Text variant="bodySm" selectable accessibilityLabel="Reviewed appointment brief">
+                {review.text}
+              </Text>
+            ) : (
+              <Text accessibilityRole="alert">
+                Saved records or privacy authority changed. This preview is no longer available.
+                Close and prepare a new brief.
+              </Text>
+            )}
+            {currentReview && review.photo ? <View className="gap-2">
+              <Text variant="bodySm">One selected Diary photo · capture local date {review.photo.date} · recorded series {review.photo.series}. No assessment or treatment conclusion.</Text>
+              <Suspense fallback={<Text>Loading selected photo for review…</Text>}>
+                <DiaryBriefPhotoPreview photo={review.photo} onReady={() => {
+                  if (alive.current && reviewRef.current === review) setPhotoPreviewReady(true);
+                }} onError={() => {
+                  if (!alive.current || reviewRef.current !== review) return;
+                  setPhotoPreviewReady(false); setFeedback('Selected photo could not be displayed. Cancel or edit the selection; it cannot be exported without review.');
+                }} />
+              </Suspense>
+            </View> : null}
+            <Text variant="bodySm" tone="muted">
+              Sharing opens your device’s share sheet. You choose the recipient. Copies in another
+              app cannot be revoked by deleting records or withdrawing consent here. {review.photo ? 'The one photo shown above is included in a local HTML document. Choose the recipient yourself.' : 'No photos are included.'} Internal account identifiers are not included.
+            </Text>
+            <Button
+              label={busy ? 'Opening share sheet…' : review.photo ? 'Export reviewed brief with selected photo' : 'Share reviewed brief'}
+              disabled={busy || !currentReview || cleanupPending || Boolean(review.photo && !photoPreviewReady)}
+              onPress={() => void share()}
+            />
+            <Button
+              label="Edit brief selection"
+              variant="ghost"
+              disabled={busy}
+              onPress={() => {
+                reviewRef.current = null;
+                setReview(null);
+                setFeedback(null);
+              }}
+            />
+          </>
+        ) : (
+          <>
+            {diaryAccess ? <View className="gap-2 rounded-card border border-hairline p-3">
+              <Pressable accessibilityRole="checkbox" accessibilityLabel="Include the opened Diary photo"
+                accessibilityState={{ checked: selectedPhoto !== null, busy: photoLoading }} aria-checked={selectedPhoto !== null}
+                className="min-h-[52px] justify-center" onPress={() => void choosePhoto()}>
+                <Text>{photoLoading ? 'Cancel photo selection' : selectedPhoto ? '✓ Include the opened Diary photo' : 'Include the opened Diary photo'}</Text>
+              </Pressable>
+              <Text variant="bodySm" tone="muted">Optional and separate from text notes. Only this opened photo can be selected. Review it before exporting; no automatic upload. Large or unavailable images stay excluded.</Text>
+              {photoFailed ? <Text accessibilityRole="alert">Photo not included. It could not be confirmed. Retry selection or continue with a text-only brief.</Text> : null}
+            </View> : null}
+            <TextInput
+              accessibilityLabel="Find products for appointment brief"
+              placeholder="Find by name or brand"
+              value={query}
+              onChangeText={setQuery}
+              maxLength={120}
+              autoCorrect={false}
+              className="min-h-[48px] rounded-card border border-hairline px-3 font-sans text-ink"
+            />
+            <Text variant="label" tone="muted">
+              {selected.length} products selected · includes archived products
+            </Text>
+            {choices.map((product, index) => (
+              <Pressable
+                key={product.id}
+                accessibilityRole="checkbox"
+                aria-checked={selected.includes(product.id)}
+                accessibilityLabel={`Include ${product.name}, ${product.status}, result ${index + 1}`}
+                accessibilityState={{ checked: selected.includes(product.id) }}
+                onPress={() =>
+                  setSelected((previous) =>
+                    previous.includes(product.id)
+                      ? previous.filter((id) => id !== product.id)
+                      : [...previous, product.id],
+                  )
+                }
+                className="min-h-[52px] rounded-card border border-hairline p-3"
+              >
+                <Text variant="bodySm">
+                  {selected.includes(product.id) ? '✓ ' : ''}
+                  {product.name}
+                </Text>
+                <Text variant="label" tone="muted">
+                  {product.brand ?? 'Brand unknown'} · {product.status}
+                </Text>
+              </Pressable>
+            ))}
+            {!choices.length ? (
+              <Text variant="bodySm">
+                No products match. You can clear the search or write questions without adding
+                products.
+              </Text>
+            ) : null}
+            <Pressable
+              accessibilityRole="checkbox"
+              aria-checked={includeDates}
+              accessibilityLabel="Include recorded product dates"
+              accessibilityState={{ checked: includeDates }}
+              onPress={() => setDates((value) => !value)}
+              className="min-h-[52px] justify-center rounded-card border border-hairline p-3"
+            >
+              <Text variant="bodySm">{includeDates ? '✓ ' : ''}Include recorded product dates</Text>
+            </Pressable>
+            <Text variant="bodySm" tone="muted">
+              Dates never imply treatment start or duration of use. Unknown dates stay unknown.
+            </Text>
+            <View className="gap-3 rounded-card border border-hairline p-3">
+              <Text variant="bodySm" className="font-sans-semibold">
+                Existing observations and saved routines
+              </Text>
+              <Text variant="bodySm" tone="muted">
+                Load choices, then select each record explicitly. Only the latest un-retracted
+                observation in a correction chain is eligible. Earlier records remain in Today
+                history. Plans are dated snapshots, not proof of use. No photos or diary media are
+                read.
+              </Text>
+              {diaryAccess ? <Text variant="bodySm">The saved text note from your opened Diary record is also eligible. Select it individually in recorded choices. Capture date is not the note creation date; no image is read or included.</Text> : null}
+              <Button
+                label={
+                  recordsLoading
+                    ? 'Loading recorded choices…'
+                    : recordSource || recordsNeedRefresh
+                      ? 'Refresh recorded choices'
+                      : 'Choose recorded notes and saved plans'
+                }
+                variant="ghost"
+                disabled={recordsLoading}
+                onPress={() => void openRecords()}
+              />
+              {recordsLoading ? <Button label="Cancel recorded loading" variant="ghost" onPress={cancelRecords} /> : null}
+              {recordsFailed ? (
+                <Text accessibilityRole="alert" variant="bodySm">
+                  Recorded choices unavailable. No records were reset. Retry loading; or clear
+                  recorded selections to prepare a products/questions-only brief.
+                </Text>
+              ) : null}
+              {recordSource && !recordsLoading && !recordsFailed ? (
+                <>
+                  <TextInput
+                    accessibilityLabel="Find recorded choices"
+                    value={recordQuery}
+                    onChangeText={setRecordQuery}
+                    placeholder="Find by date, source or recorded text"
+                    maxLength={120}
+                    autoCorrect={false}
+                    className="min-h-[48px] rounded-card border border-hairline px-3 font-sans text-ink"
+                  />
+                  {recordMatches.map((item) => (
+                    <Pressable
+                      key={item.id}
+                      accessibilityRole="checkbox"
+                      aria-checked={recordIds.includes(item.id)}
+                      accessibilityLabel={`Include ${item.title}`}
+                      accessibilityState={{
+                        checked: recordIds.includes(item.id),
+                        disabled: !item.available,
+                      }}
+                      disabled={!item.available}
+                      onPress={() => {
+                        // A queued callback from a retired chooser cannot select
+                        // newly loaded text that happens to have the same ID.
+                        if (!alive.current || !item.available || !recordSource ||
+                            recordChoiceSource.current !== recordSource || recordsFlight.current) return;
+                        try { assertOriginalHealthDataWriteLease(recordSource.lease); assertSource(recordSource.source); }
+                        catch { return; }
+                        setRecordIds((previous) =>
+                          previous.includes(item.id)
+                            ? previous.filter((id) => id !== item.id)
+                            : [...previous, item.id],
+                        );
+                      }}
+                      className="min-h-[52px] rounded-card border border-hairline p-3"
+                    >
+                      <Text variant="bodySm" className="font-sans-semibold">
+                        {recordIds.includes(item.id) ? '✓ ' : ''}
+                        {item.title}
+                      </Text>
+                      {item.lines.map((line, index) => (
+                        <Text key={index} variant="bodySm" tone="muted">
+                          {line}
+                        </Text>
+                      ))}
+                    </Pressable>
+                  ))}
+                  {!records.length ? (
+                    <Text variant="bodySm">
+                      No eligible recorded notes or saved plans are available. Nothing has been
+                      added to your brief.
+                    </Text>
+                  ) : null}
+                  {records.length > 0 && recordMatches.length === 0 ? (
+                    <Text variant="bodySm">
+                      No recorded choices match. Clear or change the recorded search.
+                    </Text>
+                  ) : null}
+                </>
+              ) : null}
+              <Text variant="label" tone="muted">
+                {recordIds.length} recorded items selected
+              </Text>
+              {recordIds.length ? (
+                <Button
+                  label="Clear recorded selections"
+                  variant="ghost"
+                  onPress={() => setRecordIds([])}
+                />
+              ) : null}
+            </View>
+            <Text variant="label" tone="muted">
+              My observations (optional, written by me)
+            </Text>
+            <TextInput

+  useLayoutEffect(() => {

+    setFocusRevision(serial.current); setFocused(navigation?.isFocused() ?? true);
+    const blur = navigation?.addListener('blur', () => { retire(); setFocusRevision(serial.current); setFocused(false); });
+    const focus = navigation?.addListener('focus', () => { retire(); setFocusRevision(serial.current); setFocused(true); });
+    return () => { retire(); blur?.(); focus?.(); };
+  }, [navigation]);

+    subscribeActiveHealthProcessingLeaseChanges,

+    () => null,

+  };
+  return authority && focused ? <CompareSession key={`${authority.generation}:${focusRevision}`} briefOnly={briefOnly} diaryAccess={diaryAccess} assertEntryCurrent={assertEntryCurrent} /> : null;
