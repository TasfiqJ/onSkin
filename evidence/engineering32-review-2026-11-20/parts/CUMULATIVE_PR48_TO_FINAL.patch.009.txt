+export type WebRuntimeStatus = 'checking' | 'admitted' | 'occupied' | 'unavailable';
+export type RuntimeLockPort = {
+  request(name: string, options: { mode: 'exclusive'; signal: AbortSignal },
+    callback: (lock: unknown | null) => Promise<void>): Promise<unknown>;
+};
+export const PRIVATE_WEB_RUNTIME_LOCK = 'layerwell.private-runtime.v1';
+export const PRIVATE_WEB_HANDOFF_WAIT_MS = 3000;
+
+/** One application runtime per browser origin. Holding the lock for the
+ * document lifetime also covers direct cleanup/key/control-record operations
+ * which do not participate in privateKV's per-key mutation queues.
+ * Never steal or release on blur/navigation/component unmount. The browser
+ * releases the lock when this context ends, including a crash or reload. */
+export function createWebRuntimeAdmission(port: RuntimeLockPort | undefined) {
+  let status: WebRuntimeStatus = 'checking';
+  let serial = 0;
+  let pending: { id: number; controller: AbortController; timer: ReturnType<typeof setTimeout> } | undefined;
+  const transitions: { attempt: number; status: WebRuntimeStatus; reason: string }[] = [];
+  const listeners = new Set<() => void>();
+  const publish = (next: WebRuntimeStatus, reason: string) => {
+    status = next;
+    transitions.push({ attempt: serial, status, reason });
+    if (transitions.length > 64) transitions.shift();
+    for (const listener of [...listeners]) {
+      try { listener(); } catch { /* A listener cannot release runtime ownership. */ }
+    }
+  };
+  const cancelPending = () => {
+    if (!pending) return;
+    const previous = pending; pending = undefined; ++serial;
+    clearTimeout(previous.timer); previous.controller.abort();
+    publish('occupied', 'pending-document-retired');
+  };
+  const retry = () => {
+    if (pending || status === 'admitted') return;
+    if (!port) { publish('unavailable', 'unsupported'); return; }
+    const id = ++serial, controller = new AbortController();
+    const isCurrent = () => serial === id;
+    const finishPending = () => { if (pending?.id === id) { clearTimeout(pending.timer); pending = undefined; } };
+    const timer = setTimeout(() => {
+      if (!isCurrent() || status === 'admitted') return;
+      finishPending(); ++serial; controller.abort();
+      publish('occupied', 'bounded-wait-ended');
+    }, PRIVATE_WEB_HANDOFF_WAIT_MS);
+    pending = { id, controller, timer }; publish('checking', 'exclusive-request');
+    try {
+      void port.request(PRIVATE_WEB_RUNTIME_LOCK, { mode: 'exclusive', signal: controller.signal }, async lock => {
+        // Aborted/superseded callbacks return immediately and retain no lock.
+        if (!isCurrent() || controller.signal.aborted) return;
+        finishPending();
+        if (lock === null) { publish('unavailable', 'invalid-null-grant'); return; }
+        publish('admitted', 'exclusive-callback');
+        await new Promise<void>(() => { /* Deliberately retained until context destruction. */ });
+      }).then(() => {
+        if (!isCurrent()) return;
+        finishPending(); publish('unavailable', 'request-ended');
+      }, () => {
+        if (!isCurrent()) return;
+        finishPending(); publish('unavailable', 'request-rejected');
+      });
+    } catch { if (isCurrent()) { finishPending(); publish('unavailable', 'request-threw'); } }
+  };
+  return { snapshot: () => status, retry, cancelPending, diagnostics: () => transitions.map(row => ({ ...row })), subscribe: (listener: () => void) => {
+    listeners.add(listener); return () => { listeners.delete(listener); };
+  } };
+}

+ * transaction's projection may be serialized, never an earlier private read. */
+export async function publishProfessional<T>(repository: ProfessionalRepository | null, token: string, request: AccessRequest,

+  if (!repository) throw new ProfessionalAccessError('unavailable');

+  const captured = Object.freeze({ ...request, fields: Object.freeze([...request.fields]) });

+    if (denied) throw new ProfessionalAccessError(denied);
+    if (authorityRevision(current) !== revision) throw new ProfessionalAccessError('changed');
+    return project(current);
+  });
