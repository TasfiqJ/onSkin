+/** Retire the wait even if a transport or body reader ignores cancellation.
+ * Late resolution/rejection is consumed and cannot resume the retired caller. */
+export function untilAborted<T>(signal: AbortSignal, work: Promise<T>): Promise<T> {
+  return new Promise((resolve, reject) => {
+    const abort = () => reject(new Error('REQUEST_RETIRED'));
+    if (signal.aborted) abort();
+    else signal.addEventListener('abort', abort, { once: true });
+    work.then(value => {
+      if (!signal.aborted) resolve(value);
+    }, reject).finally(() => signal.removeEventListener('abort', abort));
+  });
+}

+const reply = (status: number, body: unknown) => new Response(JSON.stringify(body), { status, headers });
+/** No default authenticator, provisioning or production registration. */
+export function createTemplateApi(dependencies: TemplateApiDependencies | null) {
+  const deps = dependencies ? { ...dependencies } : null;
+  return async (request: Request): Promise<Response> => {
+    if (new URL(request.url).pathname !== '/api/professional/templates') return reply(404, { status: 'unavailable' });
+    if (!deps) return reply(503, { status: 'unavailable' });
+    try {
+      if (request.method !== 'POST' || new URL(request.url).origin !== deps.origin || request.headers.get('Origin') !== deps.origin ||
+          request.headers.get('X-OnSkin-Request') !== 'templates-v1' || request.headers.get('Content-Type') !== 'application/json' ||
+          request.headers.has('Sec-Fetch-Site') && request.headers.get('Sec-Fetch-Site') !== 'same-origin') return reply(403, { status: 'denied' });

+      const id = (v: unknown) => typeof v === 'string' && /^[A-Za-z0-9][A-Za-z0-9:_-]{0,127}$/u.test(v);
+      if (!p || !p.token || !id(p.context) || !id(p.workspaceId)) return reply(401, { status: 'session' });
+      if (!request.body) return reply(400, { status: 'invalid' });
+      const reader = request.body.getReader(); const chunks: Uint8Array[] = []; let size = 0;
+      try {
+        while (true) { const next = await reader.read(); if (next.done) break;
+          size += next.value.byteLength; if (size > 16_384) { await reader.cancel(); return reply(400, { status: 'invalid' }); } chunks.push(next.value); }
+      } finally { reader.releaseLock(); }
+      const bytes = new Uint8Array(size); let offset = 0;
+      for (const chunk of chunks) { bytes.set(chunk, offset); offset += chunk.byteLength; }
+      let value: unknown;
+      try { value = JSON.parse(new TextDecoder('utf-8', { fatal: true }).decode(bytes)); } catch { return reply(400, { status: 'invalid' }); }

+      const input = value as Record<string, unknown>;

+export type RoleChoice = Readonly<{ kind: 'unresolved' | 'unshelved' | 'omit' } | { kind: 'product'; productId: string }>;
+export type TemplateMapping = Readonly<{ revisionId: string; revisionHash: string; am: readonly RoleChoice[]; pm: readonly RoleChoice[] }>;
+export type TemplateOrigin = Readonly<{ templateId: string; revisionId: string; revisionHash: string; revision: number;

+const productId = (v: unknown): v is string => typeof v === 'string' && !!v.trim() && v.length <= 128 && !/[\p{Cf}\p{Cs}\u0000-\u001f\u007f-\u009f]/u.test(v);
+function keys(v: unknown, expected: string): asserts v is Record<string, unknown> {
+  if (!v || typeof v !== 'object' || Array.isArray(v) || Object.keys(v).sort().join(',') !== expected) fail();

+export function decodeTemplateMapping(value: unknown): TemplateMapping {
+  const v = structuredClone(value); keys(v, 'am,pm,revisionHash,revisionId');
+  if (!templateId(v.revisionId) || !templateHash(v.revisionHash)) return fail();
+  for (const phase of ['am', 'pm'] as const) {
+    const rows = v[phase]; if (!Array.isArray(rows) || rows.length > 20) return fail();

+      keys(row, row?.kind === 'product' ? 'kind,productId' : 'kind');
+      if (!['product', 'omit', 'unresolved', 'unshelved'].includes(row.kind as string) || row.kind === 'product' && !productId(row.productId)) return fail();
+    }
+  }
+  return v as unknown as TemplateMapping;
+}
+/** A hash/shape is integrity, never ownership or professional authority. Call
+ * only with decoded practice history and the current trusted product source.
+ * Unresolved/unshelved roles have no product identity and cannot be issued. */
+export function mapTemplateRevision(history: readonly PracticeTemplate[], input: unknown, products: readonly RoutineStep[]) {
+  const mapping = decodeTemplateMapping(input);
+  const selected = history.find(t => t.id === mapping.revisionId && t.contentHash === mapping.revisionHash && t.status === 'active');
+  if (!selected || history.filter(t => t.templateId === selected.templateId).at(-1)?.status !== 'active') return fail('CHANGED');
+  if (!Array.isArray(products) || new Set(products.map(p => p.productId)).size !== products.length) return fail();
+  const am: RoutineStep[] = [], pm: RoutineStep[] = [];
+  const origin: { am: { role: TemplateRole; productId: string | null }[]; pm: { role: TemplateRole; productId: string | null }[] } = { am: [], pm: [] };
+  for (const phase of ['am', 'pm'] as const) {
+    if (mapping[phase].length !== selected[phase].length) return fail();
+    const result = phase === 'am' ? am : pm;
+    for (const [index, choice] of mapping[phase].entries()) {
+      if (choice.kind === 'unresolved' || choice.kind === 'unshelved') return fail('UNRESOLVED');
+      const product = choice.kind === 'product' ? products.find(p => p.productId === choice.productId) : null;
+      if (choice.kind === 'product' && (!product || result.some(p => p.productId === product.productId))) return fail('CHANGED');
+      if (product) result.push({ ...product });
+      origin[phase].push({ role: selected[phase][index]!, productId: product?.productId ?? null });
+    }
+  }
+  if (!am.length && !pm.length) return fail('UNRESOLVED');
+  const templateOrigin: TemplateOrigin = { templateId: selected.templateId, revisionId: selected.id,
+    revisionHash: selected.contentHash, revision: selected.revision, ...origin };
+  return { am, pm, templateOrigin: validateTemplateOrigin(templateOrigin, { am, pm }) };
+}
+/** Retained provenance is checked without rereading a retired practice. It
+ * records the exact original selection; it confers no new access or safety. */
+export function validateTemplateOrigin(value: unknown, sequence: { am: readonly RoutineStep[]; pm: readonly RoutineStep[] }): TemplateOrigin {
+  const v = structuredClone(value); keys(v, 'am,pm,revision,revisionHash,revisionId,templateId');
+  if (!templateId(v.templateId) || !templateId(v.revisionId) || !templateHash(v.revisionHash) ||
+      !Number.isSafeInteger(v.revision) || (v.revision as number) < 1 || (v.revision as number) > 100) return fail();
+  let total = 0;
+  for (const phase of ['am', 'pm'] as const) {
+    const rows = v[phase]; if (!Array.isArray(rows) || rows.length > 20) return fail();
+    const ids: string[] = [];
+    for (const row of rows) {
+      keys(row, 'productId,role');
+      if (!TEMPLATE_ROLES.includes(row.role as TemplateRole) || row.productId !== null && !productId(row.productId)) return fail();

+    }
+    if (new Set(ids).size !== ids.length || JSON.stringify(ids) !== JSON.stringify(sequence[phase].map(p => p.productId))) return fail();
+    total += ids.length;
+  }
+  if (!total) return fail();
+  return Object.freeze({ ...v, am: Object.freeze((v.am as TemplateOrigin['am']).map(r => Object.freeze(r))),
+    pm: Object.freeze((v.pm as TemplateOrigin['pm']).map(r => Object.freeze(r))) }) as TemplateOrigin;

+async function append(state: WorkspaceBilling, facts: ProviderBillingFacts): Promise<WorkspaceBilling> {
+  if (!factsValid(facts) || facts.workspaceId !== state.workspaceId || facts.customerId !== state.customerId || facts.subscriptionId !== state.subscriptionId) return fail();
+  const existing = state.receipts.find(receipt => receipt.eventId === facts.eventId);
+  if (existing) {

+    // substantive identity must remain exact. A fresh event is needed to renew.
+    const original = { ...existing } as Record<string, unknown>;

+    if (canonical(original) !== canonical(current)) return fail();

+const READS = new Set<keyof WorkspaceTransaction>([
+  'authority', 'lifecycle', 'grants', 'billing', 'capacity', 'plans', 'receivedPlans',

+/** A repository read is a value, never permission to adopt a later mutable
+ * alias. Keep its original bytes as well as a private copy: cloning alone
+ * hides retirement during a subsequent awaited final check. This guard does
+ * not replace the repository's serial transaction/current-source contract. */
+export function snapshotWorkspaceTransaction(tx: WorkspaceTransaction): WorkspaceTransaction {
+  const retained: { value: unknown; fingerprint: string }[] = [];
+  function assertUnchanged() {
+    if (retained.some(read => canonical(read.value) !== read.fingerprint)) {
+      throw new ProfessionalAccessError('changed');
+    }
+  }
+  tx.beforeCommit(async () => assertUnchanged());
+  // The port may itself be frozen. An empty target avoids proxy invariants
+  // that would otherwise prohibit wrapping its non-writable methods.
+  return new Proxy({} as WorkspaceTransaction, {
+    get(_target, key) {
+      const method = Reflect.get(tx, key, tx);
+      if (key === 'beforeCommit') {
+        return (check: () => Promise<void>) => tx.beforeCommit(async () => {
+          assertUnchanged();
+          await check();
+          assertUnchanged();
+        });
+      }
+      if (READS.has(key as keyof WorkspaceTransaction) && typeof method === 'function') {
+        return async (...args: unknown[]) => {
+          const value: unknown = await Reflect.apply(method, tx, structuredClone(args));
+          assertUnchanged();
+          const captured = structuredClone(value);
+          retained.push({ value, fingerprint: canonical(captured) });
+          return captured;
+        };
+      }
+      return typeof method === 'function' ? method.bind(tx) : method;
+    },
+  });
+}
