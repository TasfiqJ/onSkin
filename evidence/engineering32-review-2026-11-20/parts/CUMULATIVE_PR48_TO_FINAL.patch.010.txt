+/** Current trust/source projections MUST be resolved in the same custody
+ * transaction (including session/revocation and tombstones before restore).
+ * No caller-supplied Authority, plaintext table, secondary sync copy, default
+ * projection, provisioner, or automatic repair is provided by this adapter. */
+export interface WorkspaceTrust {
+  authority(tx: EncryptedTransaction, token: string, request: AccessRequest): Promise<Authority>;
+  lifecycle(tx: EncryptedTransaction, ownerId: string): Promise<OwnerLifecycle>;
+  planSource(tx: EncryptedTransaction, ownerId: string, workspaceId: string): Promise<ClientPlanSource>;
+  activationAvailable(tx: EncryptedTransaction, workspaceId: string): Promise<boolean>;
+  reconciliationSource?(tx: EncryptedTransaction, ownerId: string, workspaceId: string): Promise<ReconciliationSource>;
+  previsitRecords?(tx: EncryptedTransaction, ownerId: string, workspaceId: string, fields: readonly PrevisitRecordField[]): Promise<PrevisitRecordSource>;

+const changed = (): never => { throw new ProfessionalAccessError('changed'); };
+function id(value: string): string { if (!/^[A-Za-z0-9][A-Za-z0-9:_-]{0,127}$/u.test(value)) return changed(); return value; }
+export const workspaceRecordKey = {
+  grants: (owner: string) => `owner/${id(owner)}/grants`,
+  copies: (owner: string) => `owner/${id(owner)}/received`,
+  receipts: (owner: string) => `owner/${id(owner)}/receipts`,
+  plans: (owner: string, workspace: string) => `owner/${id(owner)}/workspace/${id(workspace)}/plans`,
+  reconciliation: (owner: string, workspace: string) => `owner/${id(owner)}/workspace/${id(workspace)}/reconciliation`,
+  billing: (workspace: string) => `workspace/${id(workspace)}/billing`,
+  capacity: (workspace: string) => `workspace/${id(workspace)}/capacity`,
+  templates: (workspace: string) => `workspace/${id(workspace)}/templates`,
+};
+function prefix(previous: readonly unknown[], next: readonly unknown[]): void {
+  if (next.length < previous.length || canonical(next.slice(0, previous.length)) !== canonical(previous)) changed();
+}
+async function copies(value: unknown, owner: string, now: number): Promise<readonly IntendedPlan[]> {
+  if (!Array.isArray(value) || value.length > 1000 || new Set(value.map(plan => plan?.id)).size !== value.length) return changed();
+  for (const plan of value as IntendedPlan[]) { await validateIntendedPlan(plan, now); if (plan.ownerId !== owner) changed(); }
+  return value as IntendedPlan[];
+}

+/** Production staging/encryption/strict-read implementation of the existing
+ * service port. The external serializable driver is still unconfigured. Missing
+ * history is unavailable, never an implicit empty aggregate or write success. */
+export class EncryptedWorkspaceRepository implements WorkspaceRepository {
+  constructor(private readonly custody: EncryptedCustody, private readonly trust: WorkspaceTrust | null) {}
+  async transaction<T>(scope: Readonly<{ workspaceId: string | null; ownerId: string | null }>, work: (tx: WorkspaceTransaction) => Promise<T>): Promise<T> {
+    const trust = this.trust;
+    if (!trust) throw new ProfessionalAccessError('unavailable');
+    const ownerId = scope.ownerId === null ? null : id(scope.ownerId);
+    const workspaceId = scope.workspaceId === null ? null : id(scope.workspaceId);
+    if (!ownerId && !workspaceId) return changed();
+    return this.custody.transaction(async store => {
+      const authorizedOwner = ownerId; let authorizedWorkspace = workspaceId;
+      const assertOwner = async (owner: string) => {
+        if (owner !== authorizedOwner) return changed();
+        const life = await trust.lifecycle(store, owner);
+        if (life.ownerId !== owner || life.deletionPending || !Number.isSafeInteger(life.epoch) || life.epoch < 1) return changed();
+        return life;
+      };
+      const assertWorkspace = (workspace: string) => { if (workspace !== authorizedWorkspace) changed(); };
+      // Independent owner reads discover their current workspace from canonical
+      // grants. The underlying driver handles the dynamic set atomically.
+      const discoverWorkspace = async (workspace: string) => {
+        if (authorizedWorkspace === null && authorizedOwner) {
+          const grant = await decodeGrants(await store.read(workspaceRecordKey.grants(authorizedOwner)), authorizedOwner, store.now());
+          if (grant.active?.workspaceId !== workspace) return changed();
+          authorizedWorkspace = workspace;
+        }
+        assertWorkspace(workspace);
+      };
+      const tx: WorkspaceTransaction = {
+        now: store.now,
+        authority: async (token, request) => {
+          // Trust owns source-of-truth session resolution; it must not touch idle
+          // activity during service rechecks or borrow a cached request identity.
+          if (request.ownerId !== ownerId || (workspaceId !== null && request.workspaceId !== null && request.workspaceId !== workspaceId)) return changed();
+          const authority = await trust.authority(store, token, request);
+          return authority;
+        },
+        lifecycle: async owner => assertOwner(owner),
+        templates: async workspace => { assertWorkspace(workspace);
+          return decodePracticeTemplates(await store.read(workspaceRecordKey.templates(workspace)), workspace, store.now()); },
+        saveTemplates: async value => {
+          if (!authorizedWorkspace || authorizedOwner !== null) return changed();
+          const next = await decodePracticeTemplates(value, authorizedWorkspace, store.now());
+          const previous = await tx.templates!(authorizedWorkspace); prefix(previous, next);
+          await store.write(workspaceRecordKey.templates(authorizedWorkspace), next);
+        },
+        grants: async owner => { await assertOwner(owner); return decodeGrants(await store.read(workspaceRecordKey.grants(owner)), owner, store.now()); },
+        billing: async workspace => { await discoverWorkspace(workspace); return decodeWorkspaceBilling(await store.read(workspaceRecordKey.billing(workspace)), workspace); },
+        capacity: async workspace => { await discoverWorkspace(workspace); return decodeWorkspaceCapacity(await store.read(workspaceRecordKey.capacity(workspace)), workspace, store.now()); },
+        plans: async (owner, workspace) => { await assertOwner(owner); await discoverWorkspace(workspace);
+          return decodeIntendedPlans(await store.read(workspaceRecordKey.plans(owner, workspace)), owner, workspace, store.now()); },
+        receivedPlans: async owner => { await assertOwner(owner); return copies(await store.read(workspaceRecordKey.copies(owner)), owner, store.now()); },
+        receipts: async owner => { await assertOwner(owner);
+          const value = await store.read(workspaceRecordKey.receipts(owner));
+          try { return await decodePlanReceipts(value, owner, store.now()); } catch { return changed(); } },
+        planSource: async (owner, workspace) => { await assertOwner(owner); await discoverWorkspace(workspace); return trust.planSource(store, owner, workspace); },
+        activationAvailable: async workspace => { await discoverWorkspace(workspace); return await trust.activationAvailable(store, workspace) === true; },
+        reconciliation: async (owner, workspace) => { await assertOwner(owner); await discoverWorkspace(workspace);
+          return decodeReconciliation(await store.read(workspaceRecordKey.reconciliation(owner, workspace)), owner, workspace); },
+        reconciliationSource: async (owner, workspace) => { await assertOwner(owner); await discoverWorkspace(workspace);
+          if (!trust.reconciliationSource) throw new ProfessionalAccessError('unavailable');
+          return trust.reconciliationSource(store, owner, workspace); },
+        previsitRecords: async (owner, workspace, fields) => { await assertOwner(owner); await discoverWorkspace(workspace);
+          if (!trust.previsitRecords) throw new ProfessionalAccessError('unavailable');
+          return trust.previsitRecords(store, owner, workspace, fields); },
+        saveReconciliation: async value => {
+          if (!authorizedOwner || !authorizedWorkspace) return changed();
+          const next = await decodeReconciliation(structuredClone(value), authorizedOwner, authorizedWorkspace);
+          const previous = await tx.reconciliation!(authorizedOwner, authorizedWorkspace); prefix(previous, next);
+          await store.write(workspaceRecordKey.reconciliation(authorizedOwner, authorizedWorkspace), next);
+        },
+        saveBilling: async value => {
+          const captured = structuredClone(value); assertWorkspace(captured.workspaceId);
+          const previous = await tx.billing(captured.workspaceId); const next = await decodeWorkspaceBilling(captured, captured.workspaceId);
+          if (previous.customerId !== next.customerId || previous.subscriptionId !== next.subscriptionId) return changed();
+          prefix(previous.receipts, next.receipts); await store.write(workspaceRecordKey.billing(captured.workspaceId), next);
+        },
+        saveGrants: async value => {
+          const captured = structuredClone(value);
+          const previous = await tx.grants(captured.ownerId); const next = await decodeGrants(captured, captured.ownerId, store.now());
+          prefix(previous.events, next.events); await store.write(workspaceRecordKey.grants(captured.ownerId), next);
+        },
+        saveCapacity: async value => {
+          const captured = structuredClone(value);
+          const previous = await tx.capacity(captured.workspaceId); const next = await decodeWorkspaceCapacity(captured, captured.workspaceId, store.now());
+          prefix(previous.events, next.events); await store.write(workspaceRecordKey.capacity(captured.workspaceId), next);
+        },
+        savePlans: async value => {
+          const captured = structuredClone(value);
+          if (!captured.length || !authorizedOwner || !authorizedWorkspace) return changed();
+          const previous = await tx.plans(authorizedOwner, authorizedWorkspace);
+          const next = await decodeIntendedPlans(captured, authorizedOwner, authorizedWorkspace, store.now());
+          prefix(previous, next); await store.write(workspaceRecordKey.plans(authorizedOwner, authorizedWorkspace), next);
+        },
+        saveOwnerReceiptAndCopy: async (value, plan) => {
+          value = structuredClone(value); plan = structuredClone(plan);
+          const previous = await tx.receipts(plan.ownerId); const retained = await tx.receivedPlans(plan.ownerId);
+          const next = await decodePlanReceipts(value, plan.ownerId, store.now()); await validateIntendedPlan(plan, store.now()); prefix(previous, next);
+          if (!next.some(receipt => receipt.kind === 'received' && receipt.planId === plan.id && receipt.planHash === plan.contentHash)) return changed();
+          const existing = retained.find(copy => copy.id === plan.id);
+          if (existing && canonical(existing) !== canonical(plan)) return changed();
+          const nextCopies = existing ? retained : [...retained, plan];
+          if (next.some(receipt => !nextCopies.some(copy => copy.id === receipt.planId && copy.contentHash === receipt.planHash && copy.createdAt <= receipt.recordedAt)) ||
+              nextCopies.some(copy => !next.some(receipt => receipt.kind === 'received' && receipt.planId === copy.id && receipt.planHash === copy.contentHash))) return changed();
+          // Both keys are staged and canonically read back in ONE atomic commit.
+          await store.write(workspaceRecordKey.receipts(plan.ownerId), next);
+          await store.write(workspaceRecordKey.copies(plan.ownerId), nextCopies);
+        },
+        beforeCommit: store.beforeCommit,
+      };
+      // Scope constrains internal storage, not permission. Existing services own
+      // role/purpose admission and final checks; webhook verification is separate.
+      // A workspace-only transaction cannot read any owner aggregate.
+      const protectedTx = snapshotWorkspaceTransaction(tx);
+      const result = await work(protectedTx);
+      store.beforeCommit(async () => { if (authorizedOwner) await protectedTx.lifecycle(authorizedOwner); });
+      return result;
+    });
+  }
+}

+export async function decodeGrants(value: unknown, ownerId: string, latestTime = 8.64e15): Promise<GrantState> {

+      !Array.isArray(candidate.events) || candidate.events.length > 1000) return fail();

+  for (const event of candidate.events) {
+    if (!event || Object.keys(event).sort().join(',') !== 'actorId,contentHash,grant,id,kind,ownerId,previousGrantId,previousHash,recordedAt,schemaVersion,sequence,source' || event.recordedAt > latestTime) return fail();

+      healthEpoch: event.grant?.healthEpoch ?? 1, consentReceiptId: event.grant?.consentReceiptId ?? 'historical-revoke',
+      current: true, adult: true, deletionPending: false }, { id: event.id, expectedRevision: state.revision,
+      kind: event.kind, previousGrantId: event.previousGrantId, grant: event.grant, confirmed: true }, event.recordedAt);
+    if (canonical(state.events.at(-1)) !== canonical(event)) return fail();
+  }
+  if (canonical(state) !== canonical(candidate)) return fail();
+  return state;

+type AcceptCommand = { kind: 'accept'; context: string; workspaceId: string; command: DeclineCommand['command'] & { capacityId: string; expectedCapacityRevision: number } };
+type ReceiveCommand = { kind: 'receive'; context: string; workspaceId: string; command: PlanCommand };
+type EndCommand = { kind: 'end'; context: string; workspaceId: string; command: { id: string; capacityId: string; grantId: string;
+  expectedGrantRevision: number; expectedCapacityRevision: number; confirmed: true } };
+const record = (value: unknown): value is Record<string, unknown> => !!value && typeof value === 'object' && !Array.isArray(value);
+const exact = (value: Record<string, unknown>, keys: string) => Object.keys(value).sort().join(',') === keys;
+const id = (value: unknown): value is string => typeof value === 'string' && /^[A-Za-z0-9][A-Za-z0-9:_-]{0,127}$/u.test(value);
+const uuid = (value: unknown): value is string => typeof value === 'string' && /^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/u.test(value);
+const hash = (value: unknown): value is string => typeof value === 'string' && /^[0-9a-f]{64}$/u.test(value);
+const number = (value: unknown): value is number => typeof value === 'number' && Number.isSafeInteger(value) && value >= 0;
+const list = (value: unknown, allowed: readonly string[]): value is string[] => Array.isArray(value) && new Set(value).size === value.length && value.every(item => allowed.includes(item));
+function validPlan(value: unknown): value is PlanSummary {
+  if (!record(value) || !exact(value, 'issuedAt,latestResponse,planHash,planId,sequence') || !uuid(value.planId) || !hash(value.planHash) ||
+      !number(value.sequence) || value.sequence < 1 || !number(value.issuedAt) || value.issuedAt > 8.64e15) return false;
+  const latest = value.latestResponse;
+  return latest === null || record(latest) && exact(latest, 'kind,recordedAt') && ['received', 'acknowledged', 'declined'].includes(latest.kind as string) &&
+    number(latest.recordedAt) && latest.recordedAt >= value.issuedAt && latest.recordedAt <= 8.64e15;
+}
+function validConnection(value: unknown): value is Connection {
+  return record(value) && exact(value, 'expectedCapacityRevision,expectedGrantRevision,expired,expiresAt,fields,grantId,permissions,purposes,workspaceId') &&
+    id(value.workspaceId) && id(value.grantId) && number(value.expectedGrantRevision) && value.expectedGrantRevision > 0 && number(value.expectedCapacityRevision) &&
+    number(value.expiresAt) && value.expiresAt <= 8.64e15 && typeof value.expired === 'boolean' &&
+    list(value.fields, GRANT_FIELDS) && value.fields.length > 0 && list(value.purposes, GRANT_PURPOSES) && value.purposes.length > 0 &&
+    list(value.permissions, ['read', 'plan_write', 'export']) && value.permissions.includes('read');
