+import { useLayoutEffect, useMemo, useRef, useState, useSyncExternalStore } from 'react';
+
+import { assertAccountIdentityGeneration, captureAccountIdentityGeneration, runAccountGenerationOperation } from '@/lib/auth/accountGeneration';
+import { activeHealthProcessingLeaseSnapshot, subscribeActiveHealthProcessingLeaseChanges } from '@/lib/consent/healthProcessingEpoch';
+import { beginDataRightsConfirmation, beginDataRightsOperation, IDLE_DATA_RIGHTS_STATE, type DataRightsState, type PendingDataRightsAction } from './youDataRightsState';
+
+type Confirmation = Readonly<{ scope: object; state: DataRightsState; generation: number }>;
+
+/** Exact rendered confirmation, original account and original consent view.
+ * Export has a separate flight so it can never block deletion priority. */
+export function useDataRightsConfirmation(ownerUserId: string | null, source?: unknown) {
+  const identity = captureAccountIdentityGeneration();
+  const health = useSyncExternalStore(subscribeActiveHealthProcessingLeaseChanges, activeHealthProcessingLeaseSnapshot, () => null);
+  const scope = useMemo(() => Object.freeze({ ownerUserId, identity, health, source }), [ownerUserId, identity, health, source]);
+  const liveScope = useRef<object | null>(scope);
+  const live = useRef<Confirmation | null>(null);
+  const [rendered, setRendered] = useState<Confirmation | null>(null);
+  useLayoutEffect(() => {
+    liveScope.current = scope;
+    return () => { if (liveScope.current === scope) { liveScope.current = null; live.current = null; } };
+  }, [scope]);
+  function current() {
+    try { assertAccountIdentityGeneration(identity); } catch { return false; }
+    return liveScope.current === scope && activeHealthProcessingLeaseSnapshot() === health;
+  }
+  function open(action: PendingDataRightsAction) {
+    if (!current() || live.current) return;
+    let generation: number | null = null;
+    // Capture only; this operation never retains a lease across confirmation.
+    void runAccountGenerationOperation((lease) => { generation = lease.generation; }).catch(() => {});
+    if (generation === null) return;
+    const confirmation = Object.freeze({ scope, state: beginDataRightsConfirmation(IDLE_DATA_RIGHTS_STATE, action)!, generation });
+    live.current = confirmation;
+    setRendered(confirmation);
+  }
+  function cancel(confirmation: Confirmation | null) {
+    if (!current() || !confirmation || live.current !== confirmation || confirmation.state.phase !== 'confirming') return;
+    live.current = null;
+    setRendered(null);
+  }
+  function begin(confirmation: Confirmation | null, action: PendingDataRightsAction) {
+    if (!current() || !confirmation || live.current !== confirmation || !ownerUserId) return null;
+    const running = beginDataRightsOperation(confirmation.state, action, 1);
+    if (!running) return null;
+    const operation = Object.freeze({ ...confirmation, state: running });
+    live.current = operation; // Same-turn latch before invoking any service.
+    setRendered(operation);
+    return Object.freeze({
+      expectedInitiator: { generation: confirmation.generation, userId: ownerUserId },
+      isCurrent: current,
+      finish: () => { if (live.current === operation) { live.current = null; setRendered(null); } },
+    });
+  }
+  return { busy: rendered?.scope === scope && rendered.state.phase === 'running', confirmation: rendered?.scope === scope ? rendered : null, open, cancel, begin };

+export function buildAppointmentBrief(
+  products: readonly BriefProduct[],
+  selection: AppointmentBriefSelection,
+  preparedOn: string,
+  records: readonly BriefRecordedItem[] = [],
+): string {
+  const invalid = () => {
+    throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+  };
+  const recordIds = selection.recordIds ?? [];
+  if (recordIds.length > 300 || new Set(recordIds).size !== recordIds.length) invalid();
+  const selectedRecords = recordIds.map((id) => {
+    const record = records.find((candidate) => candidate.id === id && candidate.available);
+    if (!record) return invalid();
+    return record;
+  });

+    !validLocalDate(preparedOn) ||
+    typeof selection.includeDates !== 'boolean' ||
+    (selection.includePhoto !== undefined && typeof selection.includePhoto !== 'boolean') ||
+    selection.productIds.length > 1000 ||
+    new Set(selection.productIds).size !== selection.productIds.length
+  )
+    invalid();

+    if (
+      typeof value !== 'string' ||
+      value.length > 2000 ||
+      /[\p{Cf}\p{Cs}\u0000-\u0008\u000b\u000c\u000e-\u001f\u007f-\u009f]/u.test(value)
+    )
+      invalid();
+  }
+  if (
+    !selection.productIds.length &&
+    !recordIds.length &&
+    !selection.observations.trim() &&
+    !selection.questions.trim()
+    && !selection.includePhoto
+  )
+    invalid();
+  const chosen = selection.productIds.map((id) => {
+    const product = products.find((candidate) => candidate.id === id);
+    if (!product) return invalid();
+    return product;
+  });

+ * Invoke the platform synchronously here, but await its sheet outside the queue:
+ * withdrawal/deletion must not wait for the recipient dialog to close. */
+export async function dispatchWithBriefSourceProof<T>(
+  proof: BriefSourceProof, lease: HealthDataWriteLease,
+  assertReviewCurrent: () => void, dispatch: () => Promise<T>, retirement?: AbortSignal,
+): Promise<T> {
+  const accepted = proofs.get(proof);
+  if (!accepted || accepted.lease !== lease) throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+  const guard = () => { assertOriginalHealthDataWriteLease(lease); assertReviewCurrent(); if (retirement?.aborted) throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE'); };
+  guard();
+  const { updatePrivateItemWithSources } = await import('@/lib/storage/privateKV');
+  guard();
+  let pending: Promise<T> | undefined;
+  await updatePrivateItemWithSources(accepted.anchor, [...accepted.bytes.keys()], values => {
+    guard();
+    if ([...accepted.bytes].some(([key, value]) => (values.get(key) ?? null) !== value)) {
+      throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+    }
+    return values.get(accepted.anchor) ?? null;
+  }, guard, undefined, () => {
+    guard();
+    if (proofs.get(proof) !== accepted) throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+    proofs.delete(proof);
+    pending = dispatch();
+    // Observe rejection immediately even if the storage guard itself retires.
+    void pending.catch(() => undefined);
+  });
+  guard();
+  if (!pending) throw new Error('APPOINTMENT_BRIEF_SOURCE_UNAVAILABLE');
+  let rejectRetired!: (reason: unknown) => void;
+  const retired = new Promise<never>((_resolve, reject) => { rejectRetired = reject; });
+  const checkRetirement = () => { try { guard(); } catch (error) { rejectRetired(error); } };
+  const stop = subscribeActiveHealthProcessingLeaseChanges(checkRetirement);
+  retirement?.addEventListener('abort', checkRetirement, { once: true });
+  const expiry = lease.expiresAt === null ? null : setTimeout(checkRetirement, Math.max(0, lease.expiresAt - Date.now()) + 1);
+  try {
+    // Covers a boundary between platform invocation and listener installation.
+    checkRetirement();
+    const result = await Promise.race([pending, retired]);
+    guard();
+    return result;
+  } finally {
+    stop(); retirement?.removeEventListener('abort', checkRetirement);
+    if (expiry !== null) clearTimeout(expiry);
+  }
+}
