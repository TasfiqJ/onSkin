+  if (fields.attempted !== null && fields.attempted.input.operationId !== operationId) invalid();
+  const body = { id: uuid(row.id), revision: integer(row.revision, 1), createdAt, updatedAt, operationId, fields };
+  return Object.freeze({ ...body, contentHash: bytesToHex(sha256(utf8ToBytes(JSON.stringify(body)))) });
+}
+export function decodeManualIntakeDraftCollection(raw: string | null, ownerUserId: string, processingEpoch?: number): ManualIntakeDraftCollection {
+  nonempty(ownerUserId, 128);
+  if (raw === null) return Object.freeze({ schemaVersion: 1, provenance: 'owner_unconfirmed_manual_intake', ownerUserId,
+    processingEpoch: integer(processingEpoch, 1), drafts: Object.freeze([]) });
+  if (raw.length > MANUAL_INTAKE_DRAFT_MAX_BYTES || utf8ToBytes(raw).length > MANUAL_INTAKE_DRAFT_MAX_BYTES) invalid();
+  let parsed: unknown;
+  try { parsed = JSON.parse(raw) as unknown; } catch { invalid(); }
+  const row = record(parsed, ['schemaVersion', 'provenance', 'ownerUserId', 'processingEpoch', 'drafts']);
+  if (row.schemaVersion !== 1 || row.provenance !== 'owner_unconfirmed_manual_intake') invalid();
+  if (row.ownerUserId !== ownerUserId || (processingEpoch !== undefined && row.processingEpoch !== processingEpoch)) throw new Error(MANUAL_INTAKE_DRAFT_AUTHORITY);
+  const epoch = integer(row.processingEpoch, 1);
+  if (!Array.isArray(row.drafts) || row.drafts.length > MANUAL_INTAKE_DRAFT_LIMIT) invalid();
+  const drafts = row.drafts.map(entry => {
+    const checked = record(entry, ['id', 'revision', 'createdAt', 'updatedAt', 'operationId', 'fields', 'contentHash']);
+    const { contentHash, ...body } = checked;
+    const draft = makeManualIntakeDraft(body as Omit<ManualIntakeDraft, 'contentHash'>);
+    if (draft.contentHash !== contentHash) invalid();
+    return draft;
+  });
+  if (new Set(drafts.map(draft => draft.id)).size !== drafts.length ||
+      new Set(drafts.map(draft => draft.operationId)).size !== drafts.length) invalid();
+  return Object.freeze({ schemaVersion: 1, provenance: 'owner_unconfirmed_manual_intake', ownerUserId,
+    processingEpoch: epoch, drafts: Object.freeze(drafts) });
+}

+/** Explicit Save Draft or Add write-ahead snapshot. No per-keystroke writes. */
+export async function saveManualIntakeDraft(lease: HealthDataWriteLease, request: Readonly<{
+  id: string;
+  expected: ManualIntakeDraft | null;
+  operationId: string;
+  fields: ManualIntakeDraftFields;
+}>, assertCallerCurrent: () => void): Promise<ManualIntakeDraft> {
+  const assertCurrent = () => { assertOriginalHealthDataWriteLease(lease); assertCallerCurrent(); };
+  assertCurrent();
+  const fields = decodeManualIntakeDraftFields(request.fields);
+  // Snapshot every caller value before the first await, including the CAS proof.
+  const expected = request.expected ? JSON.stringify(request.expected) : null;
+  const id = request.id, operationId = request.operationId;
+  let result: ManualIntakeDraft | undefined;
+  await updatePrivateItemWithSources(MANUAL_INTAKE_DRAFT_KEY, MANUAL_INTAKE_DRAFT_AUTHORIZATION_KEYS, values => {
+    assertCurrent();
+    const raw = values.get(MANUAL_INTAKE_DRAFT_KEY) ?? null;
+    const collection = decodeManualIntakeDraftCollection(raw, lease.ownerUserId, lease.epoch);
+    const existing = collection.drafts.find(row => row.id === id);
+    if ((existing ? JSON.stringify(existing) : null) !== expected ||
+        (existing && existing.operationId !== operationId) ||
+        collection.drafts.some(row => row.id !== id && row.operationId === operationId)) throw new Error(MANUAL_INTAKE_DRAFT_CONFLICT);
+    // Once Add has been attempted, a retry must keep the entire original
+    // snapshot. Even edits to raw fields could falsely describe what was saved.
+    if (existing?.fields.attempted && JSON.stringify(existing.fields) !== JSON.stringify(fields)) throw new Error(MANUAL_INTAKE_DRAFT_CONFLICT);
+    if (existing && JSON.stringify(existing.fields) === JSON.stringify(fields)) {
+      result = existing;
+      return raw;
+    }
+    if (!existing && collection.drafts.length >= MANUAL_INTAKE_DRAFT_LIMIT) throw new Error(MANUAL_INTAKE_DRAFT_CONFLICT);
+    const now = new Date().toISOString();
+    result = makeManualIntakeDraft({ id, revision: (existing?.revision ?? 0) + 1,
+      createdAt: existing?.createdAt ?? now, updatedAt: now, operationId, fields });
+    const next = JSON.stringify({ ...collection, drafts: [...collection.drafts.filter(row => row.id !== id), result] });
+    decodeManualIntakeDraftCollection(next, lease.ownerUserId, lease.epoch);
+    return next;
+  }, assertCurrent, undefined, undefined, manualIntakeDraftAuthorization(lease));
+  assertCurrent();
+  return result!;
+}

+export async function discardManualIntakeDraft(lease: HealthDataWriteLease, expected: ManualIntakeDraft, assertCallerCurrent: () => void): Promise<void> {
+  const assertCurrent = () => { assertOriginalHealthDataWriteLease(lease); assertCallerCurrent(); };

+  await updatePrivateItemWithSources(MANUAL_INTAKE_DRAFT_KEY, MANUAL_INTAKE_DRAFT_AUTHORIZATION_KEYS, values => {
+    assertCurrent();

+    const collection = decodeManualIntakeDraftCollection(raw, lease.ownerUserId, lease.epoch);
+    assertExactManualIntakeDraft(raw, lease, JSON.parse(accepted) as ManualIntakeDraft);
+    return JSON.stringify({ ...collection, drafts: collection.drafts.filter(row => row.id !== acceptedId) });
+  }, assertCurrent, undefined, undefined, manualIntakeDraftAuthorization(lease));
+  assertCurrent();
+}

+import { categoryLabel } from './categories';
+import { validLocalDate } from './freshness';
+import type { ShelfProduct } from './store';
+
+export type ComparableProduct = Pick<
+  ShelfProduct,
+  | 'id'
+  | 'name'
+  | 'brand'
+  | 'category'
+  | 'status'
+  | 'isOpened'
+  | 'openedAt'
+  | 'paoMonths'
+  | 'paoSource'
+  | 'expiryDate'
+  | 'expirySource'
+  | 'addedVia'
+  | 'ingredients'
+>;
+
+function facts(product: ComparableProduct): Record<string, string> {
+  return {
+    Brand: product.brand || 'Not recorded',
+    Category: categoryLabel(product.category) ?? 'Not recorded',
+    Status: { active: 'Active', finished: 'Finished', discarded: 'Discarded' }[product.status],
+    'Opened date': !product.isOpened ? 'Unopened' : (validLocalDate(product.openedAt) ?? 'Unknown'),
+    'Printed expiry':
+      product.expirySource === 'printed'
+        ? (validLocalDate(product.expiryDate) ?? 'Not recorded')
+        : 'Not recorded',
+    'Period after opening':
+      product.paoMonths !== null &&
+      (product.paoSource === 'label' || product.paoSource === 'catalog')
+        ? `${product.paoMonths} months · ${product.paoSource}`
+        : 'Not recorded',
+    'Added through': {
+      manual: 'Manual entry',
+      onboarding: 'Onboarding',
+      barcode: 'Barcode',
+      search: 'Catalog search',
+      ocr: 'Label text',
+    }[product.addedVia],
+  };
+}
+
+/** Compare recorded text only. No synonym inference, completeness claim,
+ * compatibility, clinical ranking, quota, persistence or provider lookup. */
+export function compareSavedProducts<T extends ComparableProduct>(
+  products: readonly T[],
+  leftId: string | null,
+  rightId: string | null,
+) {
+  if (!leftId || !rightId || leftId === rightId) return null;
+  const left = products.find((product) => product.id === leftId);
+  const right = products.find((product) => product.id === rightId);
+  if (!left || !right) return null;
+  const leftFacts = facts(left);
+  const rightFacts = facts(right);
+  const ingredientKey = (text: string) => text.trim().replace(/\s+/gu, ' ').toLowerCase();
+  const leftIngredients = new Map<string, string>();
+  for (const text of left.ingredients) {
+    const key = ingredientKey(text);
+    if (key && !leftIngredients.has(key)) leftIngredients.set(key, text.trim());
+  }
+  const rightIngredients = new Set(
+    right.ingredients.filter((text) => text.trim()).map(ingredientKey),
+  );
+  // Keep the first recorded spelling as well as the source's order.
+  const sharedIngredients =
+    leftIngredients.size && rightIngredients.size
+      ? [...leftIngredients].filter(([key]) => rightIngredients.has(key)).map(([, text]) => text)
+      : null;
+  return {
+    left,
+    right,
+    sharedIngredients,
+    rows: Object.entries(leftFacts).map(([label, value]) => ({
+      label,
+      left: value,
+      right: rightFacts[label]!,
+    })),
+  };
+}
