+import { categoryLabel } from './categories';
+import { validLocalDate } from './freshness';
+import type { ShelfProduct } from './store';
+
+export type ComparableProduct = Pick<
+  ShelfProduct,
+  | 'id'
+  | 'name'
+  | 'brand'
+  | 'category'
+  | 'status'
+  | 'isOpened'
+  | 'openedAt'
+  | 'paoMonths'
+  | 'paoSource'
+  | 'expiryDate'
+  | 'expirySource'
+  | 'addedVia'
+  | 'ingredients'
+>;
+
+function facts(product: ComparableProduct): Record<string, string> {
+  return {
+    Brand: product.brand || 'Not recorded',
+    Category: categoryLabel(product.category) ?? 'Not recorded',
+    Status: { active: 'Active', finished: 'Finished', discarded: 'Discarded' }[product.status],
+    'Opened date': !product.isOpened ? 'Unopened' : (validLocalDate(product.openedAt) ?? 'Unknown'),
+    'Printed expiry':
+      product.expirySource === 'printed'
+        ? (validLocalDate(product.expiryDate) ?? 'Not recorded')
+        : 'Not recorded',
+    'Period after opening':
+      product.paoMonths !== null &&
+      (product.paoSource === 'label' || product.paoSource === 'catalog')
+        ? `${product.paoMonths} months · ${product.paoSource}`
+        : 'Not recorded',
+    'Added through': {
+      manual: 'Manual entry',
+      onboarding: 'Onboarding',
+      barcode: 'Barcode',
+      search: 'Catalog search',
+      ocr: 'Label text',
+    }[product.addedVia],
+  };
+}
+
+/** Compare recorded text only. No synonym inference, completeness claim,
+ * compatibility, clinical ranking, quota, persistence or provider lookup. */
+export function compareSavedProducts<T extends ComparableProduct>(
+  products: readonly T[],
+  leftId: string | null,
+  rightId: string | null,
+) {
+  if (!leftId || !rightId || leftId === rightId) return null;
+  const left = products.find((product) => product.id === leftId);
+  const right = products.find((product) => product.id === rightId);
+  if (!left || !right) return null;
+  const leftFacts = facts(left);
+  const rightFacts = facts(right);
+  const ingredientKey = (text: string) => text.trim().replace(/\s+/gu, ' ').toLowerCase();
+  const leftIngredients = new Map<string, string>();
+  for (const text of left.ingredients) {
+    const key = ingredientKey(text);
+    if (key && !leftIngredients.has(key)) leftIngredients.set(key, text.trim());
+  }
+  const rightIngredients = new Set(
+    right.ingredients.filter((text) => text.trim()).map(ingredientKey),
+  );
+  // Keep the first recorded spelling as well as the source's order.
+  const sharedIngredients =
+    leftIngredients.size && rightIngredients.size
+      ? [...leftIngredients].filter(([key]) => rightIngredients.has(key)).map(([, text]) => text)
+      : null;
+  return {
+    left,
+    right,
+    sharedIngredients,
+    rows: Object.entries(leftFacts).map(([label, value]) => ({
+      label,
+      left: value,
+      right: rightFacts[label]!,
+    })),
+  };
+}

+    assertReviewCurrent();

+      if (!review.diaryAccess) throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+      review.diaryAccess.assertCurrent();
+    }
+  };
+  assertCurrent();
+  return runHealthDataOperation(review.lease.ownerUserId, async (lease) => {
+    assertCurrent();

+      message: review.text,
+    }));
+    lease.assertCurrent();
+    assertCurrent();

+  });

+function normalizeCompletionLog(value: unknown, strictCurrentSchema = false): Log {
+  if (!isRecord(value)) throw completionLogError(COMPLETION_LOG_INVALID);
+  const out: Log = {};
+  for (const [date, keys] of Object.entries(value)) {
+    const normalizedDate = normalizeLocalDateISO(date);
+    if (!normalizedDate || !Array.isArray(keys)) {
+      throw completionLogError(COMPLETION_LOG_INVALID);
+    }
+    if (strictCurrentSchema && normalizedDate !== date) {
+      throw completionLogError(COMPLETION_LOG_INVALID);
+    }
+    const normalizedKeys: string[] = [];
+    for (const key of keys) {
+      const normalizedKey = normalizeStepKey(key);
+      if (!normalizedKey) throw completionLogError(COMPLETION_LOG_INVALID);
+      if (
+        strictCurrentSchema &&
+        (normalizedKey !== key || normalizedKeys.includes(normalizedKey))
+      ) {
+        throw completionLogError(COMPLETION_LOG_INVALID);
+      }
+      if (!normalizedKeys.includes(normalizedKey)) normalizedKeys.push(normalizedKey);
+    }
+    if (strictCurrentSchema && normalizedKeys.length === 0) {
+      throw completionLogError(COMPLETION_LOG_INVALID);
+    }
+    if (normalizedKeys.length > 0) {
+      out[normalizedDate] = [...new Set([...(out[normalizedDate] ?? []), ...normalizedKeys])];
+    }

+  return out;

+function normalizeCompletedDays(value: unknown, days: Log): Set<string> {

+      throw completionLogError(COMPLETION_LOG_INVALID);
+    }
+    out.add(normalizedDate);
+  }
+  return out;

+export function decodeCompletionLog(raw: string | null, ownerUserId: string | null, allowOptionalChanges = true, allowActualRoutine = true): CompletionState {
+  if (raw === null) {
+    return { days: {}, completedDays: new Set(), sync: emptyCompletionSyncState() };
+  }
+  let parsed: unknown;
+  try {
+    parsed = JSON.parse(raw) as unknown;
+  } catch {
+    throw completionLogError(COMPLETION_LOG_INVALID);
+  }
+  if (!isRecord(parsed)) throw completionLogError(COMPLETION_LOG_INVALID);
+
+  if (parsed.version === 5) {
+    if (!hasOwn(parsed, 'actualRoutine')) throw completionLogError(COMPLETION_LOG_UNSUPPORTED_VERSION);
+    if (!allowActualRoutine) throw completionLogError(COMPLETION_LOG_INVALID);
+    const owner = ownerUserId;
+    if (owner === null) throw completionLogError(COMPLETION_LOG_INVALID);
+    const envelope = decodeActualRoutineEnvelope(raw, owner)!;
+    return { ...decodeCompletionLog(envelope.completionBytes, owner, true, false), actualRoutine: envelope.actualRoutine };
+  }

+  if (parsed.version === 4) {

+      throw completionLogError(COMPLETION_LOG_UNSUPPORTED_VERSION);
+    }
+    if (!allowOptionalChanges || utf8ToBytes(raw).length > OPTIONAL_CHANGE41_MAX_BYTES) {
+      throw completionLogError(COMPLETION_LOG_INVALID);
+    }

+    if (owner === null) throw completionLogError(COMPLETION_LOG_INVALID);
+    const envelope = decodeOptionalChangeEnvelope41(parsed, owner);
+    return { ...decodeCompletionLog(envelope.completionBytes, owner, false, false), changes: envelope.changes };
+  }
+
+  if (!hasOwn(parsed, 'version')) {
+    return {
+      days: normalizeCompletionLog(parsed),
+      completedDays: new Set(),
+      sync: emptyCompletionSyncState(),
+    };
+  }
+  if (parsed.version === 1) {
+    if (Object.keys(parsed).length !== 2 || !hasOwn(parsed, 'days') || !isRecord(parsed.days)) {
+      throw completionLogError(COMPLETION_LOG_INVALID);
+    }
+    return {
+      days: normalizeCompletionLog(parsed.days, true),
+      // A v1 row proves step taps, not that the then-scheduled PM routine was complete.
+      completedDays: new Set(),
+      sync: emptyCompletionSyncState(),
+    };
+  }
+  if (parsed.version === 2) {
+    if (
+      Object.keys(parsed).length !== 3 ||
+      !hasOwn(parsed, 'days') ||
+      !hasOwn(parsed, 'completedDays') ||
+      !isRecord(parsed.days)
+    ) {
+      throw completionLogError(COMPLETION_LOG_INVALID);
+    }
+    const days = normalizeCompletionLog(parsed.days, true);
+    return {
+      days,
+      completedDays: normalizeCompletedDays(parsed.completedDays, days),
+      // Historical v2 rows have neither an original timestamp nor durable
+      // server identity. Never invent remote operations for them.
+      sync: emptyCompletionSyncState(),
+    };
+  }
+  if (parsed.version !== SCHEMA_VERSION) {
+    if (
+      typeof parsed.version === 'number' &&
+      Number.isSafeInteger(parsed.version) &&
+      parsed.version > SCHEMA_VERSION
+    ) {
+      throw completionLogError(COMPLETION_LOG_UNSUPPORTED_VERSION);
+    }
+    throw completionLogError(COMPLETION_LOG_INVALID);
+  }
+  if (
+    Object.keys(parsed).length !== 4 ||
+    !hasOwn(parsed, 'days') ||
+    !hasOwn(parsed, 'completedDays') ||
+    !hasOwn(parsed, 'sync') ||
+    !isRecord(parsed.days)
+  ) {
+    throw completionLogError(COMPLETION_LOG_INVALID);
+  }
+  const days = normalizeCompletionLog(parsed.days, true);
+  const completedDays = normalizeCompletedDays(parsed.completedDays, days);
+  try {
+    const sync = decodeCompletionSyncState(parsed.sync, {
+      hasCompletedStep: (completedDate, step) => days[completedDate]?.includes(step) === true,
+      hasCompletedDay: (completedDate) => completedDays.has(completedDate),
+    });
+    return { days, completedDays, sync };
+  } catch {
+    throw completionLogError(COMPLETION_LOG_INVALID);
+  }

+export function makeOptionalChangeDraft(value: Omit<OptionalChangeDraft, 'contentHash'>): OptionalChangeDraft {
+  const hasProof = record41(value) && Object.hasOwn(value, 'eventPredecessorProof');
+  const row = record(value, ['id', 'revision', 'createdAt', 'updatedAt', 'fields', ...(hasProof ? ['eventPredecessorProof'] : [])]);
+  const createdAt = timestamp(row.createdAt), updatedAt = timestamp(row.updatedAt);
+  if (updatedAt < createdAt) invalid();

+    fields: decodeOptionalChangeDraftFields(row.fields),
+    ...(hasProof ? { eventPredecessorProof: hash(row.eventPredecessorProof) } : {}) };
+  return Object.freeze({ ...body, contentHash: hashOptionalChange41(body) });

+export function decodeOptionalChangeDraftCollection(raw: string | null, ownerUserId: string, processingEpoch?: number): OptionalChangeDraftCollection {
+  owner(ownerUserId);
+  if (raw === null) return Object.freeze({ schemaVersion: 1, provenance: 'owner_unconfirmed_optional_change', ownerUserId,
+    processingEpoch: integer(processingEpoch, 1), drafts: Object.freeze([]) });
+  if (raw.length > OPTIONAL_CHANGE_DRAFT_MAX_BYTES || utf8ToBytes(raw).length > OPTIONAL_CHANGE_DRAFT_MAX_BYTES) invalid();
+  let parsed: unknown;
+  try { parsed = JSON.parse(raw) as unknown; } catch { invalid(); }
+  const row = record(parsed, ['schemaVersion', 'provenance', 'ownerUserId', 'processingEpoch', 'drafts']);
+  if ((row.schemaVersion !== 1 && row.schemaVersion !== 3) || row.provenance !== 'owner_unconfirmed_optional_change') invalid();
+  if (row.ownerUserId !== ownerUserId || (processingEpoch !== undefined && row.processingEpoch !== processingEpoch)) throw new Error(OPTIONAL_CHANGE_DRAFT_AUTHORITY);
+  const epoch = integer(row.processingEpoch, 1);
+  if (!Array.isArray(row.drafts) || row.drafts.length > OPTIONAL_CHANGE_DRAFT_LIMIT) invalid();
+  const drafts = row.drafts.map(value => {
+    const hasProof = record41(value) && Object.hasOwn(value, 'eventPredecessorProof');
+    if (hasProof && row.schemaVersion !== 3) invalid();
+    const entry = record(value, ['id', 'revision', 'createdAt', 'updatedAt', 'fields', 'contentHash', ...(hasProof ? ['eventPredecessorProof'] : [])]);
+    const { contentHash, ...body } = entry;
+    const draft = makeOptionalChangeDraft(body as Omit<OptionalChangeDraft, 'contentHash'>);
+    if (draft.contentHash !== hash(contentHash)) invalid();
+    return draft;
+  });
+  if (new Set(drafts.map(row => row.id)).size !== drafts.length ||
+      new Set(drafts.map(row => row.fields.eventId)).size !== drafts.length) invalid();
+  return Object.freeze({ schemaVersion: row.schemaVersion, provenance: 'owner_unconfirmed_optional_change', ownerUserId,
+    processingEpoch: epoch, drafts: Object.freeze(drafts) });

+function sameDraftFacts(fields: OptionalChangeDraftFields, event: OptionalChangeEvent41): boolean {
+  return event.id === fields.eventId && event.kind === fields.kind && event.targetEventId === fields.targetEventId &&
+    event.observedOn === fields.observedOn && event.category === fields.category && event.text === fields.text &&
+    event.expectedRevision === fields.sourceRevision;
+}
+/** Exact predecessor proof supplements, never replaces, complete factual equality.
+ * Legacy compatibility succeeds only when its original proof already matches. */
+export function optionalChangeDraftEventMatches(fields: OptionalChangeDraftFields, event: OptionalChangeEvent41, eventPredecessorProof = fields.sourceProof): boolean {
+  return sameDraftFacts(fields, event) && event.sourceProof === eventPredecessorProof;
+}
+export function reconcileOptionalChangeDraft(draft: OptionalChangeDraft, snapshot: OptionalChangeSnapshot41): OptionalChangeDraftRecovery {
+  const fields = draft.fields;
+  if (fields.targetEventId !== null) {
+    const target = snapshot.events.find(event => event.id === fields.targetEventId);
+    if (!target) return 'target_missing';
+    if (target.contentHash !== fields.targetContentHash || target.observedOn !== fields.observedOn) return 'target_changed';
+  }
+  const existing = snapshot.events.find(event => event.id === fields.eventId);
+  if (existing) {
+    if (!sameDraftFacts(fields, existing)) return 'event_conflict';
+    if (optionalChangeDraftEventMatches(fields, existing, draft.eventPredecessorProof)) return 'already_recorded';
+    return draft.eventPredecessorProof === undefined ? 'proof_unverifiable' : 'event_conflict';
+  }
+  if (snapshot.revision !== fields.sourceRevision || snapshot.sourceProof !== fields.sourceProof) return 'history_changed';
+  return draft.eventPredecessorProof === undefined ? 'proof_upgrade_required' : 'current';
+}

+/** Pure decoding only. Callers acquire these bytes under the existing authority
+ * transaction. Neither proof comes from an event ID, text, or reconstructed past. */
+export function optionalChangeDraftSourceProofs(raw: string | null, ownerUserId: string) {
+  decodeCompletionLog(raw, ownerUserId);
+  const actual = decodeActualRoutineEnvelope(raw, ownerUserId);
+  return Object.freeze({ sourceProof: hashOptionalChange41(raw),
+    eventPredecessorProof: hashOptionalChange41(actual ? actual.completionBytes : raw) });
+}

+/** Explicit refresh only. A proof-only upgrade preserves the original revision
+ * and timestamps; a source refresh or renewed review advances the draft revision. */
+export function refreshOptionalChangeDraftProof(draft: OptionalChangeDraft, snapshot: OptionalChangeSnapshot41,
+  ownerUserId: string, updatedAt: string): OptionalChangeDraft {
+  const proofs = optionalChangeDraftSourceProofs(snapshot.sourceBytes, ownerUserId);
+  const source = decodeCompletionLog(snapshot.sourceBytes, ownerUserId);
+  if (proofs.sourceProof !== snapshot.sourceProof || (source.changes?.revision ?? 0) !== snapshot.revision ||
+      canonical41(source.changes?.events ?? []) !== canonical41(snapshot.events)) throw new Error(OPTIONAL_CHANGE_DRAFT_CONFLICT);
+  const status = reconcileOptionalChangeDraft(draft, snapshot);
+  if (status === 'already_recorded') return draft;
+  if (status === 'proof_unverifiable') throw new Error(OPTIONAL_CHANGE_DRAFT_PROOF_UNVERIFIABLE);
+  if (!['current', 'history_changed', 'proof_upgrade_required'].includes(status)) throw new Error(OPTIONAL_CHANGE_DRAFT_CONFLICT);
+  const fields = { ...draft.fields, sourceRevision: snapshot.revision, sourceProof: proofs.sourceProof, attempted: false };
+  const unchangedFields = JSON.stringify(fields) === JSON.stringify(draft.fields);
+  if (unchangedFields && draft.eventPredecessorProof === proofs.eventPredecessorProof) return draft;
+  const proofOnlyUpgrade = unchangedFields && draft.eventPredecessorProof === undefined;
+  return makeOptionalChangeDraft({ id: draft.id, revision: proofOnlyUpgrade ? draft.revision : draft.revision + 1,
+    createdAt: draft.createdAt, updatedAt: proofOnlyUpgrade ? draft.updatedAt : updatedAt, fields,
+    eventPredecessorProof: proofs.eventPredecessorProof });

+/** Explicit Save Draft or confirmation journal only. Target identity cannot be
+ * retargeted, and an attempted factual payload cannot be edited by a retry. */
+export async function saveOptionalChangeDraft(lease: HealthDataWriteLease, request: Readonly<{
+  id: string; expected: OptionalChangeDraft | null; fields: OptionalChangeDraftFields;
+}>, assertCallerCurrent: () => void): Promise<OptionalChangeDraft> {
+  const assertCurrent = () => { assertOriginalHealthDataWriteLease(lease); assertCallerCurrent(); };
+  assertCurrent();
+  const fields = decodeOptionalChangeDraftFields(request.fields), id = request.id;
+  const expected = request.expected ? JSON.stringify(request.expected) : null;
+  const { optionalChangeSnapshotFromBytes } = await import('./completionsStore');
+  assertCurrent();
+  let result: OptionalChangeDraft | undefined;
+  await updatePrivateItemWithSources(OPTIONAL_CHANGE_DRAFT_KEY, [...OPTIONAL_CHANGE_DRAFT_AUTHORIZATION_KEYS, HISTORY_KEY], values => {
+    assertCurrent();
+    const raw = values.get(OPTIONAL_CHANGE_DRAFT_KEY) ?? null;
+    const collection = decodeOptionalChangeDraftCollection(raw, lease.ownerUserId, lease.epoch);
+    const existing = collection.drafts.find(row => row.id === id);
+    if ((existing ? JSON.stringify(existing) : null) !== expected || collection.drafts.some(row => row.id !== id && row.fields.eventId === fields.eventId)) throw new Error(OPTIONAL_CHANGE_DRAFT_CONFLICT);
+    if (existing && (existing.fields.eventId !== fields.eventId || existing.fields.kind !== fields.kind ||
+        existing.fields.targetEventId !== fields.targetEventId || existing.fields.targetContentHash !== fields.targetContentHash ||
+        existing.fields.sourceRevision !== fields.sourceRevision || existing.fields.sourceProof !== fields.sourceProof)) throw new Error(OPTIONAL_CHANGE_DRAFT_CONFLICT);
+    if (existing?.fields.attempted && JSON.stringify(existing.fields) !== JSON.stringify(fields)) throw new Error(OPTIONAL_CHANGE_DRAFT_CONFLICT);
+    const unchangedFields = existing !== undefined && JSON.stringify(existing.fields) === JSON.stringify(fields);
+    if (existing && unchangedFields && existing.eventPredecessorProof !== undefined) {
+      result = existing;
+      return raw;
+    }
+    if (!existing && collection.drafts.length >= OPTIONAL_CHANGE_DRAFT_LIMIT) throw new Error(OPTIONAL_CHANGE_DRAFT_CONFLICT);
+    const historyRaw = values.get(HISTORY_KEY) ?? null;
+    const snapshot = optionalChangeSnapshotFromBytes(historyRaw, lease.ownerUserId);
+    if (existing && existing.eventPredecessorProof === undefined) {
+      const status = reconcileOptionalChangeDraft(existing, snapshot);
+      if (status === 'proof_unverifiable') throw new Error(OPTIONAL_CHANGE_DRAFT_PROOF_UNVERIFIABLE);
+      if (status === 'already_recorded' && unchangedFields) { result = existing; return raw; }
+      if (status !== 'proof_upgrade_required') throw new Error(OPTIONAL_CHANGE_DRAFT_CONFLICT);
+      // An uncertain legacy attempt cannot acquire proof through a retry. The
+      // explicit Refresh path must first clear attempted state for new review.
+      if (existing.fields.attempted || !unchangedFields || fields.attempted) throw new Error(OPTIONAL_CHANGE_DRAFT_PROOF_UPGRADE_REQUIRED);
+    }
+    const proofs = optionalChangeDraftSourceProofs(historyRaw, lease.ownerUserId);
+    if (proofs.sourceProof !== fields.sourceProof) throw new Error(OPTIONAL_CHANGE_DRAFT_CONFLICT);
+    const now = new Date().toISOString();
+    const proofOnlyUpgrade = existing && unchangedFields && existing.eventPredecessorProof === undefined;
+    result = makeOptionalChangeDraft({ id, revision: proofOnlyUpgrade ? existing.revision : (existing?.revision ?? 0) + 1,
+      createdAt: existing?.createdAt ?? now, updatedAt: proofOnlyUpgrade ? existing.updatedAt : now, fields,
+      eventPredecessorProof: proofs.eventPredecessorProof });
+    if (reconcileOptionalChangeDraft(result, snapshot) !== 'current') throw new Error(OPTIONAL_CHANGE_DRAFT_CONFLICT);
+    const next = JSON.stringify({ ...collection, schemaVersion: 3, drafts: [...collection.drafts.filter(row => row.id !== id), result] });
+    decodeOptionalChangeDraftCollection(next, lease.ownerUserId, lease.epoch);
+    return next;
+  }, assertCurrent, undefined, undefined, optionalChangeDraftAuthorization(lease));
+  assertCurrent();
+  return result!;
+}
+/** Explicit refresh preserves original facts and ID while rebasing only the
+ * source proof after canonical target verification. Review still belongs to a
+ * fresh runtime editor; no confirmation or callback is persisted here. */
+export async function refreshOptionalChangeDraft(lease: HealthDataWriteLease, expected: OptionalChangeDraft, assertCallerCurrent: () => void): Promise<Readonly<{ draft: OptionalChangeDraft; snapshot: OptionalChangeSnapshot41 }>> {
+  const assertCurrent = () => { assertOriginalHealthDataWriteLease(lease); assertCallerCurrent(); };
+  assertCurrent();
+  const accepted = JSON.parse(JSON.stringify(expected)) as OptionalChangeDraft;
+  const { optionalChangeSnapshotFromBytes } = await import('./completionsStore');
+  assertCurrent();
+  let result: Readonly<{ draft: OptionalChangeDraft; snapshot: OptionalChangeSnapshot41 }> | undefined;
+  await updatePrivateItemWithSources(OPTIONAL_CHANGE_DRAFT_KEY, [...OPTIONAL_CHANGE_DRAFT_AUTHORIZATION_KEYS, HISTORY_KEY], values => {
+    assertCurrent();
+    const raw = values.get(OPTIONAL_CHANGE_DRAFT_KEY) ?? null;
+    assertExactOptionalChangeDraft(raw, lease, accepted);
+    const collection = decodeOptionalChangeDraftCollection(raw, lease.ownerUserId, lease.epoch);
+    const snapshot = optionalChangeSnapshotFromBytes(values.get(HISTORY_KEY) ?? null, lease.ownerUserId);
+    const draft = refreshOptionalChangeDraftProof(accepted, snapshot, lease.ownerUserId, new Date().toISOString());
+    if (draft === accepted) {
+      result = Object.freeze({ draft: accepted, snapshot });
+      return raw;
+    }
+    const next = JSON.stringify({ ...collection, schemaVersion: 3, drafts: [...collection.drafts.filter(row => row.id !== accepted.id), draft] });
+    decodeOptionalChangeDraftCollection(next, lease.ownerUserId, lease.epoch);
+    result = Object.freeze({ draft, snapshot });
+    return next;
+  }, assertCurrent, undefined, undefined, optionalChangeDraftAuthorization(lease));
+  assertCurrent();
+  return result!;
+}
