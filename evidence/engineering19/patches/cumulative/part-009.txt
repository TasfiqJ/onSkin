+const count = (value: unknown): value is number => typeof value === 'number' && Number.isSafeInteger(value) && value >= 0;
+function decode(value: unknown): Summary {
+  if (!value || typeof value !== 'object' || Array.isArray(value)) throw new Error('WORKSPACE_STATUS_UNAVAILABLE');
+  const s = value as Summary;
+  if (Object.keys(s).sort().join(',') !== 'acceptedPlans,access,overLimit,paidThrough,planLimit,selectedPlans,status,tier,verifiedUntil' ||
+      !statuses.includes(s.status) || !['write_allowed', 'read_only', 'unknown'].includes(s.access) || ![null, 'solo', 'studio'].includes(s.tier) ||
+      !count(s.acceptedPlans) || s.planLimit !== (s.tier === 'solo' ? 20 : s.tier === 'studio' ? 60 : null) ||
+      s.overLimit !== (s.planLimit === null ? null : s.acceptedPlans > s.planLimit) ||
+      (s.selectedPlans !== null && (!count(s.selectedPlans) || s.selectedPlans > s.acceptedPlans || s.selectedPlans > (s.planLimit ?? 0))) ||

+      (s.status === 'unknown' && (s.tier !== null || s.access !== 'unknown' || s.paidThrough !== null || s.verifiedUntil !== null))) throw new Error('WORKSPACE_STATUS_UNAVAILABLE');

+export function createWorkspaceStatusController(request: typeof fetch, render: (state: 'idle' | 'loading' | 'unavailable' | Summary) => void) {
+  let generation = 0; let pending: AbortController | null = null;
+  return {
+    clear() { generation++; pending?.abort(); pending = null; render('idle'); },
+    async check() {
+      if (pending) return;
+      const admitted = ++generation; const controller = new AbortController(); pending = controller; render('loading');
+      const timeout = setTimeout(() => controller.abort(), 15_000);
+      try {
+        const response = await untilAborted(controller.signal, request('/api/professional/workspace-status', { method: 'POST', cache: 'no-store', credentials: 'same-origin', redirect: 'error',
+          headers: { 'Content-Type': 'application/json', 'X-OnSkin-Request': 'workspace-v1' }, body: JSON.stringify({ kind: 'status' }), signal: controller.signal }));
+        if (generation !== admitted) return;
+        if (!response.ok || response.headers.get('Cache-Control') !== 'no-store' || !response.headers.get('Content-Type')?.startsWith('application/json')) throw new Error('WORKSPACE_STATUS_UNAVAILABLE');
+        const raw = await untilAborted(controller.signal, response.text()); if (generation !== admitted) return;
+        if (raw.length > 4096) throw new Error('WORKSPACE_STATUS_UNAVAILABLE');
+        const value = JSON.parse(raw) as { kind: string; summary: unknown };
+        if (!value || Object.keys(value).sort().join(',') !== 'kind,summary' || value.kind !== 'workspace-status') throw new Error('WORKSPACE_STATUS_UNAVAILABLE');
