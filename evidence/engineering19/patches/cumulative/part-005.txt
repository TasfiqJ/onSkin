+export function compareRecordedFormulaVersions(
+  first: OwnedBottleHistorySnapshot38,
+  second: OwnedBottleHistorySnapshot38,
+  firstVersionId: string | null,
+  secondVersionId: string | null,
+  ownerUserId: string,
+) {

+    for (const snapshot of [first, second]) {
+      if (snapshot.ownedUnitId !== snapshot.productId || !snapshot.currentVersionId) return null;
+      validateOwnedBottleHistory38({
+        productId: snapshot.productId,
+        ownerUserId,
+        revision: snapshot.revision,
+        currentVersionId: snapshot.currentVersionId,
+        versions: snapshot.versions,
+      });
+    }
+  } catch {
+    return null;
+  }
+  const left = first.versions.find((version) => version.id === firstVersionId);
+  const right = second.versions.find((version) => version.id === secondVersionId);
+  if (!left || !right) return null;

+    (version) =>
+      version.formulaVersion.labelStatus === 'user_confirmed_label' &&
+      version.basis === 'label_in_hand',

+    ? entries(left.formulaVersion.labelText)
+    : new Map<string, string>();

+    ? entries(right.formulaVersion.labelText)
+    : new Map<string, string>();

+  return {
+    left,
+    right,
+    sameRecord: left.id === right.id,
+    manufacturerFormulaDifference: 'UNCONFIRMED' as const,

+      ? [...leftEntries].filter(([key]) => !rightEntries.has(key)).map(([, text]) => text)

+import { categoryLabel } from './categories';
+import { validLocalDate } from './freshness';
+import type { ShelfProduct } from './store';
+
+export type ComparableProduct = Pick<
+  ShelfProduct,
+  | 'id'
+  | 'name'
+  | 'brand'
+  | 'category'
+  | 'status'
+  | 'isOpened'
+  | 'openedAt'
+  | 'paoMonths'
+  | 'paoSource'
+  | 'expiryDate'
+  | 'expirySource'
+  | 'addedVia'
+  | 'ingredients'
+>;
+
+function facts(product: ComparableProduct): Record<string, string> {
+  return {
+    Brand: product.brand || 'Not recorded',
+    Category: categoryLabel(product.category) ?? 'Not recorded',
+    Status: { active: 'Active', finished: 'Finished', discarded: 'Discarded' }[product.status],
+    'Opened date': !product.isOpened ? 'Unopened' : (validLocalDate(product.openedAt) ?? 'Unknown'),
+    'Printed expiry':
+      product.expirySource === 'printed'
+        ? (validLocalDate(product.expiryDate) ?? 'Not recorded')
+        : 'Not recorded',
+    'Period after opening':
+      product.paoMonths !== null &&
+      (product.paoSource === 'label' || product.paoSource === 'catalog')
+        ? `${product.paoMonths} months · ${product.paoSource}`
+        : 'Not recorded',
+    'Added through': {
+      manual: 'Manual entry',
+      onboarding: 'Onboarding',
+      barcode: 'Barcode',
+      search: 'Catalog search',
+      ocr: 'Label text',
+    }[product.addedVia],
+  };
+}
+
+/** Compare recorded text only. No synonym inference, completeness claim,
+ * compatibility, clinical ranking, quota, persistence or provider lookup. */
+export function compareSavedProducts<T extends ComparableProduct>(
+  products: readonly T[],
+  leftId: string | null,
+  rightId: string | null,
+) {
+  if (!leftId || !rightId || leftId === rightId) return null;
+  const left = products.find((product) => product.id === leftId);
+  const right = products.find((product) => product.id === rightId);
+  if (!left || !right) return null;
+  const leftFacts = facts(left);
+  const rightFacts = facts(right);
+  const ingredientKey = (text: string) => text.trim().replace(/\s+/gu, ' ').toLowerCase();
+  const leftIngredients = new Map<string, string>();
+  for (const text of left.ingredients) {
+    const key = ingredientKey(text);
+    if (key && !leftIngredients.has(key)) leftIngredients.set(key, text.trim());
+  }
+  const rightIngredients = new Set(
+    right.ingredients.filter((text) => text.trim()).map(ingredientKey),
+  );
+  // Keep the first recorded spelling as well as the source's order.
+  const sharedIngredients =
+    leftIngredients.size && rightIngredients.size
+      ? [...leftIngredients].filter(([key]) => rightIngredients.has(key)).map(([, text]) => text)
+      : null;
+  return {
+    left,
+    right,
+    sharedIngredients,
+    rows: Object.entries(leftFacts).map(([label, value]) => ({
+      label,
+      left: value,
+      right: rightFacts[label]!,
+    })),
+  };
+}

+function normalizeCompletionLog(value: unknown, strictCurrentSchema = false): Log {
+  if (!isRecord(value)) throw completionLogError(COMPLETION_LOG_INVALID);
+  const out: Log = {};
+  for (const [date, keys] of Object.entries(value)) {
+    const normalizedDate = normalizeLocalDateISO(date);
+    if (!normalizedDate || !Array.isArray(keys)) {
+      throw completionLogError(COMPLETION_LOG_INVALID);
+    }
+    if (strictCurrentSchema && normalizedDate !== date) {
+      throw completionLogError(COMPLETION_LOG_INVALID);
+    }
+    const normalizedKeys: string[] = [];
+    for (const key of keys) {
+      const normalizedKey = normalizeStepKey(key);
+      if (!normalizedKey) throw completionLogError(COMPLETION_LOG_INVALID);
+      if (
+        strictCurrentSchema &&
+        (normalizedKey !== key || normalizedKeys.includes(normalizedKey))
+      ) {
+        throw completionLogError(COMPLETION_LOG_INVALID);
+      }
+      if (!normalizedKeys.includes(normalizedKey)) normalizedKeys.push(normalizedKey);
+    }
+    if (strictCurrentSchema && normalizedKeys.length === 0) {
+      throw completionLogError(COMPLETION_LOG_INVALID);
+    }
+    if (normalizedKeys.length > 0) {
+      out[normalizedDate] = [...new Set([...(out[normalizedDate] ?? []), ...normalizedKeys])];

+export function decodeCompletionLog(raw: string | null, ownerUserId: string | null, allowOptionalChanges = true, allowActualRoutine = true): CompletionState {

+  }

+    if (!allowActualRoutine) throw completionLogError(COMPLETION_LOG_INVALID);

+  }

+    return {
+      days: normalizeCompletionLog(parsed),
+      completedDays: new Set(),
+      sync: emptyCompletionSyncState(),
+    };
+  }
+  if (parsed.version === 1) {

+      throw completionLogError(COMPLETION_LOG_INVALID);
+    }

+      !hasOwn(parsed, 'completedDays') ||
+      !isRecord(parsed.days)

+    }

+      days,

+      // Historical v2 rows have neither an original timestamp nor durable
+      // server identity. Never invent remote operations for them.

+    }

+  }

+    !hasOwn(parsed, 'days') ||

+    !hasOwn(parsed, 'sync') ||
+    !isRecord(parsed.days)
+  ) {
+    throw completionLogError(COMPLETION_LOG_INVALID);
+  }

+    const sync = decodeCompletionSyncState(parsed.sync, {
+      hasCompletedStep: (completedDate, step) => days[completedDate]?.includes(step) === true,
+      hasCompletedDay: (completedDate) => completedDays.has(completedDate),
+    });
+    return { days, completedDays, sync };
+  } catch {
+    throw completionLogError(COMPLETION_LOG_INVALID);
+  }

+function reconcile(): void {
+  const next = new Set<HTMLElement>();
+  let branch = dialogs.at(-1);
+  // The platform owns focus-guard siblings inside its portal. Keep those
+  // available so Tab can wrap, and isolate the application at the body boundary.
+  while (branch?.parentElement && branch.parentElement.tagName !== 'BODY') branch = branch.parentElement;
+  if (branch?.parentElement?.tagName === 'BODY') {
+    for (const sibling of Array.from(branch.parentElement.children)) {
+      if (sibling !== branch && sibling instanceof HTMLElement) next.add(sibling);
+    }
+  }
+  for (const [node, prior] of isolated) {
+    if (next.has(node)) continue;
+    node.inert = prior.inert;
+    if (prior.ariaHidden === null) node.removeAttribute('aria-hidden');
+    else node.setAttribute('aria-hidden', prior.ariaHidden);
+    isolated.delete(node);
+  }
+  for (const node of next) {
+    if (!isolated.has(node)) isolated.set(node, { inert: node.inert, ariaHidden: node.getAttribute('aria-hidden') });
