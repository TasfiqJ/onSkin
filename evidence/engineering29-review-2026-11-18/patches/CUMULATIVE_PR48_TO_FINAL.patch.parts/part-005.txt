+  products: readonly BriefProduct[],
+  selection: AppointmentBriefSelection,
+  preparedOn: string,
+  records: readonly BriefRecordedItem[] = [],

+  const invalid = () => {
+    throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+  };

+    if (!record) return invalid();
+    return record;
+  });

+    typeof selection.includeDates !== 'boolean' ||
+    (selection.includePhoto !== undefined && typeof selection.includePhoto !== 'boolean') ||
+    selection.productIds.length > 1000 ||
+    new Set(selection.productIds).size !== selection.productIds.length
+  )
+    invalid();

+      typeof value !== 'string' ||

+      /[\p{Cf}\p{Cs}\u0000-\u0008\u000b\u000c\u000e-\u001f\u007f-\u009f]/u.test(value)
+    )
+      invalid();
+  }
+  if (
+    !selection.productIds.length &&
+    !recordIds.length &&
+    !selection.observations.trim() &&
+    !selection.questions.trim()
+    && !selection.includePhoto
+  )
+    invalid();
+  const chosen = selection.productIds.map((id) => {
+    const product = products.find((candidate) => candidate.id === id);
+    if (!product) return invalid();
+    return product;
+  });
+  return [

+                ? []
+                : [
+                    `${product.status === 'finished' ? 'Finished' : 'Discarded'} date: ${dates.finished}`,
+                  ]),
+            ]
+          : []),

+    productUpdatedAt: snapshot.productUpdatedAt, productIdentityHash: snapshot.productIdentityHash,
+    revision: snapshot.revision, currentVersionId: snapshot.currentVersionId });

+function decodeSource(value: unknown): BottleDraftSource {
+  const row = record(value, ['productId', 'ownedUnitId', 'productUpdatedAt', 'productIdentityHash', 'revision', 'currentVersionId']);
+  const currentVersionId = row.currentVersionId;
+  if (currentVersionId !== null && (typeof currentVersionId !== 'string' || !/^owned38:[0-9a-f]{64}$/.test(currentVersionId))) invalid();
+  const revision = integer(row.revision, 0);
+  if ((revision === 0) !== (currentVersionId === null)) invalid();
+  return Object.freeze({ productId: nonempty(row.productId, 128), ownedUnitId: nonempty(row.ownedUnitId, 200),
+    productUpdatedAt: timestamp(row.productUpdatedAt), productIdentityHash: hash(row.productIdentityHash), revision, currentVersionId });
+}
+export function decodeBottleDraftFields(value: unknown): BottleDraftFields {
+  const row = record(value, ['mode', 'label', 'market', 'variant', 'observed']);
+  const mode = row.mode;
+  if (mode !== null && mode !== 'label' && mode !== 'unknown') invalid();
+  // A draft deliberately permits incomplete dates and empty labels. Actual
+  // observation validation still owns final confirmation.
+  return Object.freeze({ mode, label: text(row.label, 8000, true), market: text(row.market, 80),
+    variant: text(row.variant, 160), observed: text(row.observed, 10) });
+}
+function draftBody(value: Omit<BottleDraft, 'contentHash'>): Omit<BottleDraft, 'contentHash'> {
+  const row = record(value, ['id', 'revision', 'createdAt', 'updatedAt', 'source', 'fields']);
+  const id = nonempty(row.id, 36);
+  if (!/^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/.test(id)) invalid();
+  const createdAt = timestamp(row.createdAt), updatedAt = timestamp(row.updatedAt);
+  if (updatedAt < createdAt) invalid();
+  return Object.freeze({ id, revision: integer(row.revision, 1), createdAt, updatedAt,
+    source: decodeSource(row.source), fields: decodeBottleDraftFields(row.fields) });
+}
+export function makeBottleDraft(value: Omit<BottleDraft, 'contentHash'>): BottleDraft {
+  const body = draftBody(value);
+  return Object.freeze({ ...body, contentHash: bytesToHex(sha256(utf8ToBytes(JSON.stringify(body)))) });
+}
+export function decodeBottleDraftCollection(raw: string | null, ownerUserId: string, processingEpoch?: number): BottleDraftCollection {
+  nonempty(ownerUserId, 128);
+  if (raw === null) {
+    return Object.freeze({ schemaVersion: 1, provenance: 'owner_unconfirmed_transcription', ownerUserId,
+      processingEpoch: integer(processingEpoch, 0), drafts: Object.freeze([]) });
+  }
+  if (raw.length > 600_000 || utf8ToBytes(raw).length > 600_000) invalid();
+  let parsed: unknown;
+  try { parsed = JSON.parse(raw) as unknown; } catch { invalid(); }
+  const row = record(parsed, ['schemaVersion', 'provenance', 'ownerUserId', 'processingEpoch', 'drafts']);
+  if (row.schemaVersion !== 1 || row.provenance !== 'owner_unconfirmed_transcription') invalid();
+  if (row.ownerUserId !== ownerUserId || (processingEpoch !== undefined && row.processingEpoch !== processingEpoch)) throw new Error(BOTTLE_DRAFT_AUTHORITY);
+  const epoch = integer(row.processingEpoch, 0);
+  if (!Array.isArray(row.drafts) || row.drafts.length > BOTTLE_DRAFT_LIMIT) invalid();
+  const drafts = row.drafts.map(value => {
+    const entry = record(value, ['id', 'revision', 'createdAt', 'updatedAt', 'source', 'fields', 'contentHash']);
+    const { contentHash, ...body } = entry;
+    const draft = makeBottleDraft(body as Omit<BottleDraft, 'contentHash'>);
+    if (draft.contentHash !== hash(contentHash)) invalid();
+    return draft;
+  });
+  if (new Set(drafts.map(d => d.id)).size !== drafts.length ||
+      new Set(drafts.map(d => d.source.productId)).size !== drafts.length) invalid();
+  return Object.freeze({ schemaVersion: 1, provenance: 'owner_unconfirmed_transcription', ownerUserId,
+    processingEpoch: epoch, drafts: Object.freeze(drafts) });
+}

+async function captureSourceProof(lease: HealthDataWriteLease, anchor: string, sources: readonly string[]): Promise<BriefSourceProof> {
+  const guard = () => assertOriginalHealthDataWriteLease(lease);
+  guard();
+  const { updatePrivateItemWithSources } = await import('@/lib/storage/privateKV');
+  guard();
+  let bytes: Map<string, string | null> | undefined;
+  await updatePrivateItemWithSources(anchor, sources, values => {
+    guard(); bytes = new Map(sources.map(key => [key, values.get(key) ?? null]));
+    return values.get(anchor) ?? null;
+  }, guard);
+  guard();
+  if (!bytes) throw new Error('APPOINTMENT_BRIEF_SOURCE_UNAVAILABLE');

+/** Authorize the handoff in the same fixed source queue as the final byte check.
+ * Invoke the platform synchronously here, but await its sheet outside the queue:
+ * withdrawal/deletion must not wait for the recipient dialog to close. */
+export async function dispatchWithBriefSourceProof<T>(
+  proof: BriefSourceProof, lease: HealthDataWriteLease,
+  assertReviewCurrent: () => void, dispatch: () => Promise<T>, retirement?: AbortSignal,
+): Promise<T> {
+  const accepted = proofs.get(proof);
+  if (!accepted || accepted.lease !== lease) throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+  const guard = () => { assertOriginalHealthDataWriteLease(lease); assertReviewCurrent(); if (retirement?.aborted) throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE'); };
+  guard();
+  const { updatePrivateItemWithSources } = await import('@/lib/storage/privateKV');
+  guard();
+  let pending: Promise<T> | undefined;
+  await updatePrivateItemWithSources(accepted.anchor, [...accepted.bytes.keys()], values => {
+    guard();
+    if ([...accepted.bytes].some(([key, value]) => (values.get(key) ?? null) !== value)) {
+      throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+    }
+    return values.get(accepted.anchor) ?? null;
+  }, guard, undefined, () => {
+    guard();
+    if (proofs.get(proof) !== accepted) throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+    proofs.delete(proof);
+    pending = dispatch();
+    // Observe rejection immediately even if the storage guard itself retires.
+    void pending.catch(() => undefined);
+  });
+  guard();
+  if (!pending) throw new Error('APPOINTMENT_BRIEF_SOURCE_UNAVAILABLE');
+  let rejectRetired!: (reason: unknown) => void;
+  const retired = new Promise<never>((_resolve, reject) => { rejectRetired = reject; });
+  const checkRetirement = () => { try { guard(); } catch (error) { rejectRetired(error); } };
+  const stop = subscribeActiveHealthProcessingLeaseChanges(checkRetirement);
+  retirement?.addEventListener('abort', checkRetirement, { once: true });
+  const expiry = lease.expiresAt === null ? null : setTimeout(checkRetirement, Math.max(0, lease.expiresAt - Date.now()) + 1);
+  try {
+    // Covers a boundary between platform invocation and listener installation.
+    checkRetirement();
+    const result = await Promise.race([pending, retired]);
+    guard();
+    return result;
+  } finally {
+    stop(); retirement?.removeEventListener('abort', checkRetirement);
+    if (expiry !== null) clearTimeout(expiry);
+  }
+}

+export function decodeManualIntakeDraftFields(value: unknown): ManualIntakeDraftFields {

+export function makeManualIntakeDraft(value: Omit<ManualIntakeDraft, 'contentHash'>): ManualIntakeDraft {
+  const row = record(value, ['id', 'revision', 'createdAt', 'updatedAt', 'operationId', 'fields']);
+  const createdAt = timestamp(row.createdAt), updatedAt = timestamp(row.updatedAt);
+  if (updatedAt < createdAt) invalid();
+  const fields = decodeManualIntakeDraftFields(row.fields), operationId = uuid(row.operationId);
+  if (fields.attempted !== null && fields.attempted.input.operationId !== operationId) invalid();
+  const body = { id: uuid(row.id), revision: integer(row.revision, 1), createdAt, updatedAt, operationId, fields };
+  return Object.freeze({ ...body, contentHash: bytesToHex(sha256(utf8ToBytes(JSON.stringify(body)))) });

+export function decodeManualIntakeDraftCollection(raw: string | null, ownerUserId: string, processingEpoch?: number): ManualIntakeDraftCollection {
+  nonempty(ownerUserId, 128);

+    processingEpoch: integer(processingEpoch, 1), drafts: Object.freeze([]) });
+  if (raw.length > MANUAL_INTAKE_DRAFT_MAX_BYTES || utf8ToBytes(raw).length > MANUAL_INTAKE_DRAFT_MAX_BYTES) invalid();
+  let parsed: unknown;
+  try { parsed = JSON.parse(raw) as unknown; } catch { invalid(); }
+  const row = record(parsed, ['schemaVersion', 'provenance', 'ownerUserId', 'processingEpoch', 'drafts']);
+  if (row.schemaVersion !== 1 || row.provenance !== 'owner_unconfirmed_manual_intake') invalid();
+  if (row.ownerUserId !== ownerUserId || (processingEpoch !== undefined && row.processingEpoch !== processingEpoch)) throw new Error(MANUAL_INTAKE_DRAFT_AUTHORITY);
+  const epoch = integer(row.processingEpoch, 1);
+  if (!Array.isArray(row.drafts) || row.drafts.length > MANUAL_INTAKE_DRAFT_LIMIT) invalid();
+  const drafts = row.drafts.map(entry => {
+    const checked = record(entry, ['id', 'revision', 'createdAt', 'updatedAt', 'operationId', 'fields', 'contentHash']);

+    const draft = makeManualIntakeDraft(body as Omit<ManualIntakeDraft, 'contentHash'>);
+    if (draft.contentHash !== contentHash) invalid();
+    return draft;
+  });
+  if (new Set(drafts.map(draft => draft.id)).size !== drafts.length ||
+      new Set(drafts.map(draft => draft.operationId)).size !== drafts.length) invalid();
+  return Object.freeze({ schemaVersion: 1, provenance: 'owner_unconfirmed_manual_intake', ownerUserId,
+    processingEpoch: epoch, drafts: Object.freeze(drafts) });

+export async function saveManualIntakeDraft(lease: HealthDataWriteLease, request: Readonly<{
+  id: string;

+  operationId: string;
+  fields: ManualIntakeDraftFields;
+}>, assertCallerCurrent: () => void): Promise<ManualIntakeDraft> {
+  const assertCurrent = () => { assertOriginalHealthDataWriteLease(lease); assertCallerCurrent(); };
+  assertCurrent();

+  // Snapshot every caller value before the first await, including the CAS proof.
+  const expected = request.expected ? JSON.stringify(request.expected) : null;
+  const id = request.id, operationId = request.operationId;
+  let result: ManualIntakeDraft | undefined;

+    assertCurrent();

+    const existing = collection.drafts.find(row => row.id === id);

+        collection.drafts.some(row => row.id !== id && row.operationId === operationId)) throw new Error(MANUAL_INTAKE_DRAFT_CONFLICT);

+    // snapshot. Even edits to raw fields could falsely describe what was saved.

+      result = existing;
+      return raw;
+    }
+    if (!existing && collection.drafts.length >= MANUAL_INTAKE_DRAFT_LIMIT) throw new Error(MANUAL_INTAKE_DRAFT_CONFLICT);
+    const now = new Date().toISOString();
+    result = makeManualIntakeDraft({ id, revision: (existing?.revision ?? 0) + 1,
+      createdAt: existing?.createdAt ?? now, updatedAt: now, operationId, fields });

+    decodeManualIntakeDraftCollection(next, lease.ownerUserId, lease.epoch);
+    return next;
+  }, assertCurrent, undefined, undefined, manualIntakeDraftAuthorization(lease));
+  assertCurrent();
+  return result!;
