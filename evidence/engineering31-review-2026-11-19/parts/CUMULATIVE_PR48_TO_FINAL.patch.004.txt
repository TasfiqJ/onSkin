+export function clearActualRoutine(current: ActualRoutineLedger, request: { id: string; createdAt: string; expectedParentHash: string }): ActualRoutineLedger {
+  if (!uuid(request.id) || !instant(request.createdAt)) fail();

+  const parent = existing ? current.versions[existing.sequence - 2] : current.versions.at(-1);
+  if (!parent || parent.source === 'owner-cleared-local' || parent.contentHash !== request.expectedParentHash) throw new Error(ACTUAL_ROUTINE_STALE);
+  const payload = { id: request.id, sequence: parent.sequence + 1, createdAt: request.createdAt,
+    sourceHash: parent.contentHash, am: [], pm: [], schemaVersion: 2 as const, source: 'owner-cleared-local' as const,
+    actorOwnerId: current.ownerUserId, parentId: parent.id, parentHash: parent.contentHash };
+  const next = { ...payload, contentHash: hash(payload) };
+  if (existing) {
+    if (existing !== current.versions.at(-1) || canonical41(existing) !== canonical41(next)) throw new Error(ACTUAL_ROUTINE_STALE);
+    return current;

+import type { PlanHistoryStep45 } from './planHistory45';
+
+type Sequence = Readonly<{ am: readonly PlanHistoryStep45[]; pm: readonly PlanHistoryStep45[] }>;
+/** Free factual comparison of exact bottle identity, names, phase and relative
+ * order. No formula equivalence, use, compatibility or advanced advice inferred. */
+export function sequenceComparisonLines(previous: Sequence | null, reviewed: Sequence): readonly string[] {
+  if (!previous) return ['No previous owner-applied sequence is available for comparison. This will be the first applied version.'];
+  const lines: string[] = [];
+  for (const phase of ['am', 'pm'] as const) {
+    const label = phase === 'am' ? 'Morning' : 'Evening';
+    const before = previous[phase]; const after = reviewed[phase];
+    const oldIds = new Set(before.map(step => step.productId)); const newIds = new Set(after.map(step => step.productId));
+    const added = after.filter(step => !oldIds.has(step.productId)); const removed = before.filter(step => !newIds.has(step.productId));
+    for (const step of added) lines.push(`${label}: add ${step.name} to this phase.`);
+    for (const step of removed) lines.push(`${label}: remove ${step.name} from this phase. Earlier versions and use entries stay unchanged.`);
+    for (const step of after) {
+      const prior = before.find(item => item.productId === step.productId);
+      if (prior && prior.name !== step.name) lines.push(`${label}: recorded name changes from ${prior.name} to ${step.name} for the same saved bottle.`);
+    }
+    const oldOrder = before.filter(step => newIds.has(step.productId)).map(step => step.productId);
+    const newOrder = after.filter(step => oldIds.has(step.productId)).map(step => step.productId);
+    const reordered = JSON.stringify(oldOrder) !== JSON.stringify(newOrder);
+    if (reordered) lines.push(`${label}: relative order of retained bottles changes.`);
+    if (!added.length && !removed.length && !reordered && after.every(step => before.find(item => item.productId === step.productId)?.name === step.name)) {
+      lines.push(`${label}: same saved bottles, recorded names and order.`);
+    }
+  }
+  lines.push('Names and sequence only. Bottle observations or other source facts may have changed even when these sequences match. This is not a formula, compatibility or use comparison.');
+  return Object.freeze(lines);
+}

+import { useLayoutEffect, useMemo, useRef, useState, useSyncExternalStore } from 'react';
+import { View } from 'react-native';
+
+import { Button, Text } from '@/components/ui';
+import { assertAccountIdentityGeneration, captureAccountIdentityGeneration } from '@/lib/auth/accountGeneration';
+import { activeHealthProcessingLeaseSnapshot, subscribeActiveHealthProcessingLeaseChanges } from '@/lib/consent/healthProcessingEpoch';
+import { retryPlaintextStagingRecovery } from '@/lib/storage/plaintextStaging';
+import { exportDataWithOutcome } from './actions';
+
+type ExportState = Readonly<{ scope: object; phase: 'idle' | 'preparing' | 'cleanup'; cleanupPending: boolean; feedback: string | null; operation?: object }>;
+
+/** The same account-rights export/recovery behavior in the active and paused
+ * shells. This scope only retires actions; storage still verifies its owner. */
+export function DataExportControls({ ownerUserId, source, disabled = false, onStart }: {
+  ownerUserId: string | null; source?: unknown; disabled?: boolean; onStart?: () => void;
+}) {
+  const identity = captureAccountIdentityGeneration();
+  const health = useSyncExternalStore(subscribeActiveHealthProcessingLeaseChanges, activeHealthProcessingLeaseSnapshot, () => null);
+  const scope = useMemo(() => Object.freeze({ ownerUserId, identity, health, source }), [ownerUserId, identity, health, source]);
+  const liveScope = useRef<object | null>(scope);
+  const disabledRef = useRef(disabled);
+  useLayoutEffect(() => { disabledRef.current = disabled; }, [disabled]);
+  const flight = useRef<{ scope: object; controller: AbortController } | null>(null);
+  const [state, setState] = useState<ExportState>(() => ({ scope, phase: 'idle', cleanupPending: false, feedback: null }));
+  const liveState = useRef(state);
+  const publish = (next: ExportState) => { liveState.current = next; setState(next); };
+  useLayoutEffect(() => {
+    liveScope.current = scope;
+    if (liveState.current.scope !== scope) publish({ scope, phase: 'idle', cleanupPending: false, feedback: null });
+    return () => {
+      if (liveScope.current === scope) liveScope.current = null;
+      if (flight.current?.scope === scope) {
+        const retired = flight.current;
+        flight.current = null;
+        retired.controller.abort();
+      }
+    };
+  }, [scope]);
+  const current = () => {
+    try { assertAccountIdentityGeneration(identity); }
+    catch { return false; }
+    return liveScope.current === scope && activeHealthProcessingLeaseSnapshot() === health;
+  };
+  const shown = state.scope === scope ? state : { scope, phase: 'idle' as const, cleanupPending: false, feedback: null };
+
+  async function start(cleanupOnly: boolean, expected: ExportState) {
+    if (liveState.current !== expected || !current() || disabledRef.current || flight.current || (!cleanupOnly && liveState.current.scope === scope && liveState.current.cleanupPending)) return;
+    const operation = { scope, controller: new AbortController() };
+    flight.current = operation;
+    publish({ scope, phase: cleanupOnly ? 'cleanup' : 'preparing', cleanupPending: cleanupOnly, feedback: null, operation });
+    onStart?.();
+    try {
+      if (cleanupOnly) {
+        await retryPlaintextStagingRecovery();
+        if (current() && flight.current === operation) publish({ scope, phase: 'idle', cleanupPending: false,
+          feedback: 'Pending temporary-file cleanup completed. Copies in another app are unchanged.' });
+      } else {
+        const result = await exportDataWithOutcome(operation.controller.signal);
+        if (!current() || flight.current !== operation) return;
+        publish({ scope, phase: 'idle', cleanupPending: result.cleanup === 'pending', feedback:
+          `${result.shared ? 'The system sharing call completed. This does not confirm delivery.' : 'The export sheet is unavailable on this device.'} ${result.cleanup === 'pending'
+            ? 'Temporary-file removal is not confirmed. Retry cleanup before another export. Copies in another app cannot be recalled.'
+            : 'The temporary export file was removed.'}` });
+      }
+    } catch {
+      if (current() && flight.current === operation) publish({ scope, phase: 'idle', cleanupPending: true, feedback:
+        cleanupOnly ? 'Temporary-file cleanup is still pending. Retry when device storage is available. No new export was created.' :
+          `${operation.controller.signal.aborted ? 'Export preparation was canceled.' : 'Export could not be confirmed.'} Temporary-file cleanup is not confirmed. Retry cleanup before another export. Copies in another app cannot be recalled.` });
+    } finally {
+      if (flight.current === operation) flight.current = null;
+    }
+  }
+  return <View>
+    <Button className="mt-2" label={shown.phase === 'preparing' ? 'Preparing...' : 'Export my data'} variant="ghost"
+      disabled={disabled || shown.phase !== 'idle' || shown.cleanupPending} onPress={() => void start(false, shown)} />
+    {shown.phase === 'preparing' ? <Button label="Cancel export preparation" variant="ghost" onPress={() => {
+      if (current() && liveState.current === shown && shown.operation === flight.current) flight.current.controller.abort();
+    }} /> : null}
+    {shown.cleanupPending ? <Button label={shown.phase === 'cleanup' ? 'Removing temporary exports…' : 'Retry temporary export cleanup'} variant="ghost"
+      disabled={disabled || shown.phase !== 'idle'} onPress={() => void start(true, shown)} /> : null}
+    {shown.feedback ? <Text accessibilityRole="alert" accessibilityLiveRegion="polite" variant="bodySm" tone="muted" className="mt-3">{shown.feedback}</Text> : null}
+  </View>;
+}
