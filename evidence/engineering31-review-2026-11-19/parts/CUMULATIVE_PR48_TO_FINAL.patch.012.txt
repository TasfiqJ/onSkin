+/** Workspace-only trusted transaction. There is no client/source/record port in
+ * this journey. Existing template permission requires current licence/billing. */
+export async function practiceTemplates(repository: WorkspaceRepository | null, token: string, workspaceId: string, command?: TemplateCommand) {
+  if (!repository) throw new ProfessionalAccessError('unavailable');
+  const captured = command ? structuredClone(command) : undefined;
+  const request: AccessRequest = { action: 'template_author_issue', operation: 'write', workspaceId, ownerId: null,
+    fields: [], purpose: null, grantVersion: null, grantEpoch: null };
+  return repository.transaction({ workspaceId, ownerId: null }, async raw => {
+    const tx = snapshotWorkspaceTransaction(raw);
+    const authority = async () => {
+      const billing = await decodeWorkspaceBilling(await tx.billing(workspaceId), workspaceId);
+      const a = { ...await tx.authority(token, request), billing: workspaceBillingAccess(billing, tx.now()) };
+      const denied = accessDenial(a, request, tx.now());
+      if (denied) throw new ProfessionalAccessError(denied);
+      return { a, billing };
+    };
+    const admitted = await authority(); const initial = canonical(admitted);
+    if (!tx.templates || captured && !tx.saveTemplates) throw new ProfessionalAccessError('unavailable');
+    const previous = await decodePracticeTemplates(await tx.templates(workspaceId), workspaceId, tx.now());
+    const next = captured ? await changePracticeTemplate(previous, workspaceId, admitted.a.session.subjectId, captured, tx.now()) : previous;
+    if (canonical(previous) !== canonical(next)) await tx.saveTemplates!(next);
+    tx.beforeCommit(async () => {
+      if (canonical(await authority()) !== initial || canonical(await tx.templates!(workspaceId)) !== canonical(next)) throw new ProfessionalAccessError('changed');
+    });
+    return next;
+  });

+export type WorkspaceBillingStatus = 'incomplete' | 'active' | 'cancel_at_period_end' | 'past_due_grace' |
+  'suspended' | 'expired' | 'refunded' | 'disputed';
+export type WorkspaceTier = 'solo' | 'studio';

+const DAY = 86_400_000;
+const statuses: readonly string[] = ['incomplete', 'active', 'cancel_at_period_end', 'past_due_grace', 'suspended', 'expired', 'refunded', 'disputed'];
+export const workspaceIdValid = (value: unknown): value is string => typeof value === 'string' && /^[A-Za-z0-9][A-Za-z0-9:_-]{0,127}$/u.test(value);
+const timestamp = (value: unknown): value is number => typeof value === 'number' && Number.isSafeInteger(value) && value >= 0;
+const fail = (): never => { throw new Error('WORKSPACE_BILLING_REFUSED'); };

+/** Only a server-owned verifier may supply these facts: verify the signed raw
+ * webhook, retrieve the subscription from the provider and bind it to the
+ * stored workspace/customer/subscription. No HTTP JSON verifier is installed. */
+export type ProviderBillingFacts = Readonly<{
+  workspaceId: string; customerId: string; subscriptionId: string; eventId: string;
+  eventAt: number; retrievedAt: number; verifiedUntil: number;
+  status: WorkspaceBillingStatus; tier: WorkspaceTier; paidThrough: number | null; pastDueSince: number | null;

+export interface WorkspaceBillingVerifier {
+  verifyAndRetrieve(rawBody: Uint8Array, signature: string): Promise<ProviderBillingFacts>;

+  previousHash: string | null; contentHash: string;

+export function emptyWorkspaceBilling(workspaceId: string, customerId: string, subscriptionId: string): WorkspaceBilling {

+function factsValid(facts: ProviderBillingFacts): boolean {
+  return !!facts && Object.keys(facts).sort().join(',') === 'customerId,eventAt,eventId,paidThrough,pastDueSince,retrievedAt,status,subscriptionId,tier,verifiedUntil,workspaceId' &&
+    [facts.workspaceId, facts.customerId, facts.subscriptionId, facts.eventId].every(workspaceIdValid) &&
+    statuses.includes(facts.status) && Object.hasOwn(WORKSPACE_LIMITS, facts.tier) &&
+    [facts.eventAt, facts.retrievedAt, facts.verifiedUntil].every(timestamp) && facts.eventAt <= facts.retrievedAt && facts.verifiedUntil > facts.retrievedAt &&
+    (facts.paidThrough === null || timestamp(facts.paidThrough)) && (facts.pastDueSince === null || timestamp(facts.pastDueSince)) &&
+    (!['active', 'cancel_at_period_end'].includes(facts.status) || (facts.paidThrough !== null && facts.paidThrough > facts.retrievedAt)) &&
+    (facts.status !== 'past_due_grace' || (facts.pastDueSince !== null && facts.pastDueSince <= facts.retrievedAt));

+async function append(state: WorkspaceBilling, facts: ProviderBillingFacts): Promise<WorkspaceBilling> {
+  if (!factsValid(facts) || facts.workspaceId !== state.workspaceId || facts.customerId !== state.customerId || facts.subscriptionId !== state.subscriptionId) return fail();
+  const existing = state.receipts.find(receipt => receipt.eventId === facts.eventId);
+  if (existing) {
+    // Retrieval time/lease may advance on exact retry; the provider event's
+    // substantive identity must remain exact. A fresh event is needed to renew.
+    const original = { ...existing } as Record<string, unknown>;
+    for (const key of ['schemaVersion', 'source', 'sequence', 'previousHash', 'contentHash', 'retrievedAt', 'verifiedUntil']) delete original[key];
+    const current = { ...facts } as Record<string, unknown>; delete current.retrievedAt; delete current.verifiedUntil;
+    if (canonical(original) !== canonical(current)) return fail();
+    return state;
+  }
+  if (state.receipts.length >= 10_000 || state.revision !== state.receipts.length) return fail();
+  const last = state.receipts.at(-1);
+  // Refund/dispute is a permanent stop for this subscription. A replacement
+  // needs a separately verified binding, never a later active webhook replay.
+  if (last && (terminal(last.status) || facts.retrievedAt < last.retrievedAt ||
+      (facts.eventAt <= last.eventAt && !terminal(facts.status)))) return fail();
+  if (facts.status === 'past_due_grace' && last?.status === 'past_due_grace' && facts.pastDueSince !== last.pastDueSince) return fail();
+  const body = { ...facts, schemaVersion: 1 as const, source: 'verified-workspace-provider' as const,
+    sequence: state.revision + 1, previousHash: last?.contentHash ?? null };
+  const receipt = Object.freeze({ ...body, contentHash: await recordHash(body) });
+  return Object.freeze({ ...state, revision: receipt.sequence, receipts: Object.freeze([...state.receipts, receipt]) });

+ * verifier, signature failure or retrieval failure leaves the aggregate intact.

+  rawBody: Uint8Array, signature: string, now: number): Promise<WorkspaceBilling> {

+const READS = new Set<keyof WorkspaceTransaction>([
+  'authority', 'lifecycle', 'grants', 'billing', 'capacity', 'plans', 'receivedPlans',
+  'receipts', 'planSource', 'activationAvailable', 'reconciliation', 'reconciliationSource', 'previsitRecords', 'templates',

+/** A repository read is a value, never permission to adopt a later mutable
+ * alias. Keep its original bytes as well as a private copy: cloning alone
+ * hides retirement during a subsequent awaited final check. This guard does
+ * not replace the repository's serial transaction/current-source contract. */
+export function snapshotWorkspaceTransaction(tx: WorkspaceTransaction): WorkspaceTransaction {
+  const retained: { value: unknown; fingerprint: string }[] = [];
+  function assertUnchanged() {
+    if (retained.some(read => canonical(read.value) !== read.fingerprint)) {
+      throw new ProfessionalAccessError('changed');
+    }
+  }
+  tx.beforeCommit(async () => assertUnchanged());
+  // The port may itself be frozen. An empty target avoids proxy invariants
+  // that would otherwise prohibit wrapping its non-writable methods.
+  return new Proxy({} as WorkspaceTransaction, {
+    get(_target, key) {
+      const method = Reflect.get(tx, key, tx);
+      if (key === 'beforeCommit') {
+        return (check: () => Promise<void>) => tx.beforeCommit(async () => {
+          assertUnchanged();
+          await check();
+          assertUnchanged();
+        });
+      }
+      if (READS.has(key as keyof WorkspaceTransaction) && typeof method === 'function') {
+        return async (...args: unknown[]) => {
+          const value: unknown = await Reflect.apply(method, tx, structuredClone(args));
+          assertUnchanged();
+          const captured = structuredClone(value);
+          retained.push({ value, fingerprint: canonical(captured) });
+          return captured;
+        };
+      }
+      return typeof method === 'function' ? method.bind(tx) : method;
+    },
+  });

+/** Owner explicitly accepts an exact issued version. Acknowledgement and
+ * capacity become atomic; neither applies the owner's actual routine. */
+export async function acceptWorkspacePlan(repository: WorkspaceRepository | null, token: string,
+  ownerId: string, workspaceId: string, command: { receiptId: string; planId: string; planHash: string; capacityId: string; expectedCapacityRevision: number; expectedResponseHash: string | null; confirmed: true }): Promise<void> {
+  if (!repository) unavailable();
+  const c = Object.freeze({ ...command });
+  if (c.confirmed !== true) changed();
+  await repository!.transaction({ workspaceId, ownerId }, async rawTx => {
+    const tx = snapshotWorkspaceTransaction(rawTx);
+    const owner = await ownerFence(tx, token, ownerId, 'write');
+    const grants = await decodeGrants(await tx.grants(ownerId), ownerId, tx.now());
+    const active = grants.active;
+    if (!active ||
+        active.workspaceId !== workspaceId || active.expiresAt <= tx.now() || !active.permissions.includes('plan_write') || !active.fields.includes('intended_plans') || !active.purposes.includes('routine_review')) changed();
+    const health = owner.initial.health;
+    if (!health?.active || health.deletionPending || health.ownerId !== ownerId || health.epoch !== active!.healthEpoch ||
+        health.consentReceiptId !== active!.consentReceiptId) throw new ProfessionalAccessError('health');
+    const plans = await decodeIntendedPlans(await tx.plans(ownerId, workspaceId), ownerId, workspaceId, tx.now());
+    const plan = plans.find(item => item.id === c.planId && item.contentHash === c.planHash);
+    if (!plan || plan.grantId !== active!.id || plan.grantEpoch !== active!.epoch || plan.grantVersion !== active!.version || plan.healthEpoch !== active!.healthEpoch) changed();
+    const billing = await currentBilling(tx, workspaceId);
+    if (workspaceBillingAccess(billing, tx.now()) !== 'write_allowed') throw new ProfessionalAccessError('billing');
+    if (!await tx.activationAvailable(workspaceId)) throw new ProfessionalAccessError('coverage');
+    const capacity = await decodeWorkspaceCapacity(await tx.capacity(workspaceId), workspaceId, tx.now());
+    const retained = await validatedReceivedCopies(tx, ownerId);
+    if (!retained.plans.some(copy => copy.id === plan!.id && copy.contentHash === plan!.contentHash)) changed();
+    const received = retained.receipts;
+    assertReviewedResponse(received, c);
+    const nextReceipts = await appendPlanReceipt(received, { ownerId, actorId: owner.initial.session.subjectId,
+      healthEpoch: active!.healthEpoch, consentReceiptId: active!.consentReceiptId, current: true, adult: true, deletionPending: false }, plan!, {
+      id: c.receiptId, kind: 'acknowledged', expectedPlanHash: c.planHash, confirmed: true,
+    }, tx.now());
+    if (nextReceipts.filter(receipt => receipt.planId === c.planId && receipt.planHash === c.planHash && receipt.kind !== 'received').at(-1)?.id !== c.receiptId) changed();
+    const tier = billing.receipts.at(-1)!.tier;
+    const reservation = capacity.active.find(item => item.ownerId === ownerId);
+    // Only a validated fresh reservation can report capacity exhaustion. Stale
+    // revisions, reused IDs, mismatched grants and corrupt history stay unknown.
+    if (/^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/u.test(c.capacityId) &&
+        c.expectedCapacityRevision === capacity.revision && !capacity.events.some(event => event.id === c.capacityId) &&
+        (!reservation || (reservation.grantId === active!.id && reservation.grantVersion === active!.version && reservation.grantEpoch === active!.epoch)) &&
+        ((!reservation && capacity.active.length >= WORKSPACE_LIMITS[tier].plans) ||
+          (capacity.active.length > WORKSPACE_LIMITS[tier].plans && !capacityPublicationAllowed(capacity, tier, ownerId)))) {
+      throw new ProfessionalAccessError('capacity');
+    }
+    const intendedReservation = { ownerId, planId: plan!.id, planHash: plan!.contentHash,
+      grantId: active!.id, grantVersion: active!.version, grantEpoch: active!.epoch };
+    const nextCapacity = await transitionWorkspaceCapacity(capacity, { id: c.capacityId, expectedRevision: c.expectedCapacityRevision,
+      kind: 'accept', selectedOwnerIds: null, reservation: intendedReservation }, owner.initial.session.subjectId, billing.receipts.at(-1)!.tier, tx.now());
+    // The retained command proves a historical acceptance, not that it still
+    // owns a place after a newer acceptance or staff release. Keep the receipts
+    // intact while refusing a misleading current-success response.
+    if (canonical(nextCapacity.active.find(item => item.ownerId === ownerId) ?? null) !== canonical(intendedReservation) ||
+        nextCapacity.events.filter(event => event.reservation?.ownerId === ownerId).at(-1)?.id !== c.capacityId) changed();
+    const source = canonical([grants, owner.initial.health, billing]);
+    tx.beforeCommit(async () => {
+      await owner.recheck();
+      const current = await tx.authority(token, ownerRequest(ownerId, 'write'));
+      const latestBilling = await currentBilling(tx, workspaceId);
+      const currentGrant = await tx.grants(ownerId);
+      if (canonical([currentGrant, current.health, latestBilling]) !== source || !currentGrant.active || currentGrant.active.expiresAt <= tx.now() ||
+          workspaceBillingAccess(latestBilling, tx.now()) !== 'write_allowed' || !await tx.activationAvailable(workspaceId) ||
+          canonical(await tx.plans(ownerId, workspaceId)) !== canonical(plans) || canonical(await tx.capacity(workspaceId)) !== canonical(nextCapacity) ||
+          canonical(await tx.receipts(ownerId)) !== canonical(nextReceipts) ||
+          canonical(await tx.receivedPlans(ownerId)) !== canonical(retained.plans)) changed();
+    });
+    await tx.saveCapacity(nextCapacity);
+    await tx.saveOwnerReceiptAndCopy(nextReceipts, plan!);
+  });

+  workspaceId: string, command: CapacityCommand): Promise<void> {

+    const plan = plans.find(item => item.id === selected.planId && item.contentHash === selected.planHash);
+    const grants = await decodeGrants(await tx.grants(ownerId), ownerId, tx.now());
+    const grant = grants.active;
+    if (!plan || !grant || grant.id !== plan.grantId || grant.epoch !== plan.grantEpoch || grant.version !== plan.grantVersion ||
+        grant.healthEpoch !== plan.healthEpoch || grant.workspaceId !== plan.workspaceId || grant.expiresAt <= tx.now() || !grant.fields.includes('intended_plans') || !grant.purposes.includes('routine_review') || !grant.permissions.includes('plan_write')) changed();
+    const health = owner.initial.health;
+    if (!health?.active || health.deletionPending || health.ownerId !== ownerId || health.epoch !== grant!.healthEpoch ||
+        health.consentReceiptId !== grant!.consentReceiptId) throw new ProfessionalAccessError('health');
+    const capacity = await decodeWorkspaceCapacity(await tx.capacity(plan!.workspaceId), plan!.workspaceId, tx.now());
+    const fingerprint = canonical([plans, receipts, grants, capacity, health]);
+    tx.beforeCommit(async () => {
+      await owner.recheck();
+      const current = await tx.authority(token, ownerRequest(ownerId, 'read'));
+      if (grant!.expiresAt <= tx.now() || canonical([await tx.receivedPlans(ownerId), await tx.receipts(ownerId), await tx.grants(ownerId), await tx.capacity(plan!.workspaceId), current.health]) !== fingerprint) changed();
+    });
+    return { text: receivedPlanText([plan!], receipts), workspaceId: plan!.workspaceId, expectedCapacityRevision: capacity.revision,
+      expectedResponseHash: planDecisions(receipts, selected.planId, selected.planHash).at(-1)?.contentHash ?? null };
+  });

+/** Minimal independently owned index. No current grant, billing, practice
+ * membership or health-processing consent is required to read retained copies.
+ * Validate every copy and receipt before projecting; never trust a stored label
+ * or allow a duplicate ID to silently select another historical version. */
+export async function ownerReceivedPlanIndex(repository: WorkspaceRepository | null, token: string, ownerId: string) {
+  if (!repository) unavailable();
+  return repository!.transaction({ workspaceId: null, ownerId }, async rawTx => {
+    const tx = snapshotWorkspaceTransaction(rawTx);
+    const owner = await ownerFence(tx, token, ownerId, 'read');
+    const { plans, receipts } = await validatedReceivedCopies(tx, ownerId);
+    const result = [];
+    for (const plan of plans) {
+      const latest = planDecisions(receipts, plan.id, plan.contentHash).at(-1) ??
+        receipts.filter(receipt => receipt.planId === plan.id && receipt.planHash === plan.contentHash).at(-1)!;
+      result.push({ planId: plan.id, planHash: plan.contentHash, sequence: plan.sequence, issuedAt: plan.createdAt,
+        latestResponse: { kind: latest.kind, recordedAt: latest.recordedAt } });
+    }
+    const fingerprint = canonical([plans, receipts]);
+    tx.beforeCommit(async () => {
+      await owner.recheck();
+      if (canonical([await tx.receivedPlans(ownerId), await tx.receipts(ownerId)]) !== fingerprint) changed();
+    });
+    return result;
+  });

+/** One explicit owner offboarding transaction: revoke the live grant epoch and
+ * release that practice's accepted-plan reservation together. It does not delete
+ * received history, the account, actual routines or the consumer subscription. */
+export async function endPracticeRelationship(repository: WorkspaceRepository | null, token: string, ownerId: string,
+  workspaceId: string, command: { id: string; capacityId: string; grantId: string; expectedGrantRevision: number; expectedCapacityRevision: number; confirmed: true }): Promise<void> {
+  if (!repository) unavailable();
+  const c = Object.freeze({ ...command });
+  await repository!.transaction({ workspaceId, ownerId }, async rawTx => {
+    const tx = snapshotWorkspaceTransaction(rawTx);
+    const owner = await ownerFence(tx, token, ownerId, 'write');
+    const grants = await decodeGrants(await tx.grants(ownerId), ownerId, tx.now());
+    const previousGrantId = grants.active?.id ?? grants.events.find(event => event.id === c.id)?.previousGrantId;
+    const grant = grants.events.find(event => event.grant?.id === c.grantId)?.grant;
+    if (!grant || grant.workspaceId !== workspaceId || previousGrantId !== c.grantId) changed();
+    const nextGrants = await transitionGrant(grants, { ownerId, actorId: owner.initial.session.subjectId,
+      healthEpoch: owner.initial.health?.epoch ?? grant!.healthEpoch, consentReceiptId: owner.initial.health?.consentReceiptId ?? grant!.consentReceiptId,
+      current: true, adult: true, deletionPending: false }, { id: c.id, expectedRevision: c.expectedGrantRevision,
+      kind: 'revoke', previousGrantId: c.grantId, grant: null, confirmed: c.confirmed }, tx.now());
+    const capacity = await decodeWorkspaceCapacity(await tx.capacity(workspaceId), workspaceId, tx.now());
+    const reservation = capacity.active.find(item => item.ownerId === ownerId);
+    let nextCapacity = capacity;
+    if (reservation) {
+      if (reservation.grantId !== c.grantId || reservation.grantEpoch !== grant!.epoch) changed();
+      // Revocation/release cannot depend on a healthy billing read. The tier in
+      // the capacity journal is metadata and cannot mint paid authority.
+      const tier = capacity.events.at(-1)?.tier;
+      if (!tier) changed();
+      nextCapacity = await transitionWorkspaceCapacity(capacity, { id: c.capacityId, expectedRevision: c.expectedCapacityRevision,
+        kind: 'release', reservation, selectedOwnerIds: null }, owner.initial.session.subjectId, tier!, tx.now());
+    } else {
+      const released = capacity.events.find(event => event.id === c.capacityId && event.kind === 'release' &&
+        event.reservation?.ownerId === ownerId && event.reservation.grantId === c.grantId);
+      if (released) {
+        nextCapacity = await transitionWorkspaceCapacity(capacity, { id: c.capacityId, expectedRevision: c.expectedCapacityRevision,
+          kind: 'release', reservation: released.reservation, selectedOwnerIds: null }, owner.initial.session.subjectId, released.tier, tx.now());
+      } else if (capacity.revision !== c.expectedCapacityRevision) changed();
+    }
+    tx.beforeCommit(async () => {
+      await owner.recheck();
+      if (canonical(await tx.grants(ownerId)) !== canonical(nextGrants) || canonical(await tx.capacity(workspaceId)) !== canonical(nextCapacity)) changed();
+    });
+    await tx.saveGrants(nextGrants);
+    await tx.saveCapacity(nextCapacity);
+  });
+}

+/** Factual selected-text pre-visit report. Read-through owner facts require
+ * their original source adapter; unavailable history is never inferred. */
+export async function professionalPreVisitReport(repository: WorkspaceRepository | null, token: string, request: AccessRequest,
+  selection: { productIds: readonly string[]; plans: readonly { planId: string; planHash: string }[];
+    expectedProducts?: readonly { productId: string; name: string }[]; records?: readonly PrevisitRecordSelection[];
+    recordFields?: readonly PrevisitRecordField[]; recordSourceHash?: string }): Promise<string> {
+  if (!repository) unavailable();
+  const { request: r, selection: selected } = structuredClone({ request, selection });
+  const recordSelection = selected.records ?? []; const recordFields = selected.recordFields ?? [];
+  const fields = [...(selected.productIds.length ? ['products'] : []), ...(selected.plans.length ? ['intended_plans'] : []), ...recordFields];
+  if (r.action !== 'client_plan_issue_respond_export' || !['read', 'export'].includes(r.operation) || r.purpose !== 'pre_visit' || !r.ownerId || !r.workspaceId ||
+      !fields.length || canonical([...r.fields].sort()) !== canonical([...fields].sort()) || selected.productIds.length > 100 || selected.plans.length > 100 || recordSelection.length > 100 ||
+      new Set(recordFields).size !== recordFields.length || recordFields.some(field => !['actual_routine', 'use_events', 'formulas'].includes(field)) ||
+      new Set(recordSelection.map(row => canonical([row.field, row.id]))).size !== recordSelection.length ||
+      (recordSelection.length ? !recordFields.length || !/^[a-f0-9]{64}$/u.test(selected.recordSourceHash ?? '') : recordFields.length > 0 || selected.recordSourceHash !== undefined) ||
+      recordSelection.some(row => !row || Object.keys(row).sort().join(',') !== 'contentHash,field,id' || !recordFields.includes(row.field) ||
+        typeof row.id !== 'string' || !row.id.trim() || row.id.length > 256 || !/^[a-f0-9]{64}$/u.test(row.contentHash)) ||
+      new Set(selected.productIds).size !== selected.productIds.length || new Set(selected.plans.map(item => item.planId)).size !== selected.plans.length ||
+      (selected.expectedProducts !== undefined && (selected.expectedProducts.length !== selected.productIds.length ||
+        selected.expectedProducts.some((product, index) => product.productId !== selected.productIds[index] ||
+          Object.keys(product).sort().join(',') !== 'name,productId')))) throw new ProfessionalAccessError('invalid_request');
+  return repository!.transaction({ workspaceId: r.workspaceId, ownerId: r.ownerId }, async rawTx => {
+    const tx = snapshotWorkspaceTransaction(rawTx);
+    const a = await tx.authority(token, r); admit(a, r, tx.now());
+    const lifecycle = await liveLifecycle(tx, r.ownerId!);
+    // A plan-only grant must not read the products projection, even internally.
+    const source = selected.productIds.length ? await validateClientPlanSource(await tx.planSource(r.ownerId!, r.workspaceId!),
+      r.ownerId!, r.workspaceId!, a.grant!.id, a.grant!.version) : null;
+    const plans = selected.plans.length ? await decodeIntendedPlans(await tx.plans(r.ownerId!, r.workspaceId!), r.ownerId!, r.workspaceId!, tx.now()) : [];
+    const products = selected.productIds.map(id => {
+      const matches = source!.products.filter(item => item.productId === id);
+      if (matches.length !== 1) changed();
+      return matches[0]!;
+    });
+    if (selected.expectedProducts !== undefined && canonical(products) !== canonical(selected.expectedProducts)) changed();
+    const chosen = selected.plans.map(item => {
+      const plan = plans.find(plan => plan.id === item.planId && plan.contentHash === item.planHash && plan.grantId === a.grant?.id &&
+        plan.grantEpoch === a.grant?.epoch && plan.grantVersion === a.grant?.version && plan.healthEpoch === a.grant?.healthEpoch);
+      if (!plan) changed(); return plan!;
+    });
+    const recordSource = await readPrevisitRecords(tx, r.ownerId!, r.workspaceId!, a.grant!, recordFields);
+    if (recordSource && recordSource.contentHash !== selected.recordSourceHash) changed();
+    const chosenRecords = recordSelection.map(item => {
+      const record = recordSource?.records.find(row => row.field === item.field && row.id === item.id && row.contentHash === item.contentHash);
+      if (!record) return changed(); return record;
+    });
+    const fingerprint = canonical([a, source, plans, lifecycle, recordSource]);
+    tx.beforeCommit(async () => {
+      const current = await tx.authority(token, r); admit(current, r, tx.now());
+      if (canonical([current, source ? await tx.planSource(r.ownerId!, r.workspaceId!) : null,
+        selected.plans.length ? await tx.plans(r.ownerId!, r.workspaceId!) : [], await liveLifecycle(tx, r.ownerId!),
+        await readPrevisitRecords(tx, r.ownerId!, r.workspaceId!, current.grant!, recordFields)]) !== fingerprint) changed();
+    });
+    return ['Selected factual pre-visit report', `Prepared ${new Date(tx.now()).toISOString()}`,
+      ...products.map(product => `Selected saved product: ${product.name}`),
+      ...chosen.flatMap(plan => [`Intended version ${plan.sequence}, issued ${new Date(plan.createdAt).toISOString()}`,
+        `Morning: ${plan.am.map(step => step.name).join(' → ') || 'No steps recorded'}`,
+        `Evening: ${plan.pm.map(step => step.name).join(' → ') || 'No steps recorded'}`, ...templateOriginLines(plan)]),
+      ...chosenRecords.flatMap(record => previsitRecordText(record, recordSource!.records)),
+      chosenRecords.length ? 'Only selected owner record facts are included. Device recorded dates are not independently trusted server times. Missing evidence is unknown; retracted use is not current evidence of use.' :
+        'Saved product names are not formula identification. Actual routine, recorded use and formula-at-use evidence are not included.',
+      'Intended plans are suggestions, not proof of use, treatment effect or a clinical assessment. No photos are included.',
+      'Previously downloaded copies cannot be recalled by withdrawing a grant or deleting records.'].join('\n\n');
+  });
