+async function awaitExportStep<T>(lease: AccountGenerationLease, operation: (signal: AbortSignal) => Promise<T>, retirement?: AbortSignal): Promise<T> {

+import { useLayoutEffect, useMemo, useRef, useState, useSyncExternalStore } from 'react';

+import { beginDataRightsConfirmation, beginDataRightsOperation, IDLE_DATA_RIGHTS_STATE, type DataRightsState, type PendingDataRightsAction } from './youDataRightsState';
+
+type Confirmation = Readonly<{ scope: object; state: DataRightsState; generation: number }>;
+
+/** Exact rendered confirmation, original account and original consent view.
+ * Export has a separate flight so it can never block deletion priority. */
+export function useDataRightsConfirmation(ownerUserId: string | null, source?: unknown) {
+  const identity = captureAccountIdentityGeneration();
+  const health = useSyncExternalStore(subscribeActiveHealthProcessingLeaseChanges, activeHealthProcessingLeaseSnapshot, () => null);
+  const scope = useMemo(() => Object.freeze({ ownerUserId, identity, health, source }), [ownerUserId, identity, health, source]);
+  const liveScope = useRef<object | null>(scope);
+  const live = useRef<Confirmation | null>(null);
+  const [rendered, setRendered] = useState<Confirmation | null>(null);
+  useLayoutEffect(() => {
+    liveScope.current = scope;
+    return () => { if (liveScope.current === scope) { liveScope.current = null; live.current = null; } };
+  }, [scope]);
+  function current() {
+    try { assertAccountIdentityGeneration(identity); } catch { return false; }
+    return liveScope.current === scope && activeHealthProcessingLeaseSnapshot() === health;
+  }
+  function open(action: PendingDataRightsAction) {
+    if (!current() || live.current) return;
+    let generation: number | null = null;
+    // Capture only; this operation never retains a lease across confirmation.
+    void runAccountGenerationOperation((lease) => { generation = lease.generation; }).catch(() => {});
+    if (generation === null) return;
+    const confirmation = Object.freeze({ scope, state: beginDataRightsConfirmation(IDLE_DATA_RIGHTS_STATE, action)!, generation });
+    live.current = confirmation;
+    setRendered(confirmation);
+  }
+  function cancel(confirmation: Confirmation | null) {
+    if (!current() || !confirmation || live.current !== confirmation || confirmation.state.phase !== 'confirming') return;
+    live.current = null;
+    setRendered(null);
+  }
+  function begin(confirmation: Confirmation | null, action: PendingDataRightsAction) {
+    if (!current() || !confirmation || live.current !== confirmation || !ownerUserId) return null;
+    const running = beginDataRightsOperation(confirmation.state, action, 1);
+    if (!running) return null;
+    const operation = Object.freeze({ ...confirmation, state: running });
+    live.current = operation; // Same-turn latch before invoking any service.
+    setRendered(operation);
+    return Object.freeze({
+      expectedInitiator: { generation: confirmation.generation, userId: ownerUserId },
+      isCurrent: current,
+      finish: () => { if (live.current === operation) { live.current = null; setRendered(null); } },
+    });
+  }
+  return { busy: rendered?.scope === scope && rendered.state.phase === 'running', confirmation: rendered?.scope === scope ? rendered : null, open, cancel, begin };

+import { validAppointmentBriefText } from './appointmentBrief';

+export const APPOINTMENT_BRIEF_DRAFT_KEY = 'layerwell.appointmentBriefDraft.v1';
+export type AppointmentBriefDraftText = Readonly<{ observations: string; questions: string }>;
+export type AppointmentBriefDraft = AppointmentBriefDraftText & Readonly<{
+  schemaVersion: 1;
+  ownerUserId: string;
+  processingEpoch: number;
+  revisionId: string;
+}>;
+
+export function decodeAppointmentBriefDraft(raw: string, owner: string, epoch?: number): AppointmentBriefDraft {
+  if (raw.length > 25_000) throw new Error('APPOINTMENT_BRIEF_DRAFT_INVALID');
+  const row = JSON.parse(raw) as AppointmentBriefDraft | null;
+  if (!row || Object.keys(row).sort().join(',') !== 'observations,ownerUserId,processingEpoch,questions,revisionId,schemaVersion' ||
+    row.schemaVersion !== 1 || !owner || row.ownerUserId !== owner || !Number.isSafeInteger(row.processingEpoch) || row.processingEpoch < 0 ||
+    epoch !== undefined && row.processingEpoch !== epoch ||
+    typeof row.revisionId !== 'string' || !/^[a-f0-9]{8}-[a-f0-9]{4}-4[a-f0-9]{3}-[89ab][a-f0-9]{3}-[a-f0-9]{12}$/u.test(row.revisionId) ||
+    !validAppointmentBriefText(row.observations) || !validAppointmentBriefText(row.questions) ||
+    !row.observations.trim() && !row.questions.trim()) throw new Error('APPOINTMENT_BRIEF_DRAFT_INVALID');
+  return Object.freeze(row);

+type DraftSave = Readonly<{ expectedRevisionId: string | null; raw: string }>;
+
+async function accessDraft(lease: HealthDataWriteLease, check: () => void,
+  change?: Readonly<{ expectedRevisionId: string | null; raw: string | null }>): Promise<AppointmentBriefDraft | null> {
+  const assertCurrent = () => { assertOriginalHealthDataWriteLease(lease); check(); };
+  assertCurrent();
+  const authorization = storedHealthAuthority(lease, 'APPOINTMENT_BRIEF_DRAFT_AUTHORITY');
+  const expected = change?.expectedRevisionId;
+  const proposed = change?.raw ?? null;
+  const next = proposed ? decodeAppointmentBriefDraft(proposed, lease.ownerUserId, lease.epoch) : null;
+  let result: AppointmentBriefDraft | null = null;
+  await updatePrivateItemWithSources(APPOINTMENT_BRIEF_DRAFT_KEY, authorization.keys, values => {
+    const raw = values.get(APPOINTMENT_BRIEF_DRAFT_KEY) ?? null;
+    const current = raw === null ? null : decodeAppointmentBriefDraft(raw, lease.ownerUserId, lease.epoch);
+    if (!change) { result = current; return raw; }
+    if (next && current?.revisionId === next.revisionId) {
+      if (raw !== proposed) throw new Error('APPOINTMENT_BRIEF_DRAFT_CONFLICT');
+      result = current;
+      return raw;
+    }
+    if ((current?.revisionId ?? null) !== expected) throw new Error('APPOINTMENT_BRIEF_DRAFT_CONFLICT');
+    result = next;
+    return proposed;
+  }, assertCurrent, undefined, undefined, authorization, assertCurrent);
+  assertCurrent();
+  return result;
+}
+
+export function loadAppointmentBriefDraft(lease: HealthDataWriteLease, check: () => void) {
+  return accessDraft(lease, check);
+}

+export function saveAppointmentBriefDraft(lease: HealthDataWriteLease, expectedRevisionId: string | null,
+  text: AppointmentBriefDraftText, check: () => void) {

+  return Object.freeze({ expectedRevisionId, raw });
