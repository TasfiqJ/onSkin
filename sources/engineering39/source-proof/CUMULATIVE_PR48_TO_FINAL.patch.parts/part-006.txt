+// A new page is staged in a slot outside the committed inventory. The small
+// encrypted commit record is the only publication point. Interruption leaves
+// the old inventory readable, or the new inventory with its exact staged page.
+async function updateJournal(lease: HealthDataWriteLease, sources: readonly string[], guard: () => void,
+  transform: (values: ReadonlyMap<string,string|null>, journal: ReturnType<typeof collection>) => { page: number; raw: string } | undefined): Promise<void> {
+  const keys = [...AUTH_KEYS, ...ROUTINE_DECISION_JOURNAL_KEYS, ...sources];
+  const update = (key: string, fn: (values: ReadonlyMap<string,string|null>) => string | null) =>
+    updatePrivateItemWithSources(key, keys, fn, guard, undefined, undefined, authorization(lease), guard);

+  const operation = journalWrites.then(async () => {
+    guard();

+    await update(ROUTINE_DECISION_COMMIT_KEY, values => {
+      const journal = collection(values, lease);
+      const next = transform(values, journal);
+      if (!next) return journal.commitRaw;
+      const refs = journal.commit?.pages ?? journal.pages.map((_, index) => ({
+        slot: index, hash: routineDecisionHash(values.get(ROUTINE_DECISION_SLOT_KEYS[index]!)),
+      }));
+      slot = ROUTINE_DECISION_SLOT_KEYS.findIndex((_, index) => !refs.some(ref => ref.slot === index));
+      if (slot < 0) throw Error(ROUTINE_DECISION_CONFLICT);
+      return journal.commitRaw ?? JSON.stringify(makeRoutineDecisionCommit(lease.ownerUserId, lease.epoch, refs, null, null));
+    });
+    if (slot === undefined) return;
+    const key = ROUTINE_DECISION_SLOT_KEYS[slot]!;
+    let prepared: { prior: string; bytes: string; page: number } | undefined;
+    await update(key, values => {
+      const journal = collection(values, lease);
+      const { commit, commitRaw } = journal;
+      if (!commit || !commitRaw || commit.pages.some(ref => ref.slot === slot)) throw Error(ROUTINE_DECISION_CONFLICT);
+      const next = transform(values, journal);
+      if (!next) return values.get(key) ?? null;
+      prepared = { prior: commitRaw, bytes: stageRoutineDecisionPage(next.raw), page: next.page };
+      return prepared.bytes;
+    });
+    if (!prepared) return;
+    const staged = prepared;
+    await update(ROUTINE_DECISION_COMMIT_KEY, values => {
+      const journal = collection(values, lease);
+      const { commit, commitRaw } = journal;
+      if (!commit || commitRaw !== staged.prior || values.get(key) !== staged.bytes) throw Error(ROUTINE_DECISION_CONFLICT);
+      const next = transform(values, journal);
+      if (!next || next.page !== staged.page || stageRoutineDecisionPage(next.raw) !== staged.bytes) throw Error(ROUTINE_DECISION_CONFLICT);
+      const refs = [...commit.pages];
+      refs[staged.page] = { slot: slot!, hash: routineDecisionHash(staged.bytes) };
+      return JSON.stringify(makeRoutineDecisionCommit(lease.ownerUserId, lease.epoch, refs, commit, commitRaw));
+    });
+    guard();
+  });
+  journalWrites = operation.catch(() => {});
+  await operation;
+}

+function verifyRetirementArchive(lease: HealthDataWriteLease, archived: ArchiveReadback | null) {
+  const scope = { ownerId: lease.ownerUserId, healthEpoch: lease.epoch };
+  if (!archived) throw Error(ROUTINE_DECISION_CONFLICT);

+      receipt.hash !== routineDecisionHash(['owner-routine-receipt-v1', scope, receipt.sequence, receipt.recordedAt, receipt.originalHash])) {

+  const expected = resolvedEvidence(journal, lease, candidate.id);

+function replacementPages(journal: ReturnType<typeof collection>, candidate: ArchiveInput) {
+  const first = journal.pages[0]!;
+  const decisions = journal.pages.flatMap(page => page.collection.decisions).filter(saved => saved.command.request.id !== candidate.id);
+  const receipts = new Map(journal.pages.flatMap(page => page.outcomes).map(outcome => [outcome.commandHash, outcome]));
+  let previous: RoutineDecisionPage | undefined;
+  const replacement: string[] = [];
+  let start = 0;
+  while (start < decisions.length) {
+    // Pack the largest ordered prefix accepted by the existing count and byte
+    // limits. Moving later commands forward makes interior capacity reusable.
+    let low = 1;
+    let high = Math.min(ROUTINE_DECISION_LIMIT, decisions.length - start);
+    let best: { page: RoutineDecisionPage; raw: string; count: number } | undefined;
+    while (low <= high) {
+      const count = Math.floor((low + high) / 2);
+      const batch = decisions.slice(start, start + count);
+      const page = { ...newRoutineDecisionPage({ ...first.collection, decisions: batch }, previous),
+        outcomes: batch.flatMap(saved => receipts.get(saved.contentHash) ?? []) };
+      const raw = encodeRoutineDecisionPage(page);
+      if (raw === null) high = count - 1;
+      else { best = { page, raw, count }; low = count + 1; }
+    }
+    if (!best || replacement.length === journal.pages.length) throw Error(ROUTINE_DECISION_CONFLICT);
+    replacement.push(best.raw);
+    previous = best.page;
+    start += best.count;
+  }
+  return replacement;
+}

+export async function prepareRoutineDecisionRetirement(lease: HealthDataWriteLease, original: ArchiveInput,
+  readCurrentArchive: (id: string, originalHash: string) => Promise<ArchiveReadback | null>, assertCallerCurrent: () => void) {

+  const archived = await readCurrentArchive(candidate.id, originalHash);
+  const retired = verifyRetirementArchive(lease, archived);
+  if (retired.originalHash !== originalHash) throw Error(ROUTINE_DECISION_CONFLICT);
+  const current = await readJournal(lease, assertCallerCurrent);
+  if (current.commitRaw !== journal.commitRaw) throw Error(ROUTINE_DECISION_CONFLICT);
+  requireOriginalEvidence(current, lease, candidate);
+  return Object.freeze({ expectedCommit: journal.commitRaw, retired: Object.freeze(retired),
+    replacementPages: Object.freeze(replacementPages(current, candidate)) });
+}

+/** Serial-store contract, exercised with the existing encrypted custodian.
+ * The host must verify archive membership, freshness and retirement authority
+ * in the same transaction as staging all writes and the receipt. The normal
+ * mobile writer does not expose this operation. */
+export function publishRoutineDecisionRetirement(lease: HealthDataWriteLease, input: RetirementPlan,
+  transact: (retired: RetirementPlan['retired'], stage: (snapshot: {
+    values: ReadonlyMap<string, string | null>;
+    archived: ArchiveReadback;
+    priorReceipt: string | null;
+  }) => { receipt: string; changes: ReadonlyMap<string, string | null> }, assertBeforeCommit: () => void) => Promise<string>,
+  assertCallerCurrent: () => void) {
+  const plan = Object.freeze({ ...input, retired: Object.freeze({ ...input.retired }), replacementPages: Object.freeze([...input.replacementPages]) });
+  const guard = () => { assertOriginalHealthDataWriteLease(lease); assertCallerCurrent(); };

+    guard();

+    if (!journal.commit) throw Error(ROUTINE_DECISION_CONFLICT);

+    if (priorReceipt !== null) {
+      if (priorReceipt !== intent || journal.entries.has(retired.id)) throw Error(ROUTINE_DECISION_CONFLICT);
+      return { receipt: intent, changes: new Map() };
+    }
+    if (journal.commitRaw !== plan.expectedCommit) throw Error(ROUTINE_DECISION_CONFLICT);
+    requireOriginalEvidence(journal, lease, archived.original);
+    const replacement = replacementPages(journal, archived.original);
+    if (routineDecisionHash(replacement) !== routineDecisionHash(plan.replacementPages)) throw Error(ROUTINE_DECISION_CONFLICT);
+    const changes = new Map<string, string | null>();
+    const refs = replacement.map((raw, index) => {
+      const slot = journal.commit!.pages[index]!.slot;
+      const bytes = stageRoutineDecisionPage(raw);
+      changes.set(ROUTINE_DECISION_SLOT_KEYS[slot]!, bytes);
+      return { slot, hash: routineDecisionHash(bytes) };
+    });
+    for (const key of ROUTINE_DECISION_SLOT_KEYS) {
+      if (!changes.has(key)) changes.set(key, null);
+    }
+    changes.set(ROUTINE_DECISION_COMMIT_KEY, JSON.stringify(makeRoutineDecisionCommit(lease.ownerUserId, lease.epoch,
+      refs, journal.commit, journal.commitRaw)));

+    if (successor.entries.has(retired.id) || successor.entries.size !== journal.entries.size - 1) throw Error(ROUTINE_DECISION_CONFLICT);
+    for (const [key, value] of changes) {
+      if ((values.get(key) ?? null) === value) changes.delete(key);
+    }
+    return { receipt: intent, changes };
+  }, guard).then(receipt => { guard(); return receipt; });

+async function prepareDecision(lease: HealthDataWriteLease, command: RoutineDecision, assertCallerCurrent: () => void,
+  source?: ReturnType<typeof bindActualRoutineSource44>): Promise<SavedRoutineDecision | null> {
+  const accepted = decodeRoutineDecision(command, lease.ownerUserId);
+  const guard = () => { assertOriginalHealthDataWriteLease(lease); source?.guard(); assertCallerCurrent(); }; guard();
+  let result: SavedRoutineDecision | null | undefined;
+  await updateJournal(lease, [COMPLETION_KEY, ...(source?.keys ?? [])], guard, (values, { entries, pages }) => {
+    source?.validate(values);
+    const candidate = makeSavedRoutineDecision(accepted, routineDecisionHash(values.get(CONSENT_KEY)), lease.ownerUserId);
+    if (entries.has(accepted.request.id)) { result = exact(entries, candidate).saved; return; }
+    const ledgerBytes = values.get(COMPLETION_KEY) ?? null;
+    const ledger = actualRoutineSnapshotFromBytes(ledgerBytes, lease.ownerUserId);
+    if (inspectRoutineDecision(ledger, accepted).kind !== 'not_found') throw Error(ROUTINE_DECISION_CONFLICT);
+    // Validate the proposed transition against the exact reserved ledger before
+    // using a recovery slot. Only the later guarded mutation can publish it.
+    let prospective: ActualRoutineLedger;
+    if (accepted.kind === 'retraction') {
+      const target = ledger.uses.find(row => row.id === accepted.event.id);
+      if (!target || target.kind !== 'use' || routineDecisionHash(target) !== routineDecisionHash(accepted.event)) throw Error(ROUTINE_DECISION_CONFLICT);
+      prospective = retractActualUse(ledger, target.id, accepted.expectedHash, accepted.request);
+    } else if (accepted.kind === 'clear') {
+      prospective = clearActualRoutine(ledger, accepted.request);
+    } else {
+      prospective = applyActualRoutine(ledger, accepted.facts, accepted.request);
+      // A no-change result requires the original live source proof as well.
+      if (source && prospective === ledger) {
+        result = null;
+        return;
+      }
+    }
+    if (prospective !== ledger) {
+      const envelope = decodeActualRoutineEnvelope(ledgerBytes, lease.ownerUserId);
+      encodeActualRoutineEnvelope(envelope ? envelope.completionBytes : ledgerBytes, prospective);
+    }
+    const empty = decodeRoutineDecisions(null, lease.ownerUserId, lease.epoch);
+    const last = pages.at(-1) ?? newRoutineDecisionPage(empty);
+    const append = (page: RoutineDecisionPage) => encodeRoutineDecisionPage({ ...page,
+      collection: { ...page.collection, decisions: [...page.collection.decisions, candidate] } });
+    const current = append(last);
+    result = candidate;
+    if (current !== null) return { page: last.index, raw: current };
+    const next = newRoutineDecisionPage(empty, last), raw = append(next);
+    if (next.index >= ROUTINE_DECISION_PAGE_KEYS.length || raw === null) throw Error('ROUTINE_DECISION_CAPACITY');
+    return { page: next.index, raw };
+  });
+  guard();
+  if (result === null) return null;
+  // Verify authenticated encrypted durable readback before dispatch is possible.
+  const readback = await readJournal(lease); guard();
+  return exact(readback.entries, result!).saved;
+}

+/** Append the exact immutable record receipt after authenticated inspection.
+ * Legacy acknowledgements stay historical evidence, with no inferred receipt. */
+export async function acknowledgeRoutineDecision(lease: HealthDataWriteLease, expected: SavedRoutineDecision, assertCallerCurrent: () => void): Promise<void> {
+  const accepted = JSON.parse(JSON.stringify(expected)) as SavedRoutineDecision;
+  const guard = () => { assertOriginalHealthDataWriteLease(lease); assertCallerCurrent(); }; guard();
+  await updateJournal(lease, [COMPLETION_KEY], guard, (values, { pages, entries }) => {
+    const { page: index, saved: found } = exact(entries, accepted);

+    const outcome = inspectRoutineDecision(ledger, found.command);
+    if ((outcome.kind !== 'recorded_current' && outcome.kind !== 'recorded_historical') || outcome.sequence === undefined) throw Error(ROUTINE_DECISION_CONFLICT);
+    const recordHash = outcomeRecordHash(ledger, found.command, outcome.sequence);
+    if (recordHash === undefined) throw Error(ROUTINE_DECISION_CONFLICT);
+    const page = pages[index]!;
+    const raw = encodeRoutineDecisionPage({ ...page, outcomes: [...page.outcomes,
+      { commandHash: found.contentHash, recordHash, sequence: outcome.sequence }] });
+    if (raw === null) throw Error('ROUTINE_DECISION_CAPACITY');
+    return { page: index, raw };
+  });
+  guard();
+}

+import type { PlanHistoryStep45 } from './planHistory45';
+
+type Sequence = Readonly<{ am: readonly PlanHistoryStep45[]; pm: readonly PlanHistoryStep45[] }>;
+/** Free factual comparison of exact bottle identity, names, phase and relative
+ * order. No formula equivalence, use, compatibility or advanced advice inferred. */
+export function sequenceComparisonLines(previous: Sequence | null, reviewed: Sequence): readonly string[] {
+  if (!previous) return ['No previous owner-applied sequence is available for comparison. This will be the first applied version.'];
+  const lines: string[] = [];
+  for (const phase of ['am', 'pm'] as const) {
+    const label = phase === 'am' ? 'Morning' : 'Evening';
+    const before = previous[phase]; const after = reviewed[phase];
+    const oldIds = new Set(before.map(step => step.productId)); const newIds = new Set(after.map(step => step.productId));
+    const added = after.filter(step => !oldIds.has(step.productId)); const removed = before.filter(step => !newIds.has(step.productId));
+    for (const step of added) lines.push(`${label}: add ${step.name} to this phase.`);
+    for (const step of removed) lines.push(`${label}: remove ${step.name} from this phase. Earlier versions and use entries stay unchanged.`);
+    for (const step of after) {
+      const prior = before.find(item => item.productId === step.productId);
+      if (prior && prior.name !== step.name) lines.push(`${label}: recorded name changes from ${prior.name} to ${step.name} for the same saved bottle.`);
+    }
+    const oldOrder = before.filter(step => newIds.has(step.productId)).map(step => step.productId);
+    const newOrder = after.filter(step => oldIds.has(step.productId)).map(step => step.productId);
+    const reordered = JSON.stringify(oldOrder) !== JSON.stringify(newOrder);
+    if (reordered) lines.push(`${label}: relative order of retained bottles changes.`);
+    if (!added.length && !removed.length && !reordered && after.every(step => before.find(item => item.productId === step.productId)?.name === step.name)) {
+      lines.push(`${label}: same saved bottles, recorded names and order.`);
+    }
+  }
+  lines.push('Names and sequence only. Bottle observations or other source facts may have changed even when these sequences match. This is not a formula, compatibility or use comparison.');
+  return Object.freeze(lines);
+}

+type ExportState = Readonly<{ scope: object; phase: 'idle' | 'preparing' | 'cleanup'; cleanupPending: boolean; feedback: string | null; operation?: object }>;
+
+/** The same account-rights export/recovery behavior in the active and paused
+ * shells. This scope only retires actions; storage still verifies its owner. */
+export function DataExportControls({ ownerUserId, source, disabled = false, onStart }: {
+  ownerUserId: string | null; source?: unknown; disabled?: boolean; onStart?: () => void;
+}) {
+  const identity = captureAccountIdentityGeneration();
+  const health = useSyncExternalStore(subscribeActiveHealthProcessingLeaseChanges, activeHealthProcessingLeaseSnapshot, () => null);
+  const scope = useMemo(() => Object.freeze({ ownerUserId, identity, health, source }), [ownerUserId, identity, health, source]);
+  const liveScope = useRef<object | null>(scope);
+  const disabledRef = useRef(disabled);
+  useLayoutEffect(() => { disabledRef.current = disabled; }, [disabled]);
+  const flight = useRef<{ scope: object; controller: AbortController } | null>(null);
+  const [state, setState] = useState<ExportState>(() => ({ scope, phase: 'idle', cleanupPending: false, feedback: null }));
+  const liveState = useRef(state);
+  const publish = (next: ExportState) => { liveState.current = next; setState(next); };
+  useLayoutEffect(() => {
+    liveScope.current = scope;
+    if (liveState.current.scope !== scope) publish({ scope, phase: 'idle', cleanupPending: false, feedback: null });
+    return () => {
+      if (liveScope.current === scope) liveScope.current = null;
+      if (flight.current?.scope === scope) {
+        const retired = flight.current;
+        flight.current = null;
+        retired.controller.abort();
+      }
+    };
+  }, [scope]);
+  const current = () => {
+    try { assertAccountIdentityGeneration(identity); }
+    catch { return false; }
+    return liveScope.current === scope && activeHealthProcessingLeaseSnapshot() === health;
+  };
+  const shown = state.scope === scope ? state : { scope, phase: 'idle' as const, cleanupPending: false, feedback: null };
+
+  async function start(cleanupOnly: boolean, expected: ExportState) {
+    if (liveState.current !== expected || !current() || disabledRef.current || flight.current || (!cleanupOnly && liveState.current.scope === scope && liveState.current.cleanupPending)) return;

+    publish({ scope, phase: cleanupOnly ? 'cleanup' : 'preparing', cleanupPending: cleanupOnly, feedback: null, operation });
+    onStart?.();
