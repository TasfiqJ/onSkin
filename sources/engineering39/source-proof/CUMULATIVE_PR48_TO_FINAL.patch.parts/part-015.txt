+export function createPrevisitController(request: typeof fetch, render: (state: PrevisitState) => void, download: (text: string) => void) {
+  let state: PrevisitState = { kind: 'idle' }; let generation = 0; let pending: AbortController | null = null;
+  let context: string | null = null; let source: Source | null = null; let reviewed: Selection | null = null;
+  const show = (next: PrevisitState) => { state = next; render(next); };
+  const clear = () => { generation++; pending?.abort(); pending = null; context = null; source = null; reviewed = null; show({ kind: 'idle' }); };
+  async function send(input: unknown, expected: 'source' | 'review' | 'export', fields?: Fields) {
+    if (pending) return;
+    const admitted = ++generation; const controller = new AbortController(); pending = controller; show({ kind: 'loading' });
+    const timeout = setTimeout(() => controller.abort(), 15_000);
+    try {
+      const response = await untilAborted(controller.signal, request('/api/professional/previsit', { method: 'POST', cache: 'no-store', credentials: 'same-origin', redirect: 'error',
+        headers: { 'Content-Type': 'application/json', 'X-OnSkin-Request': 'previsit-v1' }, body: JSON.stringify(input), signal: controller.signal }));
+      if (generation !== admitted) return;
+      if (!response.ok || response.headers.get('Cache-Control') !== 'no-store' || !response.headers.get('Content-Type')?.startsWith('application/json')) return invalid();
+      const raw = await untilAborted(controller.signal, response.text()); if (generation !== admitted) return;
+      if (raw.length > 2_500_000) return invalid();
+      const result: unknown = JSON.parse(raw); keys(result, expected === 'source' ? ['kind', 'context', 'source'] : ['kind', 'context', 'text']);
+      if (result.kind !== expected || !id(result.context) || (context !== null && result.context !== context)) return invalid();
+      context = result.context;
+      if (expected === 'source') { source = decodeSource(result.source, fields!); show({ kind: 'source', source }); }
+      else {
+        if (typeof result.text !== 'string' || !result.text.trim()) return invalid();
+        if (expected === 'review') show({ kind: 'review', text: result.text });
+        else {
+          download(result.text);
+          if (generation !== admitted) return;
+          source = null; reviewed = null; show({ kind: 'downloaded' });
+        }
+      }
+    } catch { if (generation === admitted) { source = null; reviewed = null; context = null; show({ kind: 'unavailable' }); } }
+    finally { clearTimeout(timeout); if (generation === admitted) pending = null; }
+  }
+  return {
+    clear,
+    whenCurrent(rendered: PrevisitState, action: () => unknown) { if (rendered === state) return action(); },
+    async load(fields: Fields) {
+      if (pending) return;
+      if (!fields.length || fields.length > 5 || new Set(fields).size !== fields.length || fields.some(field => !['products', 'intended_plans', ...PREVISIT_RECORD_FIELDS].includes(field))) return;
+      source = null; reviewed = null;
+      await send({ kind: 'source', context, fields }, 'source', Object.freeze([...fields]));
+    },
+    async review(products: readonly Product[], plans: readonly Plan[], records: readonly PrevisitRecord[] = []) {
+      if (pending || state.kind !== 'source' || !source || !context || products.length + plans.length + records.length === 0 || records.length > 100 ||
+          new Set(products.map(product => product.productId)).size !== products.length || new Set(plans.map(plan => plan.planId)).size !== plans.length ||
+          products.some(product => !source!.products.some(original => canonical(original) === canonical(product))) ||
+          plans.some(plan => !source!.plans.some(original => canonical(original) === canonical(plan))) ||
+          new Set(records.map(row => canonical([row.field, row.id]))).size !== records.length ||
+          records.some(row => !source!.records.some(original => canonical(original) === canonical(row)))) return;
+      reviewed = Object.freeze({ products: Object.freeze(products.map(product => Object.freeze({ ...product }))),
+        plans: Object.freeze(plans.map(plan => Object.freeze({ planId: plan.planId, planHash: plan.planHash }))),
+        ...(records.length ? { records: Object.freeze(records.map(({ field, id, contentHash }) => Object.freeze({ field, id, contentHash }))),
+          recordFields: Object.freeze(source.fields.filter((field): field is PrevisitRecordField => PREVISIT_RECORD_FIELDS.includes(field as PrevisitRecordField))),
+          recordSourceHash: source.recordSourceHash! } : {}) });
+      await send({ kind: 'review', context, grantVersion: source.grantVersion, grantEpoch: source.grantEpoch, ...reviewed }, 'review');
+    },
+    async export() {
+      if (pending || state.kind !== 'review' || !reviewed || !source || !context) return;
+      await send({ kind: 'export', context, grantVersion: source.grantVersion, grantEpoch: source.grantEpoch, ...reviewed }, 'export');
+    },
+  };
+}

+  Readonly<{ kind: 'source' | 'saved'; scope: Scope; outcome?: QueueOutcome }> |
+  Readonly<{ kind: 'editing'; scope: Scope; row: QueueRevision; draft?: Extract<QueueCommand, { kind: 'transition' }> }> |
+  Readonly<{ kind: 'review' | 'saving' | 'checking' | 'unknown'; scope: Scope; command: QueueCommand; checked?: true }>;
+const fail = (): never => { throw new Error('QUEUE_UNAVAILABLE'); };
+const hash = (v: unknown): v is string => typeof v === 'string' && /^[a-f0-9]{64}$/u.test(v);
+const safe = (v: unknown): v is string => typeof v === 'string' && v.trim().length > 0 && v.length <= 2000 && !/[\p{Cf}\p{Cs}\u0000-\u001f\u007f-\u009f]/u.test(v);
+async function decodeModel(value: unknown): Promise<DeliveredModel> {
+  if (!value || typeof value !== 'object' || Array.isArray(value)) return fail(); const m = value as DeliveredModel;
+  const fields = Object.keys(m).sort().join(',');
+  if (!['acknowledgements,events,grantEpoch,grantVersion,plans,responses,rows,sourceHash,writeAvailable',
+        'acknowledgements,events,grantEpoch,grantVersion,placementAt,plans,responses,rows,sourceHash,writeAvailable'].includes(fields) ||
+      ('placementAt' in m && (!Number.isSafeInteger(m.placementAt) || m.placementAt! < 0 || m.placementAt! > 8.64e15)) ||
+      !hash(m.sourceHash) || typeof m.writeAvailable !== 'boolean' || ![m.grantVersion, m.grantEpoch].every(v => Number.isSafeInteger(v) && v > 0) ||
+      !Array.isArray(m.events) || m.events.length > 1000 || !Array.isArray(m.plans) || m.plans.length > 100 || !Array.isArray(m.rows) || m.rows.length > 5000 ||
+      !Array.isArray(m.acknowledgements) || m.acknowledgements.length > 1000 || !Array.isArray(m.responses) || m.responses.length > 100) return fail();
+  for (const p of m.plans) if (!p || Object.keys(p).sort().join(',') !== 'planHash,planId,sequence' || !hash(p.planHash) || !safe(p.planId) || !Number.isSafeInteger(p.sequence) || p.sequence < 1) return fail();
+  if (new Set(m.plans.map(p => p.planId)).size !== m.plans.length) return fail();
+  for (const e of m.events) if (!e || Object.keys(e).sort().join(',') !== 'contentHash,detail,grantVersion,id,kind,ownerId,planHash,version,workspaceId' ||
+      !hash(e.contentHash) || !safe(e.detail) || !safe(e.id) || !Number.isSafeInteger(e.version) || e.version < 1 || e.grantVersion !== m.grantVersion ||
+      !MATERIAL_EVENTS.includes(e.kind) || !m.plans.some(p => p.planHash === e.planHash)) return fail();
+  if (new Set(m.events.map(e => `${e.id}:${e.version}`)).size !== m.events.length) return fail();
+  for (const a of m.acknowledgements) if (!a || Object.keys(a).sort().join(',') !== 'contentHash,id,planHash' || !safe(a.id) || !hash(a.contentHash) || !m.plans.some(p => p.planHash === a.planHash)) return fail();
+  for (const r of m.responses) if (!r || Object.keys(r).sort().join(',') !== 'contentHash,id,kind,planHash,planId,recordedAt' ||
+      !safe(r.id) || !hash(r.contentHash) || !['acknowledged', 'declined'].includes(r.kind) || !Number.isFinite(r.recordedAt) || r.recordedAt < 0 || r.recordedAt > 8.64e15 ||
+      !m.plans.some(p => p.planId === r.planId && p.planHash === r.planHash)) return fail();
+  if (new Set(m.responses.map(r => r.planId)).size !== m.responses.length || new Set(m.responses.map(r => r.id)).size !== m.responses.length ||
+      m.acknowledgements.length !== m.responses.filter(r => r.kind === 'acknowledged').length ||
+      new Set(m.acknowledgements.map(a => a.id)).size !== m.acknowledgements.length ||
+      m.acknowledgements.some(a => !m.responses.some(r => r.kind === 'acknowledged' && r.id === a.id && r.contentHash === a.contentHash && r.planHash === a.planHash))) return fail();
+  const first = m.rows[0]; await decodeReconciliation(m.rows, first?.ownerId ?? '', first?.workspaceId ?? '');
+  if (m.rows.some(row => row.grantVersion !== m.grantVersion || !m.plans.some(p => p.planHash === row.planHash))) return fail();
+  return structuredClone(m);

+async function decodeOutcome(value: unknown, model: DeliveredModel, command: QueueCommand | undefined): Promise<QueueOutcome> {
+  if (!value || typeof value !== 'object') return fail();
+  const record = value as QueueOutcome;
+  const fields = Object.keys(value).sort().join(',');
+  if (record.kind === 'recorded_current' || record.kind === 'recorded_historical') {
+    const latest = model.rows.findLast(row => row.caseId === record.caseId);
+    const row = model.rows.find(row => row.caseId === record.caseId && row.revision === record.revision && row.contentHash === record.recordHash);
+    if (fields === 'caseId,kind,recordHash,revision' && latest && (latest.revision === record.revision) === (record.kind === 'recorded_current') && row) {
+      if ((model.placementAt !== undefined || command?.kind === 'enqueue' && command.expectedPlacement) &&
+          (!command || row.commandEvidence?.commandHash !== await recordHash(command) ||
+          row.commandEvidence.kind !== command.kind || row.commandEvidence.sourceHash !== command.sourceHash ||
+          row.commandEvidence.grantEpoch !== model.grantEpoch)) return fail();
+      return record;
+    }
+  } else if (fields === 'kind' && ['not_found', 'conflict', 'event_present', 'unverified_history'].includes(record.kind)) return record;
+  return fail();

+export function createReconciliationController(request: typeof fetch, onChange: (state: QueueViewState) => void, newId: () => string = () => crypto.randomUUID()) {
+  let state: QueueViewState = { kind: 'idle' }; let generation = 0; let abort: AbortController | null = null;
+  const show = (next: QueueViewState) => { state = next; onChange(next); };
+  async function post(input: { kind: 'source' | 'save' | 'inspect'; context: string | null; grantVersion?: number; grantEpoch?: number; command?: QueueCommand }, operation: number) {
+    const control = new AbortController(); abort?.abort(); abort = control;
+    const timer = setTimeout(() => control.abort(), 12_000);
+    try {
+      const response = await untilAborted(control.signal, request('/api/professional/reconciliation', { method: 'POST', credentials: 'same-origin', cache: 'no-store', redirect: 'error',
+        headers: { 'Content-Type': 'application/json', 'X-OnSkin-Request': 'reconciliation-v1' }, body: JSON.stringify(input), signal: control.signal }));
+      if (generation !== operation || control.signal.aborted) return fail();
+      if (response.status === 409) throw new Error('QUEUE_CHANGED');
+      if ([400, 401, 403].includes(response.status)) throw new Error('QUEUE_DENIED');
+      if (!response.ok || !response.headers.get('Cache-Control')?.toLowerCase().split(',').some(v => v.trim() === 'no-store') || !response.headers.get('Content-Type')?.startsWith('application/json')) return fail();
+      const raw = await untilAborted(control.signal, response.text()); if (raw.length > 3_000_000) return fail();
+      const result = JSON.parse(raw); if (!result || result.kind !== input.kind || !safe(result.context) ||
+        Object.keys(result).sort().join(',') !== (input.kind === 'source' ? 'context,kind,model' : 'context,kind,model,outcome')) return fail();
+      const model = await decodeModel(result.model); if (generation !== operation || control.signal.aborted) return fail();
+      const outcome = input.kind === 'source' ? null : await decodeOutcome(result.outcome, model, input.command);
+      if (generation !== operation || control.signal.aborted) return fail();
+      return { kind: input.kind, context: result.context as string, model, outcome };
+    } finally { clearTimeout(timer); if (abort === control) abort = null; }
+  }
+  const api = {
+    get state() { return state; }, get generation() { return generation; },
+    clear() { generation++; abort?.abort(); abort = null; show({ kind: 'idle' }); },
+    async load(admitted: number) {
+      if (admitted !== generation || state.kind === 'saving' || state.kind === 'loading') return;
+      generation++; const op = generation; show({ kind: 'loading' });
+      try { const result = await post({ kind: 'source', context: null }, op); if (generation !== op) return;
+        if (result.kind !== 'source') return fail(); generation++; show({ kind: 'source', scope: { context: result.context, model: result.model } });
+      } catch (error) { if (generation === op) { generation++; show({ kind: error instanceof Error && error.message === 'QUEUE_CHANGED' ? 'changed' : 'unavailable' }); } }
+    },
+    filter(admitted: number, filter: QueueFilter) {
+      if (admitted !== generation || !['source', 'saved'].includes(state.kind) || !('scope' in state) ||
+          !['all', 'open', ...Object.keys(QUEUE_TRANSITIONS)].includes(filter.state) || !['all', ...MATERIAL_EVENTS].includes(filter.kind) ||
+          Object.keys(filter).sort().join(',') !== 'kind,state') return;
+      generation++; show({ ...state, scope: { ...state.scope, filter: Object.freeze({ ...filter }) } });
+    },
+    reviewEvent(admitted: number, eventId: string, eventVersion: number) {
+      if (admitted !== generation || !['source', 'saved'].includes(state.kind) || !('scope' in state) || !state.scope.model.writeAvailable) return;
+      const scope = state.scope;
+      const event = scope.model.events.find(e => e.id === eventId && e.version === eventVersion &&
+        (!scope.filter || scope.filter.kind === 'all' || e.kind === scope.filter.kind)); if (!event) return;
+      if (scope.model.rows.some(row => row.events.some(prior => prior.id === event.id && prior.version === event.version))) return;
+      const expectedPlacement = scope.model.placementAt === undefined ? undefined : queuePlacement(scope.model.rows, event, scope.model.placementAt);
+      generation++; show({ kind: 'review', scope: state.scope, command: { kind: 'enqueue', sourceHash: state.scope.model.sourceHash,
+        eventId, eventVersion, eventHash: event.contentHash, caseId: newId(), confirmed: true, ...(expectedPlacement && { expectedPlacement }) } });
+    },
+    edit(admitted: number, caseId: string) {
+      if (admitted !== generation || !['source', 'saved'].includes(state.kind) || !('scope' in state)) return;
+      const row = filterCurrentCases(state.scope.model.rows, state.scope.filter ?? allRecords).find(row => row.caseId === caseId); if (!row) return;
+      generation++; show({ kind: 'editing', scope: state.scope, row });
+    },
+    reviewTransition(admitted: number, fields: Omit<Extract<QueueCommand, { kind: 'transition' }>, 'kind' | 'sourceHash' | 'caseId' | 'expectedHash' | 'confirmed'>): boolean {
+      if (admitted !== generation || state.kind !== 'editing' || !state.scope.model.writeAvailable || !QUEUE_TRANSITIONS[state.row.state].includes(fields.next) || !safe(fields.reason) ||
+          (['resolved', 'declined_referred', 'relationship_closed'].includes(fields.next) && (!safe(fields.communication) ||
+            ((fields.changedPlanHash ?? state.row.changedPlanHash) !== null && fields.acknowledgementId === null && !safe(fields.nonAcknowledgementReason))))) return false;
+      const effectivePlanHash = fields.changedPlanHash ?? state.row.changedPlanHash;
+      if (fields.acknowledgementId !== null && !state.scope.model.acknowledgements.some(a => a.id === fields.acknowledgementId && a.contentHash === fields.acknowledgementHash && a.planHash === effectivePlanHash)) return false;
+      const command: QueueCommand = structuredClone({ kind: 'transition', sourceHash: state.scope.model.sourceHash, caseId: state.row.caseId, expectedHash: state.row.contentHash, confirmed: true, ...fields });
+      generation++; show({ kind: 'review', scope: state.scope, command }); return true;
+    },
+    back(admitted: number) {
+      if (admitted !== generation || !['review', 'editing'].includes(state.kind) || !('scope' in state)) return;
+      generation++; show({ kind: 'source', scope: state.scope });
+    },
+    async confirm(admitted: number, mode: 'save' | 'inspect' = 'save') {
+      if (admitted !== generation || !('command' in state) || (mode === 'inspect' ? state.kind !== 'unknown' : !['review', 'unknown'].includes(state.kind))) return;
+      const { scope, command } = state; generation++; const op = generation; show({ kind: mode === 'inspect' ? 'checking' : 'saving', scope, command });
+      try { const result = await post({ kind: mode, context: scope.context, grantVersion: scope.model.grantVersion, grantEpoch: scope.model.grantEpoch, command }, op);
+        if (generation !== op) return;
+        if (!result.outcome || result.context !== scope.context ||
+            result.model.grantVersion !== scope.model.grantVersion || result.model.grantEpoch !== scope.model.grantEpoch) return fail();
+        generation++;
+        if (result.outcome.kind === 'conflict') show({ kind: 'changed' });
+        else if (result.outcome.kind === 'not_found') show({ kind: 'unknown', scope, command, ...(mode === 'inspect' ? { checked: true } : {}) });
+        else show({ kind: 'saved', outcome: result.outcome, scope: { ...scope, model: result.model } });
+      } catch (error) { if (generation !== op) return; generation++;
+        show(error instanceof Error && error.message === 'QUEUE_CHANGED' ? { kind: 'changed' } : error instanceof Error && error.message === 'QUEUE_DENIED' ? { kind: 'unavailable' } : { kind: 'unknown', scope, command });
+      }
+    },
+  }; return api;
+}
