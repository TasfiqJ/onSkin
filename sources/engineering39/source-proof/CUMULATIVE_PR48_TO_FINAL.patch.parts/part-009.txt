+function invalid(): never { throw new Error(MANUAL_INTAKE_DRAFT_INVALID); }
+function record(value: unknown, keys: readonly string[]): Record<string, unknown> {
+  if (!value || typeof value !== 'object' || Array.isArray(value) ||
+      Object.keys(value).sort().join('|') !== [...keys].sort().join('|')) invalid();
+  return value as Record<string, unknown>;

+function text(value: unknown, max: number, multiline = false): string {
+  if (typeof value !== 'string' || value.length > max || utf8ToBytes(value).length > max * 3 ||
+      (multiline ? /[\u0000-\u0008\u000b\u000c\u000e-\u001f\u007f-\u009f\ud800-\udfff]/u : /[\u0000-\u001f\u007f-\u009f\ud800-\udfff]/u).test(value)) invalid();
+  return value;
+}
+function nullableText(value: unknown, max: number): string | null {
+  return value === null ? null : text(value, max);
+}
+function nonempty(value: unknown, max: number): string {
+  const result = text(value, max);
+  if (!result || result.trim() !== result) invalid();
+  return result;
+}
+function integer(value: unknown, min: number, max = Number.MAX_SAFE_INTEGER): number {
+  if (!Number.isSafeInteger(value) || (value as number) < min || (value as number) > max) invalid();
+  return value as number;
+}
+function uuid(value: unknown): string {
+  const result = nonempty(value, 36);
+  if (!/^[0-9a-f]{8}-[0-9a-f]{4}-4[0-9a-f]{3}-[89ab][0-9a-f]{3}-[0-9a-f]{12}$/.test(result)) invalid();
+  return result;
+}
+function timestamp(value: unknown): string {
+  const result = nonempty(value, 30);
+  if (!Number.isFinite(Date.parse(result)) || new Date(result).toISOString() !== result) invalid();
+  return result;
+}
+function date(value: unknown): string | null {
+  if (value === null) return null;
+  const result = text(value, 10);
+  if (!/^\d{4}-\d{2}-\d{2}$/.test(result) || !Number.isFinite(Date.parse(result)) ||
+      new Date(result).toISOString().slice(0, 10) !== result) invalid();
+  return result;
+}
+function category(value: unknown): IntakeDraft['category'] {
+  if (value !== null && !PRODUCT_CATEGORIES.some(row => row.id === value)) invalid();
+  return value as IntakeDraft['category'];
+}
+function paoSource(value: unknown): PaoSource {
+  if (value !== 'label' && value !== 'catalog' && value !== 'category_default' && value !== 'unknown') invalid();
+  return value;
+}
+function score(value: unknown, max: number): number | null {
+  if (value === null) return null;
+  if (typeof value !== 'number' || !Number.isFinite(value) || value < 0 || value > max) invalid();
+  return value;

+function decodeForm(value: unknown): ManualIntakeForm {
+  const row = record(value, ['name', 'brand', 'barcode', 'ingredients', 'category', 'eightDigitFormat', 'addedVia']);
+  if (row.addedVia !== 'manual' || (row.eightDigitFormat !== null && row.eightDigitFormat !== 'ean8' && row.eightDigitFormat !== 'upc_e')) invalid();
+  return Object.freeze({ name: text(row.name, 120), brand: text(row.brand, 120), barcode: text(row.barcode, 20),
+    ingredients: text(row.ingredients, 16_000, true), category: category(row.category),
+    eightDigitFormat: row.eightDigitFormat, addedVia: 'manual' });
+}
+function decodeIntake(value: unknown): IntakeDraft {
+  const row = record(value, Object.keys(createIntakeDraft()));
+  if (row.addedVia !== 'manual') invalid();
+  const quality = row.catalogMatchQuality;
+  if (quality !== null && !['manual', 'verified', 'usable', 'limited', 'unverified', 'blocked'].includes(quality as string)) invalid();
+  if (!Array.isArray(row.ingredients) || row.ingredients.length > 500) invalid();
+  const ingredients = row.ingredients.map(entry => nonempty(entry, 256));
+  Object.freeze(ingredients);
+  let recovery: IntakeDraft['catalogRecoveryToken'] = null;
+  if (row.catalogRecoveryToken !== null) {
+    const token = record(row.catalogRecoveryToken, ['barcode', 'productId']);
+    recovery = Object.freeze({ barcode: nonempty(token.barcode, 20), productId: nonempty(token.productId, 128) });
+  }
+  return Object.freeze({ name: text(row.name, 120), brand: nullableText(row.brand, 120),
+    category: category(row.category), barcode: nullableText(row.barcode, 20),
+    catalogProductId: nullableText(row.catalogProductId, 128), catalogSourceId: nullableText(row.catalogSourceId, 128),
+    catalogSource: nullableText(row.catalogSource, 128), catalogSourceName: nullableText(row.catalogSourceName, 256),
+    catalogSourceRef: nullableText(row.catalogSourceRef, 512), catalogSourceUrl: nullableText(row.catalogSourceUrl, 2048),
+    catalogSourceSnapshotDate: date(row.catalogSourceSnapshotDate), catalogMatchQuality: quality as IntakeDraft['catalogMatchQuality'],
+    dataQualityScore: score(row.dataQualityScore, 100), ingredientParseStatus: nullableText(row.ingredientParseStatus, 80),
+    ingredientParseConfidence: score(row.ingredientParseConfidence, 1), parserVersion: nullableText(row.parserVersion, 128),
+    sourceDisclosureAckAt: row.sourceDisclosureAckAt === null ? null : timestamp(row.sourceDisclosureAckAt),
+    ingredients, addedVia: 'manual', paoMonths: row.paoMonths === null ? null : integer(row.paoMonths, 1, 120),
+    paoSource: paoSource(row.paoSource), expiryDate: date(row.expiryDate), catalogRecoveryToken: recovery });
+}
+function decodeOpening(value: unknown): ManualIntakeOpening {
+  const row = record(value, ['mode', 'selectedDate', 'paoMonths', 'paoSource', 'customPaoText']);
+  const mode = row.mode;
+  if (mode !== null && mode !== 'just' && mode !== 'pick' && mode !== 'unopened') invalid();
+  return Object.freeze({ mode, selectedDate: date(row.selectedDate),
+    paoMonths: row.paoMonths === null ? null : integer(row.paoMonths, 1, 120),
+    paoSource: paoSource(row.paoSource), customPaoText: text(row.customPaoText, 3) });
+}
+function decodeSubmission(value: unknown): ManualIntakeSubmission {
+  const keys = Object.keys(createIntakeDraft()).filter(key => key !== 'catalogRecoveryToken');
+  const row = record(value, [...keys, 'operationId', 'openedAt', 'isOpened']);
+  const { operationId, openedAt, isOpened, ...payload } = row;
+  const intake = decodeIntake({ ...payload, catalogRecoveryToken: null });
+  if (typeof isOpened !== 'boolean' || !intake.name.trim()) invalid();
+  const openingDate = date(openedAt);
+  if (isOpened !== (openingDate !== null)) invalid();
+  const { catalogRecoveryToken: _recovery, ...product } = intake;
+  return Object.freeze({ ...product, operationId: uuid(operationId), openedAt: openingDate, isOpened });
+}
+export function decodeManualIntakeDraftFields(value: unknown): ManualIntakeDraftFields {
+  const row = record(value, ['form', 'intake', 'phase', 'opening', 'attempted']);
+  if (row.phase !== 'manual' && row.phase !== 'opened') invalid();
+  const form = decodeForm(row.form), intake = decodeIntake(row.intake), opening = decodeOpening(row.opening);
+  let attempted: ManualIntakeDraftFields['attempted'] = null;
+  if (row.attempted !== null) {
+    const attempt = record(row.attempted, ['input']);
+    const input = decodeSubmission(attempt.input);
+    if (row.phase !== 'opened' || opening.mode === null || input.isOpened !== (opening.mode !== 'unopened') ||
+        (opening.mode === 'pick' && input.openedAt !== opening.selectedDate)) invalid();
+    const { catalogRecoveryToken: _recovery, ...expected } = intake;
+    if (JSON.stringify(input) !== JSON.stringify({ ...expected, name: intake.name.trim(),
+      paoMonths: opening.paoMonths, paoSource: opening.paoSource, operationId: input.operationId,
+      openedAt: input.openedAt, isOpened: input.isOpened })) invalid();
+    attempted = Object.freeze({ input });
+  }
+  return Object.freeze({ form, intake, phase: row.phase, opening, attempted });
+}
+export function makeManualIntakeDraft(value: Omit<ManualIntakeDraft, 'contentHash'>): ManualIntakeDraft {
+  const row = record(value, ['id', 'revision', 'createdAt', 'updatedAt', 'operationId', 'fields']);
+  const createdAt = timestamp(row.createdAt), updatedAt = timestamp(row.updatedAt);
+  if (updatedAt < createdAt) invalid();
+  const fields = decodeManualIntakeDraftFields(row.fields), operationId = uuid(row.operationId);
+  if (fields.attempted !== null && fields.attempted.input.operationId !== operationId) invalid();
+  const body = { id: uuid(row.id), revision: integer(row.revision, 1), createdAt, updatedAt, operationId, fields };
+  return Object.freeze({ ...body, contentHash: bytesToHex(sha256(utf8ToBytes(JSON.stringify(body)))) });
+}
+export function decodeManualIntakeDraftCollection(raw: string | null, ownerUserId: string, processingEpoch?: number): ManualIntakeDraftCollection {
+  nonempty(ownerUserId, 128);
+  if (raw === null) return Object.freeze({ schemaVersion: 1, provenance: 'owner_unconfirmed_manual_intake', ownerUserId,
+    processingEpoch: integer(processingEpoch, 1), drafts: Object.freeze([]) });
+  if (raw.length > MANUAL_INTAKE_DRAFT_MAX_BYTES || utf8ToBytes(raw).length > MANUAL_INTAKE_DRAFT_MAX_BYTES) invalid();
+  let parsed: unknown;
+  try { parsed = JSON.parse(raw) as unknown; } catch { invalid(); }
+  const row = record(parsed, ['schemaVersion', 'provenance', 'ownerUserId', 'processingEpoch', 'drafts']);
+  if (row.schemaVersion !== 1 || row.provenance !== 'owner_unconfirmed_manual_intake') invalid();
+  if (row.ownerUserId !== ownerUserId || (processingEpoch !== undefined && row.processingEpoch !== processingEpoch)) throw new Error(MANUAL_INTAKE_DRAFT_AUTHORITY);
+  const epoch = integer(row.processingEpoch, 1);
+  if (!Array.isArray(row.drafts) || row.drafts.length > MANUAL_INTAKE_DRAFT_LIMIT) invalid();
+  const drafts = row.drafts.map(entry => {
+    const checked = record(entry, ['id', 'revision', 'createdAt', 'updatedAt', 'operationId', 'fields', 'contentHash']);
+    const { contentHash, ...body } = checked;
+    const draft = makeManualIntakeDraft(body as Omit<ManualIntakeDraft, 'contentHash'>);
+    if (draft.contentHash !== contentHash) invalid();
+    return draft;
+  });
+  if (new Set(drafts.map(draft => draft.id)).size !== drafts.length ||
+      new Set(drafts.map(draft => draft.operationId)).size !== drafts.length) invalid();
+  return Object.freeze({ schemaVersion: 1, provenance: 'owner_unconfirmed_manual_intake', ownerUserId,
+    processingEpoch: epoch, drafts: Object.freeze(drafts) });
+}

+/** Explicit Save Draft or Add write-ahead snapshot. No per-keystroke writes. */
+export async function saveManualIntakeDraft(lease: HealthDataWriteLease, request: Readonly<{
+  id: string;
+  expected: ManualIntakeDraft | null;
+  operationId: string;
+  fields: ManualIntakeDraftFields;
+}>, assertCallerCurrent: () => void): Promise<ManualIntakeDraft> {
+  const assertCurrent = () => { assertOriginalHealthDataWriteLease(lease); assertCallerCurrent(); };
+  assertCurrent();
+  const fields = decodeManualIntakeDraftFields(request.fields);
+  // Snapshot every caller value before the first await, including the CAS proof.
+  const expected = request.expected ? JSON.stringify(request.expected) : null;
+  const id = request.id, operationId = request.operationId;
+  let result: ManualIntakeDraft | undefined;
+  await updatePrivateItemWithSources(MANUAL_INTAKE_DRAFT_KEY, MANUAL_INTAKE_DRAFT_AUTHORIZATION_KEYS, values => {
+    assertCurrent();
+    const raw = values.get(MANUAL_INTAKE_DRAFT_KEY) ?? null;
+    const collection = decodeManualIntakeDraftCollection(raw, lease.ownerUserId, lease.epoch);
+    const existing = collection.drafts.find(row => row.id === id);
+    if ((existing ? JSON.stringify(existing) : null) !== expected ||
+        (existing && existing.operationId !== operationId) ||
+        collection.drafts.some(row => row.id !== id && row.operationId === operationId)) throw new Error(MANUAL_INTAKE_DRAFT_CONFLICT);
+    // Once Add has been attempted, a retry must keep the entire original
+    // snapshot. Even edits to raw fields could falsely describe what was saved.
+    if (existing?.fields.attempted && JSON.stringify(existing.fields) !== JSON.stringify(fields)) throw new Error(MANUAL_INTAKE_DRAFT_CONFLICT);
+    if (existing && JSON.stringify(existing.fields) === JSON.stringify(fields)) {
+      result = existing;
+      return raw;
+    }
+    if (!existing && collection.drafts.length >= MANUAL_INTAKE_DRAFT_LIMIT) throw new Error(MANUAL_INTAKE_DRAFT_CONFLICT);
+    const now = new Date().toISOString();
+    result = makeManualIntakeDraft({ id, revision: (existing?.revision ?? 0) + 1,
+      createdAt: existing?.createdAt ?? now, updatedAt: now, operationId, fields });
+    const next = JSON.stringify({ ...collection, drafts: [...collection.drafts.filter(row => row.id !== id), result] });
+    decodeManualIntakeDraftCollection(next, lease.ownerUserId, lease.epoch);
+    return next;
+  }, assertCurrent, undefined, undefined, manualIntakeDraftAuthorization(lease));
+  assertCurrent();
+  return result!;
+}

+export function compareRecordedFormulaVersions(
+  first: OwnedBottleHistorySnapshot38,
+  second: OwnedBottleHistorySnapshot38,
+  firstVersionId: string | null,
+  secondVersionId: string | null,
+  ownerUserId: string,

+    for (const snapshot of [first, second]) {
+      if (snapshot.ownedUnitId !== snapshot.productId || !snapshot.currentVersionId) return null;
+      validateOwnedBottleHistory38({
+        productId: snapshot.productId,
+        ownerUserId,
+        revision: snapshot.revision,
+        currentVersionId: snapshot.currentVersionId,
+        versions: snapshot.versions,
+      });
+    }

+    return null;
+  }
+  const left = first.versions.find((version) => version.id === firstVersionId);
+  const right = second.versions.find((version) => version.id === secondVersionId);
+  if (!left || !right) return null;

+    (version) =>
+      version.formulaVersion.labelStatus === 'user_confirmed_label' &&
+      version.basis === 'label_in_hand',

+      const display = `${token.section === 'main' ? '' : `${token.section}: `}${token.rawToken}`;
+      const key = display.trim().replace(/\s+/gu, ' ').toLowerCase();

+    }
+    return result;
+  };
+  const leftEntries = confirmed

+    : new Map<string, string>();
+  const rightEntries = confirmed
+    ? entries(right.formulaVersion.labelText)
+    : new Map<string, string>();

+  return {
+    left,
+    right,
+    sameRecord: left.id === right.id,
+    manufacturerFormulaDifference: 'UNCONFIRMED' as const,
+    onlyFirst: comparable
+      ? [...leftEntries].filter(([key]) => !rightEntries.has(key)).map(([, text]) => text)

+/** Compare recorded text only. No synonym inference, completeness claim,
+ * compatibility, clinical ranking, quota, persistence or provider lookup. */
+export function compareSavedProducts<T extends ComparableProduct>(
+  products: readonly T[],
+  leftId: string | null,
+  rightId: string | null,
+) {
+  if (!leftId || !rightId || leftId === rightId) return null;
+  const left = products.find((product) => product.id === leftId);
+  const right = products.find((product) => product.id === rightId);
+  if (!left || !right) return null;
+  const leftFacts = facts(left);
+  const rightFacts = facts(right);
+  const ingredientKey = (text: string) => text.trim().replace(/\s+/gu, ' ').toLowerCase();
+  const leftIngredients = new Map<string, string>();
+  for (const text of left.ingredients) {
+    const key = ingredientKey(text);
+    if (key && !leftIngredients.has(key)) leftIngredients.set(key, text.trim());
+  }
+  const rightIngredients = new Set(
+    right.ingredients.filter((text) => text.trim()).map(ingredientKey),
+  );
+  // Keep the first recorded spelling as well as the source's order.
+  const sharedIngredients =
+    leftIngredients.size && rightIngredients.size
+      ? [...leftIngredients].filter(([key]) => rightIngredients.has(key)).map(([, text]) => text)
+      : null;
+  return {
+    left,
+    right,
+    sharedIngredients,
+    rows: Object.entries(leftFacts).map(([label, value]) => ({
+      label,
+      left: value,
+      right: rightFacts[label]!,
+    })),
+  };
+}

+export async function shareReviewedAppointmentBrief(
+  review: ReviewedAppointmentBrief,
+  assertReviewCurrent: () => void,
+): Promise<boolean> {
+  if (review.selection.includePhoto) throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');

+      if (!review.diaryAccess) throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+      review.diaryAccess.assertCurrent();
+    }
+  };
+  assertCurrent();
+  return runHealthDataOperation(review.lease.ownerUserId, async (lease) => {
+    assertCurrent();
+    const proof = await revalidateAppointmentBrief(review, assertCurrent);
+    lease.assertCurrent(); assertCurrent();
+    const result = await dispatchWithBriefSourceProof(proof, review.lease, assertCurrent, () => Share.share({
+      title: 'My factual appointment brief',
+      message: review.text,
+    }));
+    lease.assertCurrent();
+    assertCurrent();
+    return result.action === Share.sharedAction;
+  });

+/** Common preflight for the text and explicitly selected image export paths. */
+export async function revalidateAppointmentBrief(review: ReviewedAppointmentBrief, assertReviewCurrent: () => void): Promise<BriefSourceProof> {

+  assertCurrent();

+  assertCurrent();

+    const products = await loadShelf();
+    assertCurrent();

+    assertCurrent();
+    if (review.useSource && (!review.selection.recordIds?.includes(review.useSource.id) || !matchBriefUseSource(records, review.useSource))) {
+      throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+    }
+    if (buildAppointmentBrief(products, review.selection, review.preparedOn, records) !== review.text) {
+      throw new Error('APPOINTMENT_BRIEF_INVALID_OR_STALE');
+    }
+  } catch (error) {
+    assertCurrent();
+    if (error instanceof Error && error.message === 'APPOINTMENT_BRIEF_INVALID_OR_STALE') throw error;
+    throw new Error('APPOINTMENT_BRIEF_SOURCE_UNAVAILABLE');
+  }
+  assertCurrent();
+  return proof;
