+export function createOwnerPortal(request: typeof fetch, render: (state: PortalState) => void, download: (text: string) => void,

+        const refusal = response.headers.get('X-OnSkin-Refusal');
+        if (expected === 'accepted' && retryAccept && response.status === 422 &&
+            response.headers.get('Cache-Control') === 'no-store' &&
+            response.headers.get('Content-Type')?.startsWith('application/json') &&
+            (refusal === 'capacity' || refusal === 'coverage')) {
+          // Keep the exact command: an earlier response may have been lost.
+          // A later refusal cannot prove that an earlier attempt did not save.
+          show({ kind: 'blocked-accept', reason: refusal }); return;
+        }
+        if ([400, 401, 403, 409].includes(response.status)) { retryEnd = null; retryDecline = null; retryAccept = null; retryReceive = null; context = null; }
+        throw new Error('OWNER_PORTAL_UNAVAILABLE');
+      }
+      if (!response.headers.get('Content-Type')?.startsWith('application/json') || response.headers.get('Cache-Control') !== 'no-store') throw new Error('OWNER_PORTAL_UNAVAILABLE');

+      }

+          value.plans.every(validPlan) && value.plans.every(plan => plan.latestResponse !== null) && new Set(value.plans.map(plan => plan.planId)).size === value.plans.length) next = { kind: 'index', plans: value.plans };
+      else if (expected === 'connection' && exact(value, 'connection,context,kind') && (value.connection === null || validConnection(value.connection))) next = { kind: 'connection', connection: value.connection };
+      else if (expected === 'report' && exact(value, 'context,kind,text') && typeof value.text === 'string' && value.text.length > 0 && value.text.length <= 3_000_000) next = { kind: 'report', text: value.text };
+      else if (expected === 'decline-review' && planReview && exact(value, 'context,expectedResponseHash,kind,text') &&
+          (value.expectedResponseHash === null || hash(value.expectedResponseHash)) && typeof value.text === 'string' && value.text.length > 0 && value.text.length <= 3_000_000)
+        next = { kind: 'confirm-decline', plan: planReview, text: value.text, expectedResponseHash: value.expectedResponseHash };
+      else if (expected === 'ended' && exact(value, 'context,kind')) next = { kind: 'ended' };
+      else if (expected === 'declined' && exact(value, 'context,kind')) next = { kind: 'declined' };
+      else if (expected === 'accepted' && exact(value, 'context,kind')) next = { kind: 'accepted' };
+      else if (expected === 'received' && exact(value, 'context,kind')) next = { kind: 'received' };
+      else if (expected === 'incoming' && exact(value, 'context,kind,plans,workspaceId') && id(value.workspaceId) && Array.isArray(value.plans) && value.plans.length <= 100 &&
+          value.plans.every(validPlan) && value.plans.every(plan => plan.latestResponse === null) && new Set(value.plans.map(plan => plan.planId)).size === value.plans.length) next = { kind: 'incoming', workspaceId: value.workspaceId, plans: value.plans };
+      else if (expected === 'incoming-review' && planReview && exact(value, 'context,kind,text,workspaceId') && id(value.workspaceId) && value.workspaceId === input.workspaceId &&

+        next = { kind: 'confirm-accept', plan: planReview, text: value.text, workspaceId: value.workspaceId, expectedCapacityRevision: value.expectedCapacityRevision, expectedResponseHash: value.expectedResponseHash };
+      }

+      if (admitted === generation) show(next);
+    } catch {
+      if (admitted === generation) show(retryReceive ? { kind: 'failed-receive' } : retryAccept ? { kind: 'failed-accept' } : retryDecline ? { kind: 'failed-decline' } : { kind: 'failed', retryEnd: retryEnd !== null });
+    } finally { clearTimeout(timeout); if (admitted === generation) pending = null; }
+  }
+  return {

+    whenCurrent(expected: PortalState, action: () => unknown) { if (state === expected) return action(); },
+    clear() { generation++; pending?.abort(); pending = null; context = null; retryEnd = null; retryDecline = null; retryAccept = null; retryReceive = null; show({ kind: 'idle' }); },
+    check() { if (pending) return; retryEnd = null; retryDecline = null; retryAccept = null; retryReceive = null; return run({ kind: 'index', context }, 'index'); },
+    connection() { if (!context || pending) return; retryEnd = null; retryDecline = null; retryAccept = null; retryReceive = null; return run({ kind: 'connection', context }, 'connection'); },

+      return run(retryEnd, 'ended');
+    },
+    retryEnd() { if (pending || !retryEnd) return; return run(retryEnd, 'ended'); },
+    reviewDecline(plan: PlanSummary) {
+      if (!context || pending || state.kind !== 'index' || !state.plans.some(item => item.planId === plan.planId && item.planHash === plan.planHash)) return;
+      const selected = state.plans.find(item => item.planId === plan.planId)!;

+    },
+    decline() {
+      if (!context || pending || state.kind !== 'confirm-decline') return;
+      const { planId, planHash } = state.plan;
+      retryDecline = { kind: 'decline', context, command: { receiptId: newId(), planId, planHash, expectedResponseHash: state.expectedResponseHash, confirmed: true } };
+      return run(retryDecline, 'declined');
+    },
+    retryDecline() { if (pending || !retryDecline || state.kind !== 'failed-decline') return; return run(retryDecline, 'declined'); },
+    reviewAcceptance(plan: PlanSummary) {
+      if (!context || pending || state.kind !== 'index' || !state.plans.some(item => item.planId === plan.planId && item.planHash === plan.planHash)) return;
+      const selected = state.plans.find(item => item.planId === plan.planId)!;
+      return run({ kind: 'accept-review', context, planId: selected.planId, planHash: selected.planHash }, 'accept-review', false, Object.freeze({ ...selected }));
+    },
+    accept() {
+      if (!context || pending || state.kind !== 'confirm-accept') return;
+      const { planId, planHash } = state.plan;
+      retryAccept = { kind: 'accept', context, workspaceId: state.workspaceId, command: {
+        receiptId: newId(), capacityId: newId(), expectedCapacityRevision: state.expectedCapacityRevision, expectedResponseHash: state.expectedResponseHash, planId, planHash, confirmed: true,
+      } };
+      return run(retryAccept, 'accepted');
+    },
+    retryAcceptance() { if (pending || !retryAccept || !['failed-accept', 'blocked-accept'].includes(state.kind)) return; return run(retryAccept, 'accepted'); },
+    incoming() { if (!context || pending || state.kind !== 'index') return; return run({ kind: 'incoming', context }, 'incoming'); },
+    reviewReceive(plan: PlanSummary) {
+      if (!context || pending || state.kind !== 'incoming' || !state.plans.some(item => item.planId === plan.planId && item.planHash === plan.planHash)) return;
+      const selected = state.plans.find(item => item.planId === plan.planId)!;
+      // workspaceId binds the reviewed list; server authority still comes from
+      // the current grant and must match this expected workspace.
+      return run({ kind: 'incoming-review', context, planId: selected.planId, planHash: selected.planHash, workspaceId: state.workspaceId }, 'incoming-review', false, Object.freeze({ ...selected }));
+    },
+    receive() {
+      if (!context || pending || state.kind !== 'confirm-receive') return;
+      const { planId, planHash } = state.plan;
+      retryReceive = { kind: 'receive', context, workspaceId: state.workspaceId, command: { receiptId: newId(), planId, planHash, confirmed: true } };
+      return run(retryReceive, 'received');
+    },
+    retryReceipt() { if (pending || !retryReceive || state.kind !== 'failed-receive') return; return run(retryReceive, 'received'); },
+  };

+type Leaf = Readonly<{ id: string; receiptHash: string }>;
+type Branch = Readonly<{ children: Readonly<Record<string, string>> }>;
+type Node = Leaf | Branch;
+const changed = (): never => { throw new ProfessionalAccessError('changed'); };
+const unavailable = (): never => { throw new ProfessionalAccessError('unavailable'); };
+const pathFor = (id: string) => recordHash(['owner-routine-index-id-v1', id]);
+
+/** A sparse radix tree with one committed leaf per original ID. Only the path
+ * to a changed leaf is rewritten. The encrypted head authenticates the root. */
+export class OwnerRoutineIndex {
+  constructor(private readonly tx: EncryptedTransaction, private readonly namespace: string) {}
+
+  private async read(path: string, hash: string): Promise<Node> {
+    const key = `${this.namespace}:${path}`;
+    const node = await this.tx.read(key) as Node | null;
+    if (node === null) return unavailable();
+    if (await recordHash(['owner-routine-index-node-v1', key, node]) !== hash) return changed();
+    return node;
+  }
+
+  private async write(path: string, node: Node): Promise<string> {
+    const key = `${this.namespace}:${path}`;
+    await this.tx.write(key, node);
+    return recordHash(['owner-routine-index-node-v1', key, node]);
+  }
+
+  async find(root: string | null, id: string): Promise<string | null> {
+    const path = await pathFor(id);
+    let hash = root;
+    for (let depth = 0; hash !== null && depth <= path.length; depth++) {
+      const node = await this.read(path.slice(0, depth), hash);
+      if ('id' in node) return node.id === id ? node.receiptHash : null;
+      hash = node.children[path[depth]!] ?? null;
+    }
+    return hash === null ? null : changed();
+  }
+
+  async insert(root: string | null, leaf: Leaf): Promise<string> {
+    const path = await pathFor(leaf.id);
+    const insert = async (prefix: string, hash: string | null): Promise<string> => {
+      if (hash === null) return this.write(prefix, leaf);
+      const node = await this.read(prefix, hash);
+      if (prefix.length >= path.length) return changed();
+      const next = path[prefix.length]!;
+      if ('id' in node) {
+        const oldPath = await pathFor(node.id);
+        if (oldPath === path || !oldPath.startsWith(prefix)) return changed();
+        let split = prefix.length;
+        while (path[split] === oldPath[split]) split++;
+        let joined = await this.write(path.slice(0, split), { children: {
+          [oldPath[split]!]: await this.write(oldPath.slice(0, split + 1), node),
+          [path[split]!]: await this.write(path.slice(0, split + 1), leaf),
+        } });
+        for (let depth = split - 1; depth >= prefix.length; depth--) {
+          joined = await this.write(path.slice(0, depth), { children: { [path[depth]!]: joined } });
+        }
+        return joined;
+      }
+      const children = { ...node.children, [next]: await insert(prefix + next, node.children[next] ?? null) };
+      return canonical(children) === canonical(node.children) ? hash : this.write(prefix, { children });
+    };
+    return insert('', root);
+  }
+
+  /** Enumerate commitments in digest order so each inventory page needs only
+   * its cursor path and the bounded next segment. */
+  async page(root: string | null, after: string | null, limit: number): Promise<{ leaves: Leaf[]; next: string | null }> {
+    const leaves: Leaf[] = [];
+    let last: string | null = null;
+    let more = false;
+    const visit = async (path: string, hash: string): Promise<void> => {
+      if (more || (after !== null && path < after.slice(0, path.length))) return;
+      const node = await this.read(path, hash);
+      if ('id' in node) {
+        const digest = await pathFor(node.id);
+        if (after !== null && digest <= after) return;
+        if (leaves.length === limit) { more = true; return; }
+        leaves.push(node);
+        last = digest;
+        return;
+      }
+      for (const digit of Object.keys(node.children).sort()) await visit(path + digit, node.children[digit]!);
+    };
+    if (root !== null) await visit('', root);
+    return { leaves, next: more ? last : null };
+  }
+}

+export function decodeTemplateCommand(value: unknown): TemplateCommand {
+  if (!value || typeof value !== 'object' || Array.isArray(value)) return fail();
+  const c = value as TemplateCommand;
+  const expected = c.kind === 'retire' ? 'confirmed,expectedHeadHash,id,kind,templateId' : 'am,confirmed,expectedHeadHash,id,kind,pm,templateId,title';
+  if (Object.keys(c).sort().join(',') !== expected || !templateId(c.id) || !templateId(c.templateId) || c.confirmed !== true ||
+      (c.expectedHeadHash !== null && !templateHash(c.expectedHeadHash)) || !['revise', 'retire'].includes(c.kind)) return fail();
+  if (c.kind === 'revise' && (!templateTitle(c.title) || !templateRoles(c.am) || !templateRoles(c.pm) || !c.am.length && !c.pm.length)) return fail();

+export async function decodePracticeTemplates(value: unknown, workspaceId: string, now = 8.64e15): Promise<readonly PracticeTemplate[]> {
+  const copied: unknown = structuredClone(value);
+  if (!identifier(workspaceId) || !Array.isArray(copied) || copied.length > 600 || !Number.isFinite(now) || now < 0) return fail();
+  const result: PracticeTemplate[] = [], heads = new Map<string, PracticeTemplate>(), ids = new Set<string>();
+  for (const t of copied as PracticeTemplate[]) {
+    if (!t || Object.keys(t).sort().join(',') !== 'actorId,am,contentHash,createdAt,id,parentHash,pm,revision,schemaVersion,status,templateId,title,workspaceId' ||
+        t.schemaVersion !== 1 || !templateId(t.id) || !templateId(t.templateId) || ids.has(t.id) || t.workspaceId !== workspaceId ||
+        !identifier(t.actorId) || !templateTitle(t.title) || !templateRoles(t.am) || !templateRoles(t.pm) || !t.am.length && !t.pm.length ||
+        !['active', 'retired'].includes(t.status) || !Number.isSafeInteger(t.revision) || t.revision < 1 || t.revision > (t.status === 'retired' ? 101 : 100) ||
+        !Number.isSafeInteger(t.createdAt) || t.createdAt < (result.at(-1)?.createdAt ?? 0) || t.createdAt > Math.min(now, 8.64e15)) return fail();
+    const parent = heads.get(t.templateId);

+export async function changePracticeTemplate(value: unknown, workspaceId: string, actorId: string, input: unknown, now: number) {
+  const command = decodeTemplateCommand(input);
