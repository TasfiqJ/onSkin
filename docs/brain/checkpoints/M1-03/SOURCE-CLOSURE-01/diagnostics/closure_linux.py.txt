def run(name,command,env=None):
    output=PRIVATE/(name+'.log');receipt=PRIVATE/(name+'.receipt.json')
    assert not output.exists() and not receipt.exists()
    commit=subprocess.check_output(['git','-C',str(ROOT),'rev-parse','HEAD']).decode().strip()
    tree=subprocess.check_output(['git','-C',str(ROOT),'rev-parse','HEAD^{tree}']).decode().strip()
    start=datetime.datetime.now(datetime.timezone.utc).isoformat()
    with output.open('wb') as out:result=subprocess.run(command,cwd=ROOT,stdout=out,stderr=subprocess.STDOUT,env=env)
    raw=output.read_bytes();text=raw.decode(errors='replace');cases=re.search(r'Ran (\d+) tests?',text);counts=None
    if cases:
        end=text[text.rfind('\nRan '):];failed=sum(int(n) for n in re.findall(r'(?:failures|errors)=(\d+)',end));skipped=sum(int(n) for n in re.findall(r'skipped=(\d+)',end))
        counts=dict(passed=int(cases[1])-failed-skipped,failed=failed,skipped=skipped)
    value=dict(schema_version=1,kind='LOCAL_EXECUTION',claim_origin='CODEX_REPORTED_EXECUTION',command=' '.join(command),
        working_directory=str(ROOT),environment='WSL2 onskin-r6-verification; Ubuntu22.04 Linux x86_64; Python3.12.3',
        source_commit=commit,source_tree=tree,status='PASS' if result.returncode==0 else 'FAIL',exit_code=result.returncode,
        started_at=start,finished_at=datetime.datetime.now(datetime.timezone.utc).isoformat(),admission='NONE',
        output=dict(path='docs/brain/checkpoints/M1-03/SOURCE-CLOSURE-01/evidence/'+output.name,sha256=hashlib.sha256(raw).hexdigest()))
    if counts is not None:value['counts']=counts
    receipt.write_text(json.dumps(value,indent=2)+'\n')
    print(name,result.returncode,counts,text[-700:],flush=True)
    return result.returncode
