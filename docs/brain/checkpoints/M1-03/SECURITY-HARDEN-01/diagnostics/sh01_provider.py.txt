"""Read-only provider metadata/artifacts. Credentials never leave memory or print."""
import datetime,hashlib,io,json,pathlib,re,subprocess,sys,urllib.request,urllib.error,zipfile
WS=pathlib.Path(__file__).parent;REPO=WS/'overnight-delivery-clean'
PRIVATE=WS/'sh01-provider-private';PRIVATE.mkdir(exist_ok=True)
raw=subprocess.check_output(['git','credential','fill'],input=b'protocol=https\nhost=github.com\n\n',cwd=REPO,stderr=subprocess.DEVNULL)
credential=dict(line.split('=',1) for line in raw.decode().splitlines() if '=' in line)
token=credential['password']
def get(path,binary=False):
    request=urllib.request.Request('https://api.github.com/repos/TasfiqJ/onSkin/'+path,
        headers={'Authorization':'Bearer '+token,'Accept':'application/vnd.github+json','X-GitHub-Api-Version':'2022-11-28'})
    class NoRedirect(urllib.request.HTTPRedirectHandler):
        def redirect_request(self,*args,**kwargs):return None
    try:
        with urllib.request.build_opener(NoRedirect).open(request,timeout=90) as response:data=response.read()
    except urllib.error.HTTPError as error:
        if error.code not in (301,302,303,307,308):raise
        # Never forward GitHub authorization to signed object-store URLs.
        with urllib.request.urlopen(error.headers['Location'],timeout=90) as response:data=response.read()
    return data if binary else json.loads(data)
sha=sys.argv[1]
pr=get('pulls/9')
runs=get('actions/runs?head_sha='+sha+'&per_page=100')['workflow_runs']
result=dict(observed_utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),source_commit=sha,
    evidence_class='AUTHENTICATED_PROVIDER_OBSERVED',pull_request={k:pr.get(k) for k in ['number','state','draft','merged','merge_commit_sha','html_url']},
    main_sha=pr['base']['sha'],stage_sha=pr['head']['sha'],runs=[])
for run in runs:
    row={k:run.get(k) for k in ['id','workflow_id','name','head_sha','head_branch','event','run_attempt','status','conclusion','created_at','updated_at','html_url']}
    jobs=get('actions/runs/'+str(run['id'])+'/jobs?per_page=100')['jobs']
    row['jobs']=[{k:j.get(k) for k in ['id','name','head_sha','status','conclusion','started_at','completed_at','html_url','steps']} for j in jobs]
    row['artifacts']=[{k:a.get(k) for k in ['id','name','size_in_bytes','expired','digest']} for a in get('actions/runs/'+str(run['id'])+'/artifacts?per_page=100')['artifacts']]
    if '--logs' in sys.argv:
        for job in row['jobs']:
            if job['status']!='completed' or job['conclusion']=='skipped':continue
            body=get('actions/jobs/'+str(job['id'])+'/logs',True)
            (PRIVATE/(str(job['id'])+'.log')).write_bytes(body)
            text=body.decode(errors='replace')
            job['log_sha256']=hashlib.sha256(body).hexdigest()
            job['test_counts']=re.findall(r'Ran (\d+) tests?',text)
            job['failure_codes']=sorted(set(re.findall(r'\b(?:SECRET|SOURCE|SEMGREP)_[A-Z_]+\b',text)))
            job['checkout_hash_lines']=re.findall(r'(?m)^\S+ ([0-9a-f]{40})\s*$',text)
    if '--download' in sys.argv:
        for artifact in row['artifacts']:
            name=artifact['name'];body=get('actions/artifacts/'+str(artifact['id'])+'/zip',True)
            (PRIVATE/(str(artifact['id'])+'.zip')).write_bytes(body)
            with zipfile.ZipFile(io.BytesIO(body)) as z:
                artifact['zip_sha256']=hashlib.sha256(body).hexdigest();artifact['members']=z.namelist()
                for member in z.namelist():
                    assert not member.startswith(('/', '\\')) and '..' not in pathlib.PurePosixPath(member).parts
                z.extractall(PRIVATE/str(artifact['id']))
    result['runs'].append(row)
out=WS/('sh01-provider-'+sha+'.json');out.write_bytes((json.dumps(result,indent=2)+'\n').encode())
print(json.dumps(dict(source=sha,pr_draft=result['pull_request']['draft'],pr_merged=result['pull_request']['merged'],
    runs=[dict(id=r['id'],name=r['name'],status=r['status'],conclusion=r['conclusion'],jobs=[dict(name=j['name'],conclusion=j['conclusion']) for j in r['jobs']]) for r in result['runs']])))
