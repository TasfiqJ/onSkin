def run(name,cmd,cwd=ROOT,env=None):
    original=name;attempt=1
    while (PRIVATE/(name+'.json')).exists():attempt+=1;name=original+'-'+str(attempt)
    start=datetime.datetime.now(datetime.timezone.utc).isoformat()
    with (PRIVATE/(name+'.log')).open('wb') as out:
        p=subprocess.run(cmd,cwd=cwd,stdout=out,stderr=subprocess.STDOUT,env=env)
    raw=(PRIVATE/(name+'.log')).read_bytes()
    receipt=dict(command=cmd,started=start,finished=datetime.datetime.now(datetime.timezone.utc).isoformat(),raw_exit=p.returncode,
        output_sha256=hashlib.sha256(raw).hexdigest(),output_bytes=len(raw),environment_overrides={k:v for k,v in (env or {}).items() if k in {'ONSKIN_TEST_CANDIDATE','TURBO_FORCE'}},
        source_commit=subprocess.check_output(['git','-C',str(ROOT),'rev-parse','HEAD']).decode().strip(),
        source_tree=subprocess.check_output(['git','-C',str(ROOT),'rev-parse','HEAD^{tree}']).decode().strip(),admission='NONE')
    (PRIVATE/(name+'.json')).write_text(json.dumps(receipt,indent=2)+'\n')
    print(name,p.returncode,raw[-1200:].decode(errors='replace'),flush=True)
    return p.returncode
