def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def git(*args):
    return subprocess.check_output(['git', '-C', str(REPO), '-c', 'gc.auto=0', *args],
                                   env=dict(os.environ, GIT_NO_LAZY_FETCH='1'))

def save(name, value):
    target = OUT / name
    with target.open('xb') as stream:
        stream.write((json.dumps(value, indent=2, ensure_ascii=True) + '\n').encode())

def run(name, argv, cwd=REPO, timeout=1800):
    assert git('rev-parse', 'HEAD').decode().strip() == HEAD
    started = datetime.datetime.now(datetime.timezone.utc).isoformat()
    clock = time.monotonic()
    logfile = OUT / (name + '.log')
    with logfile.open('xb') as output:
        try:
            process = subprocess.run(argv, cwd=cwd, stdout=output, stderr=subprocess.STDOUT,
                                     timeout=timeout, env=dict(os.environ, GIT_NO_LAZY_FETCH='1'))
            code, status = process.returncode, 'PASS' if process.returncode == 0 else 'FAIL'
        except subprocess.TimeoutExpired:
            code, status = None, 'TIMEOUT'
    raw = logfile.read_bytes()
    value = dict(schema_version=1, kind='RECONCILIATION_EXECUTION', admission='NONE',
                 evidence_class='INDEPENDENTLY_REPRODUCED', argv=argv, working_directory=str(cwd),
                 started_utc=started, ended_utc=datetime.datetime.now(datetime.timezone.utc).isoformat(),
                 seconds=round(time.monotonic()-clock, 3), host=platform.platform(),
                 python=sys.version, source_commit=HEAD, source_tree=git('rev-parse', HEAD+'^{tree}').decode().strip(),
                 status=status, exit_code=code, log=logfile.name, log_bytes=len(raw), log_sha256=sha(raw))
    save(name+'.receipt.json', value)
    print(json.dumps(value), flush=True)
    print(raw.decode('utf-8', errors='replace')[-2200:], flush=True)
    return code
